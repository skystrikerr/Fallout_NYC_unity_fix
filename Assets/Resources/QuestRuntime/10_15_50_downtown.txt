  // ------------------------------------------------------------------
  // Events & triggers
  // ------------------------------------------------------------------
  /** living security robots at the Met (location spawns + the quest's own) */
  function metBots() {
    if (G.isStub('actors') || !G.actors.list) return [];
    return G.actors.list().filter((a) => a && !a.dead && !a.removed && (a.type === 'protectron' || a.type === 'mister_handy') && a.id !== 'met_curator'
      && Math.hypot(a.pos.x - MET.x, a.pos.z - MET.z) < 110);
  }
  /**
   * Completability net: the robots are the Met's location spawns. If the player is at the museum with the job
   * open and fewer robots exist than are still needed (spawns culled, respawn timer, an older save), the
   * museum's security wakes two more at reachable spots on the Met's steps.
   */
  let metCheckT = 0;
  L.hook('update', function (dt) {
    metCheckT += dt; if (metCheckT < 3) return; metCheckT = 0;
    if (Q.stage('sq_exhibit') !== 'security' || L.flag('met_closed')) return;
    const P = G.player.pos; if (Math.hypot(P.x - MET.x, P.z - MET.z) > 70 || P.x >= G.LAYOUT.INTERIOR_X_MIN) return;
    const need = 2 - Math.min(2, L.counter('met_bots'));
    if (metBots().length >= need) return;
    const n = Number(L.flag('met_bot_waves')) || 0; if (n >= 3) return;
    L.incFlag('met_bot_waves');
    const at = L.anchor('met', ['entrance', 'steps', 'curator'], { x: MET.x + 36, z: MET.z });
    for (let i = 0; i < need; i++) {
      const s = L.freeSpot(at.x + (i ? 3 : -3), at.z + 2, at.y, { maxR: 10 });
      L.spawnNPC({ id: 'met_secbot' + n + '_' + i, type: 'protectron', name: 'Museum Security Protectron', faction: 'robots', hostile: true, level: 5, noRespawn: true,
        pos: { x: s.x, y: s.y, z: s.z }, ai: 'guard', home: { x: s.x, z: s.z, r: 8 },
        barks: ['PATRON DETECTED. THE MUSEUM IS CLOSED.', 'PLEASE DO NOT TOUCH THE EXHIBITS. OR THE ROBOTS.', 'EXPELLING ART THIEF.'] });
    }
  });
  /**
   * Same net for Shell Shock: the Battery's mirelurks are location spawns (3 or so, 3-day respawn). With the
   * job open, the player on the Battery shore, fewer than 4 killed and none left alive nearby, a few more crawl
   * out of the harbour at the nests (max 3 waves) — before, the marker pointed at empty nests for days.
   */
  let shellCheckT = 0;
  L.hook('update', function (dt) {
    shellCheckT += dt; if (shellCheckT < 3) return; shellCheckT = 0;
    if (Q.stage('sq_shellshock') !== 'hunt' || !Q.isActive('sq_shellshock') || L.flag('shell_net')) return;
    const P = G.player.pos; if (Math.hypot(P.x + 40, P.z - 950) > 90 || P.x >= G.LAYOUT.INTERIOR_X_MIN) return;
    const need = 4 - Math.min(4, L.counter('mirelurks')); if (need <= 0) return;
    const live = G.actors.list().filter((a) => a && !a.dead && !a.removed && /^mirelurk/.test(a.type) && Math.hypot(a.pos.x + 40, a.pos.z - 950) < 160);
    if (live.length) return;
    const n = Number(L.flag('shell_waves')) || 0; if (n >= 3) return;
    L.incFlag('shell_waves');
    for (let i = 0; i < Math.min(need, 3); i++) {
      const s = L.freeSpot(-40 + (i - 1) * 5, 956, undefined, { maxR: 14 }); if (!s || s.forced) continue;
      L.spawnNPC({ id: 'bp_lurk' + n + '_' + i, type: 'mirelurk', name: 'Mirelurk', faction: 'creatures', hostile: true, level: 4, noRespawn: true,
        pos: { x: s.x, y: s.y, z: s.z }, ai: 'guard', home: { x: s.x, z: s.z, r: 10 } });
    }
  });
  G.events.on('actorKilled', (a) => {
    if (!a) return;
    if (a.id === 'chr_dolores') { L.setFlag('dolores_dead'); if (Q.stage('sq_sniper') === 'find') Q.setStage('sq_sniper', 'report'); }
    if (a.id === 'bb_tony') { L.setFlag('tony_dead'); L.setFlag('bridge_resolved'); if (Q.stage('sq_bridge') === 'tony') Q.setStage('sq_bridge', 'report'); ['bb_knuckles', 'bb_deedee'].forEach((id) => L.hostile(id, true)); }
    if (a.type === 'mirelurk' || a.type === 'mirelurk_queen') {
      L.bump('mirelurks');
      if (Q.stage('sq_shellshock') === 'hunt') { if (L.counter('mirelurks') >= 4) { L.setFlag('shell_culled'); Q.setStage('sq_shellshock', 'report'); } else L.refresh('sq_shellshock'); }
    }
    if ((a.type === 'protectron' || a.type === 'mister_handy') && a.id !== 'met_curator' && Math.hypot(a.pos.x - MET.x, a.pos.z - MET.z) < 110) {
      L.bump('met_bots');
      if (Q.stage('sq_exhibit') === 'security') { if (L.counter('met_bots') >= 2) { L.setFlag('met_bots_destroyed'); Q.setStage('sq_exhibit', 'report'); } else L.refresh('sq_exhibit'); }
    }
    if (a.id === 'met_curator') L.setFlag('curator_dead');
  });
  G.events.on('itemAdded', (id) => {
    if (id === 'q_caldwell_strongbox' && (Q.stage('sq_caravan') === 'woolworth' || Q.stage('sq_caravan') === 'wreck')) Q.setStage('sq_caravan', 'return');
    if (id === 'q_caldwell_strongbox' && !Q.isStarted('sq_caravan')) Q.start('sq_caravan', { stage: 'return' });
  });
  L.trigger({ id: 'caravan_wreck', x: WRECK.x, z: WRECK.z, r: 14, cond: () => Q.stage('sq_caravan') === 'wreck',
    fire() {
      L.notify('The Caldwell caravan: a gutted pack-bot, a dead driver, and raider boot prints heading south down Broadway.');
      Q.setStage('sq_caravan', 'woolworth');
      if (!L.flag('pete_rescued')) L.later(4, () => { L.setFlag('pete_trail'); L.notify('...and a second trail: huge bare footprints, dragging something heavy north toward 34th Street.'); L.refresh('sq_caravan'); });
    } });
  // Tony's gang warns you when you step onto the bridge.
  L.trigger({ id: 'bridge_warn', x: 250, z: 637, r: 24, y: 24, dy: 5, cond: () => !L.flag('bridge_resolved') && L.alive('bb_tony'),
    fire() { L.say('bb_tony', 'Hey, HEY! Toll booth! You walk past this booth without payin\', you\'re walkin\' into the river!'); } });

  // ------------------------------------------------------------------
  // Chrysler service elevator (lobby ⇄ the sniper nest on the SW gargoyle, ~110 m up)
  // ------------------------------------------------------------------
  const CHR_LIFT = { lobby: { x: 316.2, y: 1.2, z: -202.5 }, lobbyOut: { x: 314.6, y: 0, z: -202.5, yaw: Math.PI / 2 } };
  function nestSpot() { return L.anchor('chrysler', ['sniper'], null); }
  function chryslerMarker(p, label) {
    const P = G.player.pos;
    if (!p || p.y < 20 || P.y > 90) return { x: p.x, z: p.z, y: p.y, label };
    if (Math.hypot(P.x - CHR_LIFT.lobby.x, P.z - CHR_LIFT.lobby.z) < 60) return { x: CHR_LIFT.lobby.x, z: CHR_LIFT.lobby.z, label: label + ' — Service Elevator' };
    return { x: 298, z: -207.5, label: label + ' (Chrysler lobby)' };
  }
  L.chryslerMarker = chryslerMarker;
  L.hook('world', function () {
    const nest = nestSpot();
    if (!nest || nest.y < 50) return;   // landmark without a crown nest: nothing to connect
    G.world.addDoor({ pos: L.V3(CHR_LIFT.lobby.x, CHR_LIFT.lobby.y, CHR_LIFT.lobby.z), label: 'Service Elevator — Crown', verb: 'Ride', radius: 1.2,
      target: { x: nest.x + 0.6, y: nest.y, z: nest.z - 0.6, yaw: Math.PI * 0.75 }, sound: 'door_open', locked: 0,
      onEnter: () => L.notify('The raiders\' rope-and-winch elevator creaks up seventy floors. You try not to look at the rope.') });
    G.world.addDoor({ pos: L.V3(nest.x + 1.6, nest.y + 1.1, nest.z - 1.4), label: 'Service Elevator — Lobby', verb: 'Ride', radius: 1.0,
      target: CHR_LIFT.lobbyOut, sound: 'door_open' });
  });

  // ------------------------------------------------------------------
  // World & NPCs
  // ------------------------------------------------------------------
  L.hook('world', function () {
    const lbMet = !G.isStub('landmarks') && G.landmarks.getAnchor ? G.landmarks.getAnchor('met', 'terminal') : null;
    if (lbMet && isFinite(lbMet.x)) L.world.term_met_security = { pos: L.V3(lbMet.x, (lbMet.y || 0) + 1.1, lbMet.z), external: true };
    else { const t = L.anchor('met', ['terminal', 'security', 'office', 'entrance'], { x: MET.x + 32, z: MET.z + 10 }); if (t) L.spawnTerminal('met_security', t, -Math.PI / 2); }
    const tf = L.anchor('flatiron', ['terminal', 'office'], { x: 133, z: 110 });
    if (tf) L.spawnTerminal('flatiron_post', tf, Math.PI);
    // Ossie's return boat on Liberty Island (static interactable).
    const dock = libertyDock();
    if (dock) {
      L.world.libertyBoat = G.world.addInteractable({
        pos: L.V3(dock.x, (dock.y || 0) + 1, dock.z), radius: 1.2, label: 'Ossie\'s Ferry (to Battery Park)', verb: 'Ride', priority: 1,
        onInteract() { const p = batteryPier(); L.sfx('door_open'); L.travel(p.x, p.y || 0, p.z, 0.6, () => L.notify('Ossie grumbles the whole way back.')); },
      });
    }
  });

  L.hook('spawn', function () {
    // ---- Chrysler sniper ----
    if (!L.flag('dolores_left')) {
      const recruited = L.flag('dolores_recruited');
      L.spawnNPC({ id: 'chr_dolores', type: 'human', name: 'Deadeye Dolores', outfit: recruited ? 'combat_armor' : 'raider', gender: 'f', faction: recruited ? 'settlers' : 'raiders',
        ai: 'guard', home: 'here', homeR: 2, hostile: false, level: 9, hp: 180, weapon: ['sniper_rifle', 'hunting_rifle'],
        at: recruited ? { site: 'grand_central', anchors: ['guard3', 'rooftop', 'entrance'], fallback: { x: 256, z: -186 } }
          : { site: 'chrysler', anchors: ['sniper', 'rooftop', 'crown', 'boss'], fallback: { x: CHRYSLER.x - 6, z: CHRYSLER.z + 22 } },
        dialogue: 'dlg_chr_dolores', inventory: [['q_note_dolores_diary', 1], ['ammo_308', 20]],
        barks: recruited ? ['Caravan\'s clear. I\'ve got eyes on 42nd.', 'Nobody touches the pack-bots on my watch.'] : ['Nice day for it.', 'Keep walking, blue.'] });
    }
    // ---- Brooklyn Bridge toll gang ----
    const bb = (anchors, dx, dz) => ({ site: 'brooklyn_bridge', anchors, fallback: { x: 178 + dx, z: 640 + dz } });
    {
      const hostile = !!L.flag('bridge_hostile');
      L.spawnNPC({ id: 'bb_tony', type: 'human', name: 'Tollbooth Tony', outfit: 'raider_heavy', gender: 'm', faction: 'raiders', ai: 'guard', home: 'here', homeR: 3,

  // ------------------------------------------------------------------
  // Ferry. landmarks_b moors a "Harbor Skiff" door at Battery Park: it is
  // Ossie's boat, so it only runs once Ossie has agreed to take you.
  // ------------------------------------------------------------------
  function ferryAllowed() {
    return L.has('q_ferry_token') || !!L.flag('ferry_paid') || !!L.flag('ossie_free_ride') || !!L.flag('ending_played') ||
      (L.flag('mq_faction') === 'bos' && Q.isActive('mq06'));
  }
  L.ferryAllowed = ferryAllowed;
  function gateSkiff() {
    for (const d of L.doors()) {
      const p = d.def && d.def.pos, t = d.def && d.def.target; if (!p || !t) continue;
      if (p.x > -200 && p.x < 150 && p.z > 900 && p.z < 1060 && Math.hypot(t.x - G.LAYOUT.LIBERTY.x, t.z - G.LAYOUT.LIBERTY.z) < 80 && !d._questGated) {
        d._questGated = true;
        const prev = d.enabled;
        d.enabled = () => ferryAllowed() && (!prev || prev());
        // A stand-in prompt so the player knows who to talk to.
        G.world.addInteractable({ pos: p, radius: d.radius || 1.6, label: 'Harbor Skiff (talk to Ossie)', verb: 'Ride', priority: 0.5,
          enabled: () => !ferryAllowed(),
          onInteract() { L.say('bp_ossie', 'Hey! HEY! That\'s my boat! Hundred caps, then you ride.'); L.sfx('door_locked'); } });
      }
    }
  }
  L.hook('world', gateSkiff);

  // Ferry travel
  // ------------------------------------------------------------------
  function ferryToLiberty() {
    const d = libertyDock();
    L.sfx('door_open');
    L.travel(d.x, d.y || 0, d.z, Math.atan2(-(G.LAYOUT.LIBERTY.x - d.x), -(G.LAYOUT.LIBERTY.z - d.z)), () => {
      L.notify('The ferry noses up against Liberty Island\'s crumbling pier. Ossie is already turning back.');
      G.events.emit('ferryArrived', 'liberty');
    });
  }
  L.ferryToLiberty = ferryToLiberty;

  // ------------------------------------------------------------------
  // "Shell Shock" alternative: the pre-war shark-net generator on the Battery pier
  // (Repair 50 or Science 65). A small procedural machine: 4 meshes, shared materials.
  // ------------------------------------------------------------------
  let netMats = null;
  function netMaterials() {
    if (netMats) return netMats;
    const T = window.THREE;
    netMats = {
      body: new T.MeshStandardMaterial({ color: 0x5d6b62, roughness: 0.8, metalness: 0.55 }),
      rust: new T.MeshStandardMaterial({ color: 0x7a4a2a, roughness: 0.95, metalness: 0.3 }),
      lamp: new T.MeshStandardMaterial({ color: 0x331a10, emissive: 0xff5a1a, emissiveIntensity: 0.4, roughness: 0.4 }),
    };
    for (const k in netMats) netMats[k].userData.shared = true;
    return netMats;
  }
  function netLamp(on) {
    const m = netMaterials().lamp;
    m.color.setHex(on ? 0x103318 : 0x331a10); m.emissive.setHex(on ? 0x39ff6a : 0xff5a1a); m.emissiveIntensity = on ? 1.4 : 0.4;
  }
  function repairNet() {
    if (L.flag('shell_net')) { L.notify('The shark-net generator hums contentedly. Out on the water, something with too many legs changes its mind.'); return; }
    if (!Q.isActive('sq_shellshock') || Q.stage('sq_shellshock') !== 'hunt') { L.notify('A pre-war harbour shark-net generator, "PROPERTY OF THE PORT AUTHORITY". Dead as a doornail.'); return; }
    const rep = L.skill('repair'), sci = L.skill('science');
    if (rep < 50 && sci < 65) {
      L.notify('[Repair 50 / Science 65] The windings are corroded solid. You can\'t make sense of it — yet.'); L.sfx('ui_error'); return;
    }
    L.setFlag('shell_net'); L.setFlag('ossie_net_known');
    netLamp(true);
    L.sfx('hum_electric', { volume: 0.8 });
    if (G.player.shake) G.player.shake(0.15, 0.6);
    L.xp(75, rep >= 50 ? 'Repair: Harbour defence' : 'Science: Harbour defence');
    L.notify('You scrape the contacts, bridge the dead relay and throw the lever. Out in the harbour the old net lights up in a line of blue sparks, and the water starts to boil with retreating mirelurks.');
    Q.setStage('sq_shellshock', 'report');
    L.news('Harbour news: the old Port Authority shark nets off Battery Park are LIVE again for the first time since the war. Ossie says the crabs are furious. Ossie says that\'s the best part.');
  }
  L.hook('spawn', function () {
    const pier = batteryPier(); if (!pier) return;
    const p = L.freeSpot(pier.x + 6, pier.z - 3, pier.y, { maxR: 8, radius: 0.8 });
    if (!p) return;
    const T = window.THREE, M = netMaterials();
    const g = new T.Group();
    const body = new T.Mesh(new T.BoxGeometry(1.3, 1.1, 0.8), M.body); body.position.y = 0.55; g.add(body);
    const coil = new T.Mesh(new T.CylinderGeometry(0.26, 0.26, 0.9, 10), M.rust); coil.rotation.z = Math.PI / 2; coil.position.set(0, 1.28, 0); g.add(coil);
    const lever = new T.Mesh(new T.BoxGeometry(0.06, 0.45, 0.06), M.rust); lever.position.set(0.52, 1.25, 0.3); lever.rotation.z = -0.4; g.add(lever);
    const lamp = new T.Mesh(new T.SphereGeometry(0.07, 8, 6), M.lamp); lamp.position.set(-0.45, 1.16, 0.41); g.add(lamp);
    g.position.set(p.x, p.y, p.z);
    g.rotation.y = Math.atan2(-(pier.x - p.x), -(pier.z - p.z));
    G.scene.add(g);
    netLamp(!!L.flag('shell_net'));
    const collider = G.world.addCollider(p.x, p.z, 1.3, 0.8, p.y, p.y + 1.5, g.rotation.y, { tag: 'prop' });
    const interactable = G.world.addInteractable({ pos: L.V3(p.x, p.y + 1.1, p.z), radius: 1.0, label: 'Shark-Net Generator', verb: 'Repair', onInteract: repairNet });
    L.dyn({ object3d: g, collider, interactable, dispose() { g.traverse((o) => { if (o.geometry) o.geometry.dispose(); }); } });
    L.world.sharkNet = { x: p.x, z: p.z };
  });
