 *     the Opry, drinks at the Rad Room, benches, bunks; FO3-style ambient
 *     two-person conversations, greetings and hawker cries.
 *
 * Placement: ONLY the shared anchor names of the plan, resolved at newGame
 * (G.landmarks.getAnchor('times_square', n), then G.pois.getAnchor(

 * indoors (speakeasy) are skipped until their anchor exists.
 *
 * Performance: the routine scheduler ticks at 2 Hz and only while the player
 * is within ~260 m of the square or inside the Neon's interior cell; combat
 * already detaches actors beyond 175 m and hides them beyond 130 m.
 * ===================================================================== */

  function lookup(n) {
    let v = null;
    try { if (!G.isStub('landmarks') && typeof G.landmarks.getAnchor === 'function') v = G.landmarks.getAnchor('times_square', n); } catch (e) { v = null; }

  function rawAnchor(n) {
    const v = lookup(n);
    if (v) return { x: v.x, y: v.y || 0, z: v.z, yaw: v.yaw || 0, name: n };
    const al = ALIAS[n];
    if (al) {
      const b = lookup(al[0]);
      if (b) {
        const yaw = b.yaw || 0, lat = al[1] || 0, fwd = al[2] || 0;
        return { x: b.x + Math.cos(yaw) * lat + Math.sin(yaw) * fwd, y: b.y || 0, z: b.z - Math.sin(yaw) * lat + Math.cos(yaw) * fwd, yaw, name: n + '→' + al[0], alias: al[0] };
      }
    }
    return null;
  }

   * Resolve a standing spot. names: anchor candidates; fb: {near:[anchor names], dx, dz, yaw} or {x,z};
   * opts: {requireAnchor, maxR}. → {x,y,z,yaw,name,anchored,fallback} | null
   */
  function spot(names, fb, opts) {
    opts = opts || {};
    const key = [].concat(names || []).join(',') + '|' + JSON.stringify(fb || null) + '|' + (opts.requireAnchor ? 1 : 0);
    if (cache.has(key)) { const c = cache.get(key); return c ? Object.assign({}, c) : null; }
    let out = null;
    for (const n of [].concat(names || [])) {
      const a = rawAnchor(n); if (!a) continue;
      const s = L.freeSpot(a.x, a.z, a.y, { maxR: opts.maxR || 3 });
      out = s && !s.forced ? { x: s.x, y: s.y, z: s.z, yaw: a.yaw, name: a.name, anchored: true } : { x: a.x, y: a.y, z: a.z, yaw: a.yaw, name: a.name, anchored: true };
      break;
    }
    if (!out && fb && !opts.requireAnchor) {
      let base = null;
      if (fb.near) base = has(fb.near);
      if (!base) base = { x: fb.x !== undefined ? fb.x : TS.x, y: fb.y || 0, z: fb.z !== undefined ? fb.z : TS.z, yaw: 0 };
      // fallbacks are street-level: an elevated reference anchor (a roof, a catwalk) only gives x/z
      const by = (base.y || 0) > 3 ? 0 : base.y;
      const s = L.freeSpot(base.x + (fb.dx || 0), base.z + (fb.dz || 0), by, { maxR: 12 });
      if (s) out = { x: s.x, y: s.y, z: s.z, yaw: fb.yaw !== undefined ? fb.yaw : base.yaw || 0, name: 'fallback', fallback: true };
    }
    cache.set(key, out ? Object.assign({}, out) : null);
    return out;
  }

  function activityAt(routine, h) {

    // entries are in daily order starting at r[0][0]; find the last entry whose start <= h (wrapping)

  const town = { recs: new Map(), bunks: [], spawned: false };
  function spawnAll() {
    town.recs.clear(); town.bunks = series('bunk_', 16); town.bunkUsed = new Set(); town.skipped = [];
    for (const def of ROSTER) {
      try {
        // an NPC who IS a place's service (Annie's gallery, Dottie's beds) only exists once that place is built
        if (def.needsPlace && !placeOf(def.place)) { town.skipped.push(def.id + ' (no ' + def.place + ')'); continue; }
        const w = def.where || [[], null];
        const s = spot(w[0], w[1], { requireAnchor: !!def.requireAnchor || !w[1], maxR: def.maxR });
        if (!s) { town.skipped.push(def.id + ' (no anchor)'); continue; }
        const d = Object.assign({}, def);
        for (const k of ['role', 'where', 'routine', 'guard', 'sniper', 'gate', 'requireAnchor', 'maxR', 'homeR', 'place', 'needsPlace']) delete d[k];
        d.pos = { x: s.x, y: s.y, z: s.z };
        d.yaw = s.yaw || 0; d.lookYaw = s.yaw || 0;
        d.home = { x: s.x, z: s.z, r: def.homeR || (def.ai === 'wander' ? 6 : def.ai === 'patrol' ? 3 : 1.5) };
        if (def.ai === 'patrol') d.patrol = patrolRoute(def.id === 'neon_pat2');
        if (def.sniper) { d.holdPos = true; d.aggroRange = 150; }
        const a = L.spawnNPC(d);
        if (!a) continue;
        if (def.guard) { a.safeGuard = 'times_square'; a.neonGuard = true; }
        if (def.sniper) a.accuracy = 1.4;
        // crowd LOD budget: the square shows ~30 people at once; step down to the light body mesh beyond 12 m
        // (characters' default is ~34 m; faces keep their close-up LOD within 10 m either way)
        if (a.char && typeof a.char.lodDist === 'number') a.char.lodDist = Math.min(a.char.lodDist, def.sniper ? 40 : 12);
        const rec = { id: def.id, def, a, post: s, act: null, dest: null, leg: null, door: null, moveT: 0, lastX: s.x, lastZ: s.z, greetT: 0, hawkT: 0, pickT: 0 };
        const sm = /^market_stall_(\d+)(?:→|$)/.exec(s.name || ''); if (sm && def.role === 'merchant') rec.stall = Number(sm[1]);
        town.recs.set(def.id, rec);
      } catch (e) { G.reportError('quests:neon spawn ' + def.id, e); }
    }
    town.spawned = true;
  }
  function patrolRoute(reverse) {
    const pts = series('crowd_', 20).map((a) => ({ x: a.x, z: a.z }));
    let out = pts.length >= 4 ? (reverse ? pts.filter((p, i) => i % 2) : pts.filter((p, i) => !(i % 2))) : [];
    if (out.length < 3) {
      // fallback loop through the bow-tie plazas and past each gate (validated, off geometry)
      const raw = [['plaza', -2, 0], ['center', 0, 12], ['entrance_south', 3, -4], ['center', 14, 0], ['center', 0, -20], ['entrance', 4, 6], ['plaza', -6, -10]];
      out = [];
      for (const [n, dx, dz] of raw) { const s = spot([], { near: [n], dx, dz }); if (s) out.push({ x: s.x, z: s.z }); }
      if (reverse) out.reverse();
    }
    return out.length ? out : [{ x: TS.x - 8, z: TS.z }, { x: TS.x + 8, z: TS.z }];
  }

  function freeBunk(rec) {
    if (rec.bunk && !rec.bunk.gone) return rec.bunk;
    for (let i = 0; i < town.bunks.length; i++) if (!town.bunkUsed.has(i)) { town.bunkUsed.add(i); const b = town.bunks[i]; rec.bunk = { x: b.x, y: b.y, z: b.z, yaw: b.yaw, i, real: true }; return rec.bunk; }
    // every cot taken: a bedroll on the Bunkhouse floor between the cots (never out in the street)
    if (town.bunks.length) {
      for (let k = 0; k < 4; k++) {
        const b = pick(town.bunks);
        const s = L.freeSpot(b.x + (Math.random() - 0.5) * 2.4, b.z + (Math.random() - 0.5) * 2.4, b.y, { maxR: 2.5 });
        if (s && !s.forced) { rec.bunk = { x: s.x, y: s.y, z: s.z, yaw: Math.random() * 6.28, real: false, floor: true }; return rec.bunk; }
      }
    }
    // no bunk anchors (yet): sleep in a quiet corner of the bunkhouse / the NW shacks
    const s = spot([], { near: ['bunk_door', 'settler6', 'entrance'], dx: (Math.random() - 0.5) * 10, dz: 4 + Math.random() * 6 });
    rec.bunk = s ? { x: s.x, y: s.y, z: s.z, yaw: Math.random() * 6.28, real: false } : null;
    return rec.bunk;
  }
  function randomOf(names, fb) {
    const pool = [];
    for (const n of names) { const a = rawAnchor(n); if (a) pool.push(a); }
    const a = pool.length ? pick(pool) : null;
    if (a) { const s = L.freeSpot(a.x, a.z, a.y, { maxR: 3 }); if (s && !s.forced) return { x: s.x, y: s.y, z: s.z, yaw: a.yaw, real: true }; }
    if (!fb) return null;
    const s = spot([], { near: fb.near, dx: fb.dx + (Math.random() - 0.5) * (fb.spread || 8), dz: fb.dz + (Math.random() - 0.5) * (fb.spread || 8) });
    return s ? { x: s.x, y: s.y, z: s.z, yaw: Math.random() * 6.28, real: false } : null;
  }
  const nums = (b, n) => { const o = []; for (let i = 1; i <= n; i++) o.push(b + i); return o; };
  /** Which stall number each market merchant keeps (filled at spawn) — customers queue at stalls that are open. */
  const FOOD_STALLS = [5, 6, 12];
  /**
   * Customer spot in front of stall n: the exterior's `market_stall_<n>_customer` anchor when it exists,
   * else ~2 m in front of the merchant's spot (actor yaw faces (-sin, -cos)). Faces the counter.
   */
  function customerSpot(n, openOnly) {
    const m = rawAnchor('market_stall_' + n); if (!m) return null;
    if (openOnly) {
      const keeper = Array.from(town.recs.values()).find((r) => r.stall === n);
      if (keeper && !(keeper.act === 'work' && keeper.leg === 'at')) return null;
    }
    const c = rawAnchor('market_stall_' + n + '_customer');
    const jit = (Math.random() - 0.5) * 1.4;
    const bx = c ? c.x : m.x - Math.sin(m.yaw) * 2.1 + Math.cos(m.yaw) * jit, bz = c ? c.z : m.z - Math.cos(m.yaw) * 2.1 - Math.sin(m.yaw) * jit;
    const s = L.freeSpot(bx, bz, m.y || 0, { maxR: 1.6 });
    if (!s || s.forced) return null;
    return { x: s.x, y: s.y, z: s.z, yaw: Math.atan2(-(m.x - s.x), -(m.z - s.z)), real: true, stall: n };
  }
  function stallVisit(list, openOnly) {
    const order = list.slice().sort(() => Math.random() - 0.5);
    for (const n of order) { const s = customerSpot(n, openOnly); if (s) return s; }
    return null;
  }
  function destFor(rec, act) {
    const def = rec.def;
    switch (act) {
      case 'work': return Object.assign({}, rec.post, { mode: def.ai === 'wander' ? 'static' : def.ai, r: 1.2 });
      case 'perform': return Object.assign({}, rec.post, { mode: 'static', r: 1, perform: true });
      case 'sleep': { const b = freeBunk(rec); return b ? Object.assign({}, b, { mode: 'sit', r: 0.8, sleep: true }) : null; }
      case 'barracks': {
        // off-shift Neon Guards sleep in the barracks bunks (subway kiosk), not the Bunkhouse
        const d = randomOf(nums('barracks_bunk_', 4), null);
        if (d) return Object.assign(d, { mode: 'sit', r: 0.8, sleep: true });
        const b = freeBunk(rec); return b ? Object.assign({}, b, { mode: 'sit', r: 0.8, sleep: true }) : null;
      }
      case 'eat': {
        // queue at an OPEN food stall (Sato's noodles, Bev's grill, the hot-dog cart), face the counter, eat, move on
        const d = stallVisit(FOOD_STALLS, true) || stallVisit(FOOD_STALLS, false);
        if (d) return Object.assign(d, { mode: 'static', r: 1, repick: 45, eating: true });
        return destFor(rec, 'shop');
      }
      case 'shop': {
        // FO3 market life: browse at a stall counter (facing the trader) or mill about the plaza
        if (Math.random() < 0.65) { const d = stallVisit(nums('', 12).map(Number).filter((n) => FOOD_STALLS.indexOf(n) < 0), true); if (d) return Object.assign(d, { mode: 'static', r: 1, repick: 35 }); }
        const d = randomOf(nums('crowd_', 20), { near: ['merchant1', 'plaza', 'center'], dx: 2, dz: 2, spread: 14 }); return d && Object.assign(d, { mode: 'wander', r: 3.5, repick: 70 });
      }

  function observed(x, y, z) {
    const P = G.player; if (!P) return false;
    if (isInt(P.pos.x) !== isInt(x)) return false;
    const dx = x - P.pos.x, dz = z - P.pos.z, d = Math.hypot(dx, dz);
    if (d < 18) return true;
    if (d > 90) return false;
    const f = P.forward || { x: -Math.sin(P.yaw), z: -Math.cos(P.yaw) };
    return (dx * f.x + dz * f.z) / (d || 1) > 0.45;
  }
  function findDoor(fromPos, dest) {
    const wantIn = isInt(dest.x);
    let best = null, bd = 1e9;
    for (const d of L.doors()) {
      const p = d.def && d.def.pos, t = d.def && d.def.target; if (!p || !t) continue;
      if (isInt(p.x) === wantIn || isInt(t.x) !== wantIn) continue;
      const dd = Math.hypot(t.x - dest.x, t.z - dest.z) + Math.hypot(p.x - fromPos.x, p.z - fromPos.z);
      if (Math.hypot(t.x - dest.x, t.z - dest.z) > 90 || Math.hypot(p.x - fromPos.x, p.z - fromPos.z) > 160) continue;
      if (dd < bd) { bd = dd; best = { from: { x: p.x, y: p.y || 0, z: p.z }, to: { x: t.x, y: t.y || 0, z: t.z } }; }
    }
    return best;
  }

  // ------------------------------------------------------------------
  // Update: the crowd sleeps when the player is far away (performance cap)
  // ------------------------------------------------------------------
  let tickT = 0, lifeT = 0;
  L.hook('update', function (dt) {
    if (!town.spawned || !G.player || dt <= 0) return;
    const P = G.player.pos;
    const o = NEON_CELL();
    const near = isInt(P.x) ? (Math.abs(P.x - o.x) < 170 && Math.abs(P.z - o.z) < 170) : Math.hypot(P.x - TS.x, P.z - TS.z) < 260;
    if (!near) return;
    tickT += dt; lifeT += dt;
    if (tickT >= 0.5) {
      const step = tickT; tickT = 0;
      for (const rec of town.recs.values()) {
        if (!rec.def.routine || busy(rec.a)) continue;
        try { steer(rec, step); } catch (e) { G.reportError('quests:neon steer ' + rec.id, e); rec.def = Object.assign({}, rec.def, { routine: null }); }
      }
    }
    if (lifeT >= 1) { lifeT = 0; try { lifeTick(); } catch (e) { G.reportError('quests:neon life', e); } }
  });

  // Debug (dev console): G.quests.debug.neon()
  Q.debug.neon = function () {
    return Array.from(town.recs.values()).map((r) => ({ id: r.id, act: r.act, leg: r.leg, anchor: r.post && r.post.name, how: r.post ? (r.post.fallback ? 'fallback' : /→/.test(r.post.name || '') ? 'built-alias' : 'plan') : '?',
      x: +r.a.pos.x.toFixed(1), z: +r.a.pos.z.toFixed(1), state: r.a.state, mode: r.a.aiMode, dead: r.a.dead }));
  };
  /** Where the Neon stands: placement mix, skipped services, which places exist (and through which anchor). */
  Q.debug.neonPlaces = function () {
    const rows = Q.debug.neon(), count = { plan: 0, 'built-alias': 0, fallback: 0 };
    for (const r of rows) count[r.how] = (count[r.how] || 0) + 1;
    const places = {}; for (const k in PLACES) { const a = placeOf(k); places[k] = a ? a.name : null; }
    return { spawned: rows.length, placement: count, skipped: (town.skipped || []).slice(), places, fallbacks: rows.filter((r) => r.how === 'fallback').map((r) => r.id) };
  };
