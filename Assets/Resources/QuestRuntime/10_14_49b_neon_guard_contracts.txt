/* =====================================================================
 * FALLOUT: NEW YORK — QUESTS — 49b: THE NEON, GUARD CONTRACTS & RUMOURS
 *
 *   Sergeant Otis Mulroney (Guard Barracks, `barracks_sergeant`):
 *     - quartermaster of the Neon Guard: the armory trades only with sworn
 *       guards (FO3 Megaton / NV NCR style "members only" vendor);
 *     - "Neon Guard Contract" — a repeatable (radiant, one per day) job off
 *       the barracks bounty board: a feral nest / raider crew / mole rats /
 *       wild dogs / giant ants / radscorpions / a super mutant patrol / a
 *       yao guai, levelled to the player, dropped on a real Midtown
 *       intersection 150–520 m from the square and named by its streets
 *       ("West 47th Street & 8th Avenue"). Kill counter in the Pip-Boy,
 *       report back for caps; milestones at 3 / 5 / 10 contracts.
 *   "Word on the Street" — FO-style misc leads: every rumour you hear in
 *       the Neon (Lucky Lou, the townsfolk) adds a Pip-Boy objective with a
 *       marker on the person to talk to; ticked when that quest starts.
 * All writing original.
 * ===================================================================== */

  const FRIENDLY = ['times_square', 'radio_city', 'grand_central', 'rockefeller', 'st_patricks', 'carnegie_hall', 'plaza_hotel', 'flatiron', 'vault120'];
  let siteCache = null;
  /** Real Midtown intersections for contracts: open street, off every site / location, on land. */
  function contractSites() {
    if (siteCache) return siteCache;
    const Ly = G.LAYOUT, out = [];
    if (!Ly || !Ly.avenues || !Ly.streets) return (siteCache = out);
    for (const a of Ly.avenues) for (const s of Ly.streets) {
      if (s.park) continue;
      const x = a.x, z = s.z;
      if (z < a.z0 + 8 || z > a.z1 - 8 || x < s.x0 + 8 || x > s.x1 - 8) continue;
      const d = Math.hypot(x - TS.x, z - TS.z); if (d < 150 || d > 520) continue;
      if (!Ly.isLand(x, z) || (Ly.inPark && Ly.inPark(x, z))) continue;
      if (Ly.siteAt(x, z, 18)) continue;
      if (Ly.locationAt(x, z)) continue;
      let bad = false;
      for (const id of FRIENDLY) { const l = Ly.getLocation(id); if (l && Math.hypot(x - l.x, z - l.z) < l.r + 70) { bad = true; break; } }
      if (bad) continue;
      const st = Ly.streetNameAt(x, z), av = Ly.avenueNameAt(x, z);
      out.push({ x, z, name: (st || 'the cross street') + ' & ' + (av || 'the avenue') });
    }
    return (siteCache = out);
  }
  const C = () => L.flag('neon_contract') || null;
  function lvl() { return Math.max(1, L.level()); }
  function newContract() {
    const l = lvl();
    const kinds = KINDS.filter((k) => l >= k.minL && (!k.maxL || l <= k.maxL));
    const last = C();
    const kind = pick(kinds.filter((k) => !last || k.k !== last.kind).length ? kinds.filter((k) => !last || k.k !== last.kind) : kinds);
    const sites = contractSites().filter((s) => !last || Math.hypot(s.x - last.x, s.z - last.z) > 60);
    let site = null;
    for (let tries = 0; tries < 12 && sites.length; tries++) {
      const s = pick(sites);
      const f = L.freeSpot(s.x, s.z, 0, { maxR: 10 });
      if (f && !f.forced) { site = { x: f.x, y: f.y, z: f.z, name: s.name }; break; }
    }
    if (!site) return null;
    const n = (Number(L.flag('neon_contract_n')) || 0) + 1;
    const pack = kind.pack(l).filter(Boolean);
    const ids = [];
    pack.forEach(([type, cnt]) => { for (let i = 0; i < cnt; i++) ids.push({ id: 'nc' + n + '_' + ids.length, type }); });
    const pay = Math.round((55 + l * 12 + ids.length * 8) / 5) * 5;
    return { n, kind: kind.k, what: kind.what, brief: kind.brief, x: site.x, y: site.y, z: site.z, place: site.name, ids, lvl: l, pay, xp: 60 + l * 12 };
  }
  function spawnContract(c) {
    if (!c) return;
    const K = KINDS.find((k) => k.k === c.kind) || KINDS[0];
    c.ids.forEach((t, i) => {
      if (L.isDead(t.id)) return;
      const ang = i * 2.39996, r = 2 + (i % 3) * 2.2;
      const s = L.freeSpot(c.x + Math.cos(ang) * r, c.z + Math.sin(ang) * r, c.y || 0, { maxR: 6 }) || { x: c.x, y: c.y || 0, z: c.z };
      const boss = t.type === 'glowing_one' || t.type === 'yao_guai' || (K.k === 'raiders' && i === 0);
      const def = { id: t.id, type: t.type, level: c.lvl + (boss ? 2 : 0), pos: { x: s.x, y: s.y, z: s.z }, ai: 'guard', home: { x: s.x, z: s.z, r: 9 }, hostile: true, noRespawn: true };
      if (t.type === 'raider') { def.faction = 'raiders'; def.voice = 'raider'; if (boss) { def.name = pick(['Toll Boss Vic', 'Crowbar Kate', 'Two-Tooth Tommy', 'Big Mouth Benny']); def.weapon = c.lvl >= 6 ? 'assault_rifle' : 'combat_shotgun'; } }
      L.spawnNPC(def);
    });
  }
  function aliveTargets(c, extraDead) { return c ? c.ids.filter((t) => t.id !== extraDead && !L.isDead(t.id)) : []; }
  function contractMarker() {
    const c = C(); if (!c) return [];
    const alive = aliveTargets(c).map((t) => L.npc(t.id)).filter((a) => a && !a.dead && a.pos);
    if (alive.length && alive.some((a) => Math.hypot(a.pos.x - c.x, a.pos.z - c.z) < 60)) {
      let x = 0, z = 0; alive.forEach((a) => { x += a.pos.x; z += a.pos.z; });
      return [{ x: x / alive.length, z: z / alive.length, label: 'Contract: ' + c.what }];
    }
    return [{ x: c.x, z: c.z, label: 'Contract: ' + c.what }];
  }
  function acceptContract() {
    const c = newContract();
    if (!c) { L.notify('The bounty board is empty. Come back tomorrow.'); return false; }
    // radiant: the quest id is reused — clear its old completed/failed record first
    if (Q.isComplete(QID) || Q.isFailed(QID)) L.STATE.delete(QID);
    cleanupOld();
    L.setFlag('neon_contract_n', c.n);
    L.setFlag('neon_contract', c);
    spawnContract(c);
    Q.start(QID);
    Q.setTracked(QID);
    return true;
  }
  function cleanupOld() {
    const prev = L.flag('neon_contract_prev');
    if (prev && prev.ids) for (const t of prev.ids) L.removeNPC(t.id);
    L.setFlag('neon_contract_prev', null);
  }
  function payContract() {
    const c = C(); if (!c) return;
    const done = L.incFlag('neon_contracts_done');
    const def = L.DEFS.get(QID);
    def.rewards = { xp: c.xp, caps: c.pay, karma: 10 };
    Q.complete(QID);
    L.setFlag('neon_contract_day', G.time.day);
    L.setFlag('neon_contract_prev', { ids: c.ids });
    L.setFlag('neon_contract', null);
    if (done === 3) { L.giveCaps(100); L.notify('Contract bonus: +100 caps. "Three for three. You\'re starting to look like a guard."'); }
    if (done === 5) { L.setFlag('neon_marksman', true); marksman(); L.notify('Neon Marksman: Sergeant Mulroney pins a tin sharpshooter\'s badge on your collar. +1 PER, +3 Guns.'); L.news('Word from the Neon: the Guard\'s newest contract hunter has cleared five nests around Midtown. Sergeant Mulroney says the avenues are "almost boring now. Almost."'); }
    if (done === 10) { L.giveCaps(300); L.karma(50, 'Ten contracts for the Neon Guard'); L.news('The Neon Guard toasts its busiest contractor: ten nests, crews and packs cleared from the streets around Times Square. Mayor Crowe has proposed a statue. Nobody has seconded it.'); }
  }
  function abandonContract() {
    const c = C(); if (!c) return;
    Q.fail(QID);
    L.setFlag('neon_contract_prev', { ids: c.ids });
    L.setFlag('neon_contract', null);
    L.setFlag('neon_contract_day', G.time.day);
  }
  /** 5-contract reward: a permanent perk-like effect (re-applied after every load, like the companion perks). */
  function marksman() {
    if (!L.flag('neon_marksman') || G.isStub('rpg') || !G.rpg.addEffect) return;
    try { G.rpg.removeEffect && G.rpg.removeEffect('neon_marksman'); G.rpg.addEffect({ id: 'neon_marksman', name: 'Neon Marksman', kind: 'buff', duration: 0, mods: { special: { P: 1 }, skills: { guns: 3 } }, source: 'quest', desc: 'Mulroney\'s tin sharpshooter\'s pin. +1 PER, +3 Guns.' }); } catch (e) { G.reportError('quests:neonMarksman', e); }
  }

  const heard = (ld) => ld.flags.some((f) => !!L.flag(f));
  function leadMarker(ld) {
    if (ld.npc) return [npcM(ld.npc, ld.label)];
    const a = N() && N().has(ld.anchor); return a ? [{ x: a.x, z: a.z, y: a.y, label: ld.label }] : [];
  }
  function rumorsTick() {
    const any = LEADS.some(heard);
    if (!any) return;
    if (!Q.isStarted(RID)) { Q.start(RID); return; }
    if (Q.isActive(RID) && LEADS.every((ld) => { try { return ld.done(); } catch (e) { return false; } })) Q.complete(RID);
  }

    if (capt && capt.nodes && capt.nodes.hello && !capt.nodes.hello.responses.some((r) => r._sarge)) {
      capt.nodes.hello.responses.splice(Math.max(0, capt.nodes.hello.responses.length - 2), 0,
        { _sarge: true, text: 'Any more work for the Guard?', cond: () => L.alive('neon_sarge') && (Q.isComplete('sq_neon_guard') || (member() && !Q.isActive('sq_neon_guard'))), next: 'sarge_work' });
      // Completability: if Mulroney is not in the world (barracks not built, or he died), Kincaid runs the board herself.
      const sarge = L.DIALOGUES.get('dlg_neon_sarge');
      const names = ['board', 'taken', 'remind', 'quit', 'pay'];
      for (const n of names) {
        const src = sarge.nodes[n]; if (!src) continue;
        const rs = Array.isArray(src.responses) ? src.responses.map((r) => Object.assign({}, r, names.indexOf(r.next) >= 0 ? { next: 'sg_' + r.next } : null)) : src.responses;
        capt.nodes['sg_' + n] = Object.assign({}, src, { responses: rs });
      }
      const noSarge = () => !L.alive('neon_sarge');
      capt.nodes.hello.responses.splice(Math.max(0, capt.nodes.hello.responses.length - 2), 0,
        { _sarge: true, text: 'Mulroney\'s not at his post. Anything on the bounty board?', cond: () => noSarge() && member() && !Q.isActive(QID), next: 'sg_board' },
        { _sarge: true, text: () => { const c = C(); return 'About the contract — ' + (c ? c.what + ' at ' + c.place : '…') + '.'; }, cond: () => noSarge() && Q.stage(QID) === 'hunt', next: 'sg_remind' },
        { _sarge: true, text: 'Contract\'s done. They\'re dead.', cond: () => noSarge() && Q.stage(QID) === 'report', next: 'sg_pay' });
      capt.nodes.sarge_work = { text: 'Always. Mulroney runs the contract board down in the barracks — the subway kiosk by the island. One job a day, paid on proof. He\'ll also sell you guard kit at cost. Tell him I said to stop hoarding the good grenades.', responses: [{ text: 'I\'ll see him.', next: 'hello' }] };
    }
    // Barnaby and the townsfolk now drop the Speakeasy / Oracle rumours
    const bell = L.DIALOGUES.get('dlg_neon_bellhop');
    if (bell && bell.nodes && bell.nodes.subway && !bell.nodes.subway._r) { bell.nodes.subway._r = true; const prev = bell.nodes.subway.onEnter; bell.nodes.subway.onEnter = function (a) { if (prev) prev(a); L.setFlag('neon_rumor_speakeasy'); }; }
    if (bell && bell.nodes && bell.nodes.tour && !bell.nodes.tour._r) { bell.nodes.tour._r = true; const prev = bell.nodes.tour.onEnter; bell.nodes.tour.onEnter = function (a) { if (prev) prev(a); L.setFlag('neon_rumor_oracle'); }; }
  });

  // ==================================================================
  // World, spawn, events, update
  // ==================================================================
  L.hook('world', function () {
    const b = N() && N().has(['bounty_board']);
    if (b) G.world.addInteractable({ pos: L.V3(b.x, (b.y || 0) + 1.6, b.z), radius: 1.0, label: 'Guard Bounty Board', verb: 'Read', priority: 0.8,
      onInteract() { L.startDialogue('dlg_neon_bounty_board', null); } });
  });
  L.hook('spawn', function () {
    siteCache = null;
    marksman();
    const c = C();
    if (c && Q.stage(QID) === 'hunt') spawnContract(c);
  });
  G.events.on('actorKilled', (a) => {
    if (!a || !a.id || !/^nc\d+_/.test(a.id)) return;
    const c = C(); if (!c || Q.stage(QID) !== 'hunt' || !c.ids.some((t) => t.id === a.id)) return;
    const left = aliveTargets(c, a.id).length;
    if (left <= 0) { Q.setStage(QID, 'report'); L.notify('Contract target cleared. Report to Sergeant Mulroney.'); } else L.refresh(QID);
  });
  G.events.on('questFlag', (name) => { if (typeof name === 'string' && name.indexOf('neon_rumor_') === 0) L.later(0.5, rumorsTick); });
  let t1 = 0;
  L.hook('update', function (dt) {
    if (dt <= 0) return;
    t1 += dt; if (t1 < 1) return; t1 = 0;
    if (Q.isActive(RID)) { try { rumorsTick(); } catch (e) { G.reportError('quests:neon rumors', e); } }
  });

  // Debug (dev console)
  Q.debug.neonContract = () => { L.setFlag('neon_guard_member', true); L.setFlag('neon_contract_day', -1); acceptContract(); return C(); };
  Q.debug.neonContractSites = () => contractSites().map((s) => s.name + ' (' + Math.round(s.x) + ',' + Math.round(s.z) + ')');
  L.neonContracts = { newContract, spawnContract, acceptContract, contractSites, LEADS, KINDS };
})();
