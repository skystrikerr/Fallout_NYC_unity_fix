  L.hook('init', function () {
    for (const n of NOTES) {
      const d = Object.assign({ weight: 0, value: 0, quest: n.quest !== undefined ? n.quest : false }, n);
      d.desc = d.desc || (n.type === 'holotape' ? 'A Vault-Tec holotape. Play it in your Pip-Boy (DATA → Notes).' : 'A note. Read it in your Pip-Boy (DATA → Notes).');
      L.defineItem(d);
    }
    for (const it of QUEST_ITEMS) L.defineItem(Object.assign({}, it));
    for (const it of APPAREL) L.defineItem(Object.assign({ noCondition: false }, it));
  });

  // Personalize the eviction notice with the player's name at pickup time.
  L.hook('newGame', function () {
    if (G.isStub('items') || !G.items.get) return;
    const d = G.items.get('q_note_eviction');
    const src = NOTES.find((n) => n.id === 'q_note_eviction');
    if (d && src) d.text = src.text.replace('{NAME}', (G.player.name || 'Wanderer').toUpperCase());
  });

  // Carrying the Blessed Isotope slowly irradiates you (it's blessed, not safe).
  // Implemented as a small rad source riding on the player, so the core applies
  // Rad-X / Rad Resistance, the Geiger counter clicks and the HUD shows +RADS.
  // Only polled while the Atom quest could involve it (no per-frame inventory scan).
  let isoSrc = null, isoT = 0, carrying = false;
  function isoOff() { if (isoSrc) { try { G.world.removeRadSource(isoSrc); } catch (e) { /* */ } isoSrc = null; } }
  L.hook('update', function (dt, realDt) {
    isoT -= realDt;
    if (isoT <= 0) {
      isoT = 1;
      const relevant = G.quests.isActive('sq_atom');
      carrying = relevant && L.has('q_blessed_isotope');
    }
    if (!carrying) { isoOff(); return; }
    if (!isoSrc && G.world.addRadSource) isoSrc = G.world.addRadSource({ x: 0, y: 0, z: 0, radius: 2.5, strength: 0.6, tag: 'blessed_isotope' });
    if (isoSrc) { const p = G.player.pos; isoSrc.x = p.x; isoSrc.y = p.y + 1; isoSrc.z = p.z; }
  });
  L.hook('reset', function () { isoOff(); carrying = false; isoT = 0; });

  L.NOTES = NOTES;
})();
