/* =====================================================================
 * FALLOUT: NEW YORK — QUESTS — 20: VAULT 120 (MQ01 "Lease on Life")
 *
 * The first ten minutes. Vault 120 is a Vault-Tec "rent-stabilized"
 * vault beneath Central Park, run for two centuries by an eviction-happy
 * Mister Handy called THE SUPERINTENDENT ("Mr. Super"). Today is the
 * player's 19th birthday — and Lease Renewal Day.
 *
 * Beats: birthday party (Overseer gives the Pip-Boy) → Chief Doyle's
 * "present" (10mm + pest control tutorial) → radroaches → Zeke's locker
 * (holotape + Liberty Key) → Mr. Super's eviction → Overseer's door code
 * → vault door terminal → the surface (exit cinematic) → MQ02.
 *
 * The vault interior geometry belongs to the map module, so every
 * position is derived at runtime from LAYOUT.playerStart, the exit door
 * found in G.world.doors, and optional map anchors, and then validated
 * with a free-spot search (no NPC spawns inside walls).
 * ===================================================================== */
(function () {
  'use strict';
  const G = window.G;
  const L = G.quests.lib;
  const Q = G.quests;

  // ------------------------------------------------------------------
  // Vault geometry helpers
  // ------------------------------------------------------------------
  const V = {};   // resolved vault spots for this session
  function vaultAnchor(name) {
    // Optional map-provided anchors (see NOTES "Requests to other modules").
    const tries = [
      () => G.map.getAnchor && G.map.getAnchor('vault120', name),
      () => G.map.vaultAnchors && G.map.vaultAnchors[name],
      () => G.map.anchors && G.map.anchors.vault120 && G.map.anchors.vault120[name],
      () => G.landmarks.getAnchor && G.landmarks.getAnchor('vault120', name),
    ];
    for (const t of tries) {
      let a = null; try { a = t(); } catch (e) { a = null; }
      if (a && isFinite(a.x) && isFinite(a.z)) return { x: a.x, y: a.y || 0, z: a.z };
    }
    return null;
  }
  function computeVaultSpots() {
    const S = G.LAYOUT.playerStart;
    const yaw = S.yaw || 0;
    const f = { x: -Math.sin(yaw), z: -Math.cos(yaw) }, r = { x: Math.cos(yaw), z: -Math.sin(yaw) };
    const at = (fw, rt) => ({ x: S.x + f.x * fw + r.x * rt, z: S.z + f.z * fw + r.z * rt });
    const door = L.vaultExitDoor();
    const y0 = S.y || 0;
    const los = { x: S.x, y: y0, z: S.z };
    const free = (p, y, opts) => L.freeSpot(p.x, p.z, y !== undefined ? y : y0, Object.assign({ maxR: 6 }, opts || {}));
    // Map anchors (map module builds the vault: atrium party stage, office, clinic, reactor...)
    const atrium = vaultAnchor('atrium'), stage = vaultAnchor('atrium_stage');
    const off = (a, dx, dz, y) => a ? free({ x: a.x + dx, z: a.z + dz }, y !== undefined ? y : a.y) : null;
    V.start = { x: S.x, y: y0, z: S.z };
    const exitDoor = vaultAnchor('vault_door') || (door ? { x: door.def.pos.x, y: 0, z: door.def.pos.z } : null);
    V.door = door ? { x: door.def.pos.x, z: door.def.pos.z } : (exitDoor || at(28, 0));
    const party = stage || atrium;
    // Where everyone looks during the party: the corridor mouth the player arrives from.
    const corridor = vaultAnchor('corridor_n') || atrium || S;
    V.face = (p) => Math.atan2(-(corridor.x - p.x), -(corridor.z - p.z));
    if (party) {
      // A tight, readable party: everyone gathered around the cake table in front of
      // the stage in a loose semicircle, facing the corridor the birthday tenant
      // walks in from (FO3's surprise-party framing). Offsets are relative to the
      // stage centre (the stage is 8 x 4 m; +z is toward the corridor).
      const S0 = stage || { x: atrium.x, y: 0.4, z: atrium.z - 7 };
      V.partyCenter = { x: S0.x, y: 0, z: S0.z + 4.6 };           // the cake table
      V.punch = { x: S0.x + 4.3, y: 0, z: S0.z + 5.2 };           // punch bowl table
      V.stage = { x: S0.x, y: S0.y || 0.4, z: S0.z };
      V.overseer = off(S0, 0, 1.1, S0.y || 0.4);                  // on the stage, front edge
      V.lucy = off(S0, 1.9, 6.6, 0);                              // front-right: first to greet you
      V.doctor = off(S0, -2.3, 3.7, 0);                           // behind-left of the cake
      V.super = off(S0, 2.6, 3.4, 0);                             // hovering at the punch ("staring at it")
      V.doyle = off(S0, 5.8, 4.1, 0);                             // lurking by the punch bowl, as promised
      V.tommy = off(S0, -4.6, 7.2, 0);                            // leaning, apart from the fun
    } else {
      V.overseer = free(at(4.2, 0.4), y0, { los }); V.lucy = free(at(3.0, 2.1), y0, { los }); V.doyle = free(at(4.6, -2.3), y0, { los });
      V.doctor = free(at(6.0, 1.4), y0, { los }); V.tommy = free(at(7.5, -3.6), y0, { los }); V.super = free(at(6.4, 3.2), y0, { los });
    }
    // After the party people go back to work.
    V.overseerOffice = vaultAnchor('overseer') || V.overseer;
    V.doyleWork = off(vaultAnchor('security'), 0, 1.5, 0) || V.doyle;
    V.doctorWork = vaultAnchor('doctor') || V.doctor;
    V.mabel = vaultAnchor('cafeteria_cook') ? off(vaultAnchor('cafeteria'), 0, 0, 0) : free(at(-3.5, 3), y0, { maxR: 12 });
    V.roach = vaultAnchor('reactor') || free({ x: S.x + (V.door.x - S.x) * 0.62, z: S.z + (V.door.z - S.z) * 0.62 }, y0, { maxR: 10 });
    const bed = vaultAnchor('player_bed');
    V.locker = bed ? off(bed, -1.2, -1.0, 0) : free(at(1.5, -3.2), y0, { los, maxR: 8 });
    const con = vaultAnchor('door_console');
    if (con) { V.doorTerm = off(con, 0.2, 2.4, 0); V.doorTermYaw = Math.PI / 2; }
    else {
      const D = V.door, dx = S.x - D.x, dz = S.z - D.z, dl = Math.hypot(dx, dz) || 1;
      V.doorTerm = free({ x: D.x + dx / dl * 3.2 + (-dz / dl) * 1.8, z: D.z + dz / dl * 3.2 + (dx / dl) * 1.8 });
      V.doorTermYaw = Math.atan2(-(S.x - D.x), -(S.z - D.z));
    }
    const ov = vaultAnchor('overseer');
    V.overseerTerm = ov ? off(ov, -4.5, 3.6, ov.y) : free(at(9, 2.5), y0, { maxR: 8 });
    const sec = vaultAnchor('security');
    V.securityTerm = sec ? off(sec, -3.5, 4.5, 0) : null;
    V.superTerm = atrium ? off(atrium, -9, 5, 0) : free(at(9, -2.5), y0, { maxR: 8 });
    V.faceStart = (p) => Math.atan2(-(S.x - p.x), -(S.z - p.z)); // yaw that looks at the start point
    return V;
  }

  function spawnRoaches() {
    if (!V.roach || G.isStub('actors')) return;
    ROACH_IDS.forEach((id, i) => {
      if (L.flag('dead_' + id)) return;
      const a = (i / 3) * Math.PI * 2;
      L.spawnNPC({ id, type: 'radroach', name: 'Radroach', faction: 'creatures', level: 1, hostile: true, ai: 'wander', noRespawn: true,
        hp: 12, damage: 2, at: { fallback: { x: V.roach.x + Math.cos(a) * 1.6, z: V.roach.z + Math.sin(a) * 1.6, y: V.roach.y }, maxR: 5 },
        home: { x: V.roach.x, z: V.roach.z, r: 5 } });
    });
  }

  function onRoachKilled(actor) {
    if (!actor || ROACH_IDS.indexOf(actor.id) < 0) return;
    L.setFlag('dead_' + actor.id);
    if (Q.stage('mq01') !== 'roaches') return;
    if (roachesLeft() === 0) { L.xp(15, 'Pest control'); Q.setStage('mq01', 'locker'); }
    else L.refresh('mq01');
  }

  function superEviction() {
    const sup = L.npc('v120_super');
    // Mr. Super hovers over to the player and serves the notice.
    if (sup && sup.moveTo) { try { sup.moveTo(G.player.pos.x + 1.5, G.player.pos.z + 1.5); } catch (e) { /* ignore */ } }
    L.sfx('alarm', { volume: 0.35 });
    L.later(0.3, () => L.notify('ATTENTION TENANTS: The lease of Tenant ' + pname().toUpperCase() + ' has been TERMINATED. Please join us in wishing them a pleasant apocalypse!'));
    L.later(1.2, () => L.say('v120_super', 'EVICTION! Eviction! Oh, happy day for the waiting list!'));
    L.later(2.0, () => { L.give('q_note_eviction', 1); });
    L.later(4.0, () => L.say('v120_lucy', 'They can\'t do that! ...Can they do that?'));
  }

  function unlockDoorTerminal() { doorTerm.lock = 0; L.terminal('v120_door', doorTerm); if (!G.isStub('terminals') && G.terminals.unlock) { try { G.terminals.unlock('v120_door'); } catch (e) { /* */ } } }
  function openVaultDoor() {
    L.setFlag('vault_door_open');
    L.sfx('vault_alarm', { volume: 0.6 });
    L.later(0.6, () => L.sfx('vault_door'));
    // Map module owns the giant cog door; use its hooks if present.
    if (!G.isStub('map') && G.map.lockVaultDoor) { try { G.map.lockVaultDoor(false); } catch (e) { /* */ } }
    const hooks = ['openVaultDoor', 'openVault', 'vaultDoorOpen'];
    for (const h of hooks) if (typeof G.map[h] === 'function' && !G.isStub('map')) { try { G.map[h](); } catch (e) { G.reportError('quests:vaultdoor', e); } break; }
    if (G.map.vault && typeof G.map.vault.open === 'function') { try { G.map.vault.open(); } catch (e) { /* */ } }
    G.player.shake(0.35, 2.2);
    L.notify('The great cog door grinds open. Two hundred years of stale air rushes out to meet the morning.');
    if (Q.isActive('mq01')) Q.setStage('mq01', 'leave');
    G.events.emit('vaultDoorOpened');
  }

  // Crowd bark governor. Seven NPCs in one room would otherwise all fire an idle
  // bark on their first AI tick. Only one vault resident speaks at a time, with a
  // breath between lines (combat's per-actor bark gate is actor._lastBark).
  // ------------------------------------------------------------------
  const VAULT_IDS = ['v120_overseer', 'v120_lucy', 'v120_doyle', 'v120_doctor', 'v120_tommy', 'v120_super', 'v120_mabel'];
  const barkSeen = new Map();
  let vaultQuietUntil = 0;
  /** Suppress idle barks from every vault resident for `sec` seconds, staggered. */
  function holdVaultBarks(sec) {
    const now = G.time.elapsed;
    vaultQuietUntil = Math.max(vaultQuietUntil, now + sec);
    VAULT_IDS.forEach((id, i) => {
      const a = L.npc(id); if (!a) return;
      // combat's bark(): silent while (now - _lastBark) < 4
      a._lastBark = Math.max(a._lastBark || -1e9, now + sec - 4 + i * 2.5);
      barkSeen.set(id, a._lastBark);
      if (a.nextThink !== undefined) a.nextThink = Math.max(a.nextThink || 0, now + sec * 0.5 + i * 1.3);
    });
  }
  L.hook('update', function () {
    if (!L.inInterior()) return;
    const V0 = G.LAYOUT.INTERIORS.vault120 && G.LAYOUT.INTERIORS.vault120.origin;
    if (V0 && Math.hypot(G.player.pos.x - V0.x, G.player.pos.z - V0.z) > 260) return;
    const now = G.time.elapsed;
    let fresh = null;
    for (const id of VAULT_IDS) {
      const a = L.npc(id); if (!a) continue;
      const lb = a._lastBark;
      if (lb === undefined || lb === barkSeen.get(id)) continue;
      // a new idle bark this frame
      if (fresh || now < vaultQuietUntil) {
        // someone else already spoke this frame, or a scripted speech is running: hush it
        const hb = (G.combat && G.combat.hideBark) || (G.__cmb && G.__cmb.hideBark);   // same frame: never visible
        if (typeof hb === 'function') { try { hb(a); } catch (e) { /* */ } }
        barkSeen.set(id, lb);
        continue;
      }
      fresh = id; barkSeen.set(id, lb);
    }
    if (fresh) {
      // give the speaker the floor: everyone else waits ~4.5 s
      for (const id of VAULT_IDS) {
        if (id === fresh) continue;
        const a = L.npc(id); if (!a) continue;
        a._lastBark = Math.max(a._lastBark || -1e9, now + 0.5);
        barkSeen.set(id, a._lastBark);
      }
    }
  });
  L.vaultHoldBarks = holdVaultBarks;
