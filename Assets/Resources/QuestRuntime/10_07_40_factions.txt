  /** Kong lives on the observation deck: guide the player via lobby door → express elevator. */
  function kongMarker() {
    const k = L.npc('esb_kong');
    const kp = k && !k.dead ? k.pos : null;
    const P = G.player.pos;
    if (!kp) return { x: ESB.x, z: ESB.z, label: 'Empire State Building' };
    if (kp.y < 60 || P.y > 100) return { x: kp.x, z: kp.z, y: kp.y, label: 'KONG' };
    const doors = L.doors();
    const inLobby = P.x >= G.LAYOUT.INTERIOR_X_MIN && Math.hypot(P.x - 4000, P.z - 800) < 200;
    if (inLobby) { const e = doors.find((d) => d.def.pos.x >= 2500 && d.def.target && d.def.target.y > 100); if (e) return { x: e.def.pos.x, z: e.def.pos.z, label: 'Express Elevator (to KONG)' }; }
    const lobby = L.doorInto('empire_lobby');
    if (lobby) return { x: lobby.def.pos.x, z: lobby.def.pos.z, label: 'Empire State Lobby (KONG is on the 86th floor)' };
    return { x: kp.x, z: kp.z, label: 'KONG' };
  }

        if (bookCount() >= 3) Q.setStage('sq_overdue', 'return');
        else L.refresh('sq_overdue');

    if (id === 'q_note_betty_ledger' && (Q.stage('sq_exchange') === 'hunt' || !Q.isStarted('sq_exchange'))) {
      L.setFlag('betty_ledger_stolen');
      // A clean lift: Betty alive and nobody at the Exchange ever spotted you.

      if (!Q.isStarted('sq_exchange')) Q.start('sq_exchange', { stage: 'report' }); else Q.setStage('sq_exchange', 'report');
    }
    if (id === 'q_holo_directive_7a') L.setFlag('knows_directive');
  });
  // "Hostile Takeover" ghost run: were you ever seen at the Exchange?
  function nearExchange() {
    const P = G.player.pos, pts = [];
    const b = L.npc('ws_betty'); if (b && !b.dead) pts.push(b.pos);
    if (L.pickups.betty_ledger) pts.push(L.pickups.betty_ledger.pos);
    return pts.some((q) => Math.hypot(P.x - q.x, P.z - q.z) < 60);
  }

  L.hook('update', function (dt, realDt) {
    xcT -= realDt; if (xcT > 0) return; xcT = 0.5;

    const inCombat = !G.isStub('combat') && G.combat.inCombat && G.combat.inCombat();
    if (inCombat && nearExchange()) { L.setFlag('exchange_alerted'); if (Q.isActive('sq_exchange')) L.refresh('sq_exchange'); }
  });

    if (a.id === 'esb_kong') { L.setFlag('kong_dead'); }
    if (a.id === 'ws_betty') { L.setFlag('betty_dead'); if (Q.stage('sq_exchange') === 'hunt') Q.setStage('sq_exchange', 'report'); else if (!Q.isStarted('sq_exchange')) { Q.start('sq_exchange', { stage: 'report' }); } }

      L.setFlag('hale_dead');

    }
    if (a.id === 'bos_stroud') { L.setFlag('stroud_dead'); }
  });

  function onCore() {
    if (L.flag('core_obtained')) return;
    L.setFlag('core_obtained');
    if (!Q.isStarted('mq05')) {
      // Explorer got here early: fold the earlier quests together.
      if (Q.isActive('mq04')) { L.setFlag('mq_faction', faction() || 'none'); Q.complete('mq04'); }
      if (!Q.isStarted('mq05')) Q.start('mq05', { stage: 'core' });
    }
    if (Q.isActive('mq04')) { if (!faction()) L.setFlag('mq_faction', 'none'); Q.complete('mq04'); }
    const f = faction();
    if (f === 'bos') Q.setStage('mq05', 'return_bos');
    else if (f === 'enclave') Q.setStage('mq05', 'return_enclave');
    else { L.setFlag('mq_faction', 'none'); Q.complete('mq05'); }
  }

  /** Remove an item from an actor's inventory whatever its representation. */
  function actorTake(actor, id) {
    if (!actor || !actor.inventory) return;
    const inv = actor.inventory;
    try {
      if (inv instanceof Map) inv.delete(id);
      else if (Array.isArray(inv)) { for (let i = inv.length - 1; i >= 0; i--) if ((Array.isArray(inv[i]) ? inv[i][0] : inv[i].id) === id) inv.splice(i, 1); }
      else if (typeof inv.remove === 'function') inv.remove(id, 99);
      else if (typeof inv === 'object') delete inv[id];
    } catch (e) { /* ignore */ }
  }
  function kongGives(how) {

    actorTake(L.npc('esb_kong'), 'q_knickerbocker_core');
    L.give('q_knickerbocker_core', 1);
  }
