    const t = C.PART_LABELS[type];
    if (t && t[key]) return t[key];
    // G.chars supplies per-type labels (e.g. creature limbs)
    if (!t && !G.isStub('chars') && typeof G.chars.info === 'function') { try { const inf = G.chars.info(type); if (inf && inf.labels && inf.labels[key]) return inf.labels[key]; } catch (e) { /* ignore */ } }

    return String(key).replace(/([A-Z])/g, ' $1').replace(/^./, (c) => c.toUpperCase());
  };

  C.weighted = (pairs) => {
    let tot = 0; for (const p of pairs) tot += p[1];
    let r = Math.random() * tot;
    for (const p of pairs) { r -= p[1]; if (r <= 0) return p[0]; }
    return pairs[pairs.length - 1][0];
  };

  /** Does the player have a perk? Defensive against different rpg shapes. */
  C.hasPerk = function (id) {
    if (G.isStub('rpg')) return false;
    const R = G.rpg;
    try {
      if (typeof R.hasPerk === 'function') return !!R.hasPerk(id);
      if (typeof R.perkRank === 'function') return R.perkRank(id) > 0;
      const t = R.takenPerks || R.perksTaken || R.myPerks;
      if (t) { if (t instanceof Set) return t.has(id); if (Array.isArray(t)) return t.includes(id) || t.some((p) => p && p.id === id); if (typeof t === 'object') return !!t[id]; }
    } catch (e) { /* ignore */ }
    return false;
  };
