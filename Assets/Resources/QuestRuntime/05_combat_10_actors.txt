/* =====================================================================
 * FALLOUT: NEW YORK — COMBAT: actors
 * Owned by: combat.
 *
 * G.actors: spawning, registry, queries, streaming (actors far from the
 * player are detached from the scene and frozen), Talk/Search
 * interactables, NPC weapons in hands, per-part hit capsules (for hitscan
 * and V.A.T.S.), floating subtitle barks.
 *
 * The Actor class is exposed privately as G.__cmb.Actor; the AI (20_ai.js)
 * and damage (30_combat.js) files add behaviour to it.
 * ===================================================================== */
(function () {
  'use strict';
  const G = window.G;
  const THREE = window.THREE;
  const C = G.__cmb;

  C.list = [];               // all actors
  C.byId = new Map();        // stable id → actor
  C.pendingState = new Map();// saved state for id'd actors not yet spawned (applied on spawn)
  C.ACTIVE_R = 175;          // AI + in-scene radius
  C.DETACH_R = 215;          // leave the scene beyond this (hysteresis)
  let uid = 0;

  const _v1 = new THREE.Vector3(), _v2 = new THREE.Vector3(), _m = new THREE.Matrix4();

  // ------------------------------------------------------------------
  // Character creation (G.chars or placeholder)
  // ------------------------------------------------------------------
  /** chars type naming — raider/enclave_soldier/brotherhood_paladin are chars types too. */
  function charTypeFor(type) {
    if (G.isStub('chars')) return type;
    try {
      const types = typeof G.chars.types === 'function' ? G.chars.types() : null;
      if (types && types.length && types.indexOf(type) < 0) {
        if (type === 'raider' || type === 'enclave_soldier' || type === 'brotherhood_paladin') return 'human';
        if (type === 'sentry_bot') return 'protectron';
      }
    } catch (e) { /* ignore */ }
    return type;
  }
  function createCharacter(type, def) {
    const opts = { seed: def.seed !== undefined ? def.seed : Math.floor(Math.random() * 1e9), variant: def.variant, outfit: def.outfit,
      gender: def.gender, skinTone: def.skinTone, hair: def.hair, ghoul: def.ghoul, level: def.level };
    if (def.hat) { opts.hat = def.hat; if (def.hatColor !== undefined) opts.hatColor = def.hatColor; }
    if (!opts.outfit) {
      if (type === 'raider') opts.outfit = (def.level || 1) > 6 && Math.random() < 0.5 ? 'raider_heavy' : 'raider';
      else if (type === 'enclave_soldier') opts.outfit = 'enclave_power_armor';
      else if (type === 'brotherhood_paladin') opts.outfit = 'bos_power_armor';
    }
    if (!G.isStub('chars') && typeof G.chars.create === 'function') {
      const ct = charTypeFor(type);
      let known = true;
      if (typeof G.chars.info === 'function') { try { known = !!G.chars.info(ct); } catch (e) { known = false; } }
      if (known) {
        try {
          const ch = G.chars.create(ct, Object.assign({ addToScene: false }, opts));
          if (ch && ch.root) { ch.autoUpdate = false; ch._cmbSeed = opts.seed; ch._cmbOutfit = opts.outfit; return ch; }
        } catch (e) { G.reportError('combat:chars.create ' + type, e); }
      }
    }
    const ph = C.createPlaceholder(type, opts);
    if (ph) { ph._cmbSeed = opts.seed; ph._cmbOutfit = opts.outfit; }
    return ph;
  }

  // ------------------------------------------------------------------
  // NPC weapons
  // ------------------------------------------------------------------
  /** Normalised NPC weapon info: {id, name, damage, rof, automatic, magSize, reloadTime, spread, range, pellets,
   *  projectile, projectileSpeed, explosionRadius, sound, damageType, melee, beam, skill, critMult, apCost, ammo} */
  C.weaponInfo = function (wid) {
    if (!wid) return null;
    let d = null;
    if (!G.isStub('weapons') && typeof G.weapons.getDef === 'function') { try { d = G.weapons.getDef(wid); } catch (e) { d = null; } }
    if (!d && !G.isStub('items') && typeof G.items.get === 'function') { try { d = G.items.get(wid); } catch (e) { d = null; } }
    let w = d ? (d.weapon || d) : null;
    const fb = C.FALLBACK_WEAPONS[wid];
    if (!w || typeof w.damage !== 'number') { if (!fb) return null; w = fb; }
    const melee = !!(w.melee || w.skill === 'melee_weapons' || w.skill === 'unarmed' || w.reloadType === 'melee');
    const proj = w.projectile && w.projectile !== 'hitscan' ? w.projectile : null;
    let beam = w.beamColor || null;
    if (!beam && w.damageType === 'energy' && !proj) beam = /plasma/.test(wid) ? 'green' : 'red';
    return {
      id: wid, name: (d && d.name) || wid, damage: w.damage || 5, rof: w.rof || 1, automatic: !!w.automatic, magSize: w.magSize || 0,
      reloadTime: w.reloadTime || 2, spread: w.spread !== undefined ? w.spread : 0.03, adsSpread: w.adsSpread !== undefined ? w.adsSpread : (w.spread || 0.03) * 0.4,
      range: melee ? (w.meleeRange || w.range || 2) : (w.range || 100), pellets: w.pellets || 1, projectile: proj, projectileSpeed: w.projectileSpeed || 40,
      explosionRadius: w.explosionRadius || 0, explosionKind: w.explosionKind || null, sound: w.sound || (melee ? 'melee_swing' : 'gun_pistol'),
      sfxPitch: w.sfxPitch || 1, damageType: w.damageType || 'ballistic', melee, beam, skill: w.skill || 'guns', critMult: w.critMult || 1,
      critChance: w.critChance || 1, apCost: w.apCost || 20, ammo: w.ammo || null, meleeArc: w.meleeArc || 1, flash: w.flash || null,
    };
  };
  /** Pick a weapon for an NPC of this pool/level; returns an id that resolves. */
  C.pickWeapon = function (poolKey, level) {
    const pool = C.WEAPON_POOLS[poolKey]; if (!pool) return null;
    const pairs = pool(level || 1).filter((p) => C.weaponInfo(p[0]));
    if (!pairs.length) return null;
    return C.weighted(pairs);
  };
  function weaponModel(wid, info) {
    if (!G.isStub('weapons') && typeof G.weapons.createWorldModel === 'function') {
      try { const m = G.weapons.createWorldModel(wid); if (m) return m; } catch (e) { G.reportError('combat:createWorldModel ' + wid, e); }
    }
    return C.placeholderWeaponModel(wid, info);
  }

  // ------------------------------------------------------------------
  // Actor
  // ------------------------------------------------------------------
  class Actor {
    constructor(def) {
      const type = def.type || 'human';
      const td = C.typeDef(type);
      this.uid = ++uid;
      this.def = def;
      this.id = def.id || ('actor_' + this.uid);
      this.stableId = !!def.id;
      this.type = type;
      this.td = td;
      this.level = Math.max(1, Math.round(def.level || 1));
      this.faction = def.faction || td.faction || 'settlers';
      this.name = def.name || C.rankName(type, this.level, def) || td.name || type;
      this.voice = def.voice !== undefined ? def.voice : td.voice;
      this.customBarks = def.barks || null;
      this.essential = !!def.essential;
      this.dialogue = def.dialogue || null;
      this.merchant = def.merchant || null;
      this.ai = def.ai || (td.human ? 'wander' : 'wander');
      this.home = def.home ? { x: def.home.x, z: def.home.z, r: def.home.r || 10 } : { x: def.pos.x, z: def.pos.z, r: def.ai === 'guard' ? 2 : 12 };
      this.patrol = def.patrol || null;
      this.patrolIdx = 0;
      this.isRobot = !!td.isRobot; this.isCreature = !!td.isCreature; this.gore = td.gore || 'red';
      // Stats (scaled by level)
      const hp = def.hp || Math.round(td.hp + (td.hpLvl || 5) * (this.level - 1));
      this.maxHp = hp; this.hp = hp;
      this.dr = def.dr !== undefined ? def.dr : (td.dr || 0);
      this.dt = td.dt || 0;
      this.perception = td.per || 5;
      this.sight = def.aggroRange || td.sight || 50;
      this.walkSpeed = (def.speed ? def.speed * 0.4 : td.walk) || 1.6;
      this.runSpeed = def.speed || td.run || 5;
      this.damageMult = def.damage || 1;
      this.accuracy = td.accuracy || 1;
      this.xp = def.xp || td.xp || 10; // base XP; rpg scales it by level on actorKilled
      this.dead = false;
      this.hostOverride = def.hostile === undefined ? undefined : !!def.hostile;
      this.inventory = [];
      this.loot = def.loot || null;
      // Limbs: each targetable part has its own condition (crippling)
      this.limbs = {};
      // AI state
      this.state = 'idle'; this.stateT = 0;
      this.target = null; this.lastKnown = null; this.awareness = 0; this.detectedPlayer = false;
      this.inCombat = false; this.fleeing = false;
      this.moveGoal = null; this.followTarget = null;
      this.yaw = def.yaw || 0;
      this.vy = 0;
      this.onDeath = def.onDeath || null; this.onTalk = def.onTalk || null;
      this.noRespawn = !!def.noRespawn;
      this.group = def._group || null;   // population group (location / encounter)
      this.roaming = !!def._roaming;
      this._t = Math.random() * 10;      // per-actor timer offset (staggering)
      // Character
      this.char = createCharacter(type, Object.assign({}, def, { level: this.level }));
      this.height = this.char.height || td.height || 1.8;
      this.radius = Math.max(0.25, Math.min(1.6, this.char.radius || td.radius || 0.35));
      this.pos = this.char.root.position;
      this.char.root.userData.actor = this;
      this.hover = def.hover !== undefined ? def.hover : (td.hover || 0);
      this.flying = this.hover > 0;
      // G.chars builds the hover height into the rig itself: keep the root on the ground then
      if (!this.char.placeholder && this.hover && def.hover === undefined) {
        let ci = null; try { ci = G.chars.info(type); } catch (e) { ci = null; }
        if (ci && ci.hover) this.hover = 0;
      }
      const y0 = def.pos.y !== undefined ? def.pos.y : 0;
      this.pos.set(def.pos.x, y0, def.pos.z);
      if (!this.hover) this.pos.y = G.world.groundHeight(def.pos.x, def.pos.z, y0 + 0.6, this.radius * 0.5);
      else this.pos.y = G.world.groundHeight(def.pos.x, def.pos.z, y0 + 0.6, 0.2) + this.hover;
      this.applyYaw();
      this.char.root.traverse((o) => { if (o.isMesh) o.userData.actor = this; });
      // Weapon
      this.weaponId = null; this.weapon = null; this.weaponModel = null; this.magLeft = 0;
      this.natural = td.natural ? C.NATURAL[td.natural] : null;
      this._resolveWeapon(def);
      // Inventory & loot
      this._buildInventory(def);
      this.inScene = false;
      this.hitParts = null;
      this._anim = null;
    }

    // ---- basic helpers --------------------------------------------------
    applyYaw() { this.char.root.rotation.y = this.yaw + (this.char.yawOffset || 0); }
    get alive() { return !this.dead; }
    forward(out) { return (out || new THREE.Vector3()).set(-Math.sin(this.yaw), 0, -Math.cos(this.yaw)); }
    /** Eye/head position (world). */
    eyePos(out) {
      out = out || new THREE.Vector3();
      const h = this.char.parts && this.char.parts.head;
      if (h && this.inScene && !this.dead) { h.getWorldPosition(out); out.y += 0.08; return out; }
      return out.set(this.pos.x, this.pos.y + this.height * 0.9, this.pos.z);
    }
    centerPos(out) { return (out || new THREE.Vector3()).set(this.pos.x, this.pos.y + (this.dead ? 0.3 : this.height * 0.55), this.pos.z); }
    distTo(p) { return Math.hypot(p.x - this.pos.x, p.z - this.pos.z); }
    distToPlayer() { const p = G.player.pos; return Math.hypot(p.x - this.pos.x, p.z - this.pos.z); }

    /** Hostile to the player? */
    hostile() {
      if (this.dead) return false;
      if (this.frenzied || this._zoneAggro) return true; // _zoneAggro: town guards answering gunfire (45_safezone)
      if (this.hostOverride !== undefined) return this.hostOverride;
      return G.combat.relation(this.faction, 'player') === 'hostile';
    }
    setHostile(b) {
      const was = this.hostile();
      this.hostOverride = b === undefined ? undefined : !!b;
      // stood down (truce, surrender, a visitor pass…): leave combat properly — clearing only the target left
      // `inCombat` set with the actor idle, so canTalk() stayed false for good (untalkable quest NPCs)
      if (!this.hostile() && this.target === 'player') { if (C.leaveCombat) C.leaveCombat(this, 'idle'); else { this.target = null; this.inCombat = false; this.setState('idle'); } }
      if (!was && this.hostile() && this.dialogue && G.dialogue && !G.isStub('dialogue') && G.dialogue.isActive && G.dialogue.isActive()) {
        try { G.dialogue.end(); } catch (e) { /* ignore */ }
      }
    }
    /** Is `other` (actor | 'player') an enemy of this actor? */
    isEnemy(other) {
      if (other === 'player') return this.hostile();
      if (!other || other.dead || other === this) return false;
      if (this.frenzied || other.frenzied) return true;
      // Town guards (45_safezone): anything player-hostile inside their zone is a target, and it shoots back
      if (this.safeGuard && C.safeIntruder && C.safeIntruder(this, other)) return true;
      if (other.safeGuard && C.safeIntruder && C.safeIntruder(other, this)) return true;
      // Player-aligned followers fight whatever the player fights
      const fa = this.following === 'player' ? 'player' : this.faction;
      const fb = other.following === 'player' ? 'player' : other.faction;
      if (fa === 'player' && other.hostile()) return true;
      if (fb === 'player' && this.hostile()) return true;
      return G.combat.relation(fa, fb) === 'hostile';
    }

    setState(s) { if (this.state === s) return; this.prevState = this.state; this.state = s; this.stateT = 0; }
    damage(amount, info) { return C.applyDamage(this, amount, info || {}); }
    kill(info) { if (this.dead) return; C.killActor(this, info || {}); }
    heal(n) { if (this.dead) return; this.hp = Math.min(this.maxHp, this.hp + n); }

    /** Floating subtitle bark. */
    say(text, dur) { if (!text || this.dead) return; C.showBark(this, text, dur); }
    bark(cat, chance) {
      if (chance !== undefined && Math.random() > chance) return;
      const now = G.time.elapsed;
      if (this._lastBark && now - this._lastBark < 4 && cat !== 'allyDown' && cat !== 'flee') return;
      let line = null;
      if (this.customBarks) {
        if (Array.isArray(this.customBarks)) { if (cat === 'idle' || cat === 'greet') line = C.pick(this.customBarks); }
        else if (this.customBarks[cat]) line = C.pick(this.customBarks[cat]);
      }
      if (!line) line = C.bark(this.voice, cat);
      if (line) { this._lastBark = now; this.say(line); }
      // creatures/robots vocalise with sounds instead
      const snd = this.td.sounds && (cat === 'alert' ? this.td.sounds.alert : cat === 'hurt' ? this.td.sounds.hurt : cat === 'idle' ? this.td.sounds.idle : null);
      if (snd && !line) C.sfx(snd, { pos: this.pos, volume: 0.8, pitch: this.type === 'super_mutant_behemoth' || this.type === 'mirelurk_queen' ? 0.6 : 0.9 + Math.random() * 0.2 });
      else if (snd && (this.isRobot || this.type === 'super_mutant') && cat === 'alert') C.sfx(snd, { pos: this.pos, volume: 0.7 });
    }

    moveTo(x, z) { this.moveGoal = { x, z }; if (this.state === 'idle' || this.state === 'follow') this.setState('move'); }
    follow(t) { this.followTarget = t || null; this.following = t === 'player' || t === G.player ? 'player' : (t || null); if (t) this.setState('follow'); else this.setState('idle'); }
    stop() { this.moveGoal = null; if (this.state === 'move') this.setState('idle'); }
    faceTowards(x, z, rate, dt) {
      const want = Math.atan2(-(x - this.pos.x), -(z - this.pos.z));
      if (!rate || !dt) this.yaw = want;
      else this.yaw += C.clamp(G.util.angleDiff(this.yaw, want), -rate * dt, rate * dt);
      this.applyYaw();
    }

    canTalk() {
      if (this.dead || this.knockedOut || this.hostile() || this.inCombat) return false;
      if (this.onTalk || this.dialogue || this.merchant) return true;
      return !this.isCreature && !this.isRobot && this.type !== 'eyebot';
    }
    talk() {
      this.faceTowards(G.player.pos.x, G.player.pos.z);
      if (this.onTalk) { try { this.onTalk(this); } catch (e) { G.reportError('combat:onTalk', e); } return; }
      if (this.dialogue && !G.isStub('dialogue') && typeof G.dialogue.start === 'function') { G.dialogue.start(this.dialogue, this); return; }
      if (this.merchant && !G.isStub('barter') && typeof G.barter.open === 'function') {
        this.say(C.bark(this.voice === 'merchant' ? 'merchant' : 'merchant', 'greet'), 3);
        G.barter.open(this); return;
      }
      let line = null;
      if (this.customBarks) line = Array.isArray(this.customBarks) ? C.pick(this.customBarks) : (this.customBarks.greet ? C.pick(this.customBarks.greet) : null);
      // Faction armour on the player changes how people greet you (FO3/NV)
      if (!line) { const af = C.playerArmorFaction(); const t = af && C.ARMOR_GREET[af]; const v = this.voice || (this.faction === 'enclave' ? 'enclave' : this.faction === 'brotherhood' ? 'brotherhood' : null);
        if (t && v && t[v] && Math.random() < 0.8) line = C.pick(t[v]); }
      if (!line) line = C.bark(this.voice, 'greet') || C.bark('settler', 'greet');
      this.say(line, 4.5);
      if (this.char.playOneShot) try { this.char.playOneShot('talk'); } catch (e) { /* ignore */ }
    }
    /** FO3: activating an unaware NPC while sneaking opens their pockets. */
    canPickpocket() {
      return !this.dead && G.player.sneaking && !this.hostile() && !this.inCombat && this.td.human && !this.talking && !this.knockedOut && this.following !== 'player'
        && !G.isStub('containers') && !G.isStub('ui') && typeof G.containers.fromActor === 'function';
    }
    pickpocket() {
      let c = null;
      try { c = this.container || G.containers.fromActor(this); } catch (e) { G.reportError('combat:pickpocket', e); }
      if (!c) { this.talk(); return; }
      this.container = c;
      const actor = this;
      c.owner = this.faction; c.pickpocket = true;
      c.onTake = function (id, n) {
        if (actor.dead) return;
        if (id === actor.weaponId) actor.equipWeapon(null);
        let d = null; try { d = G.items.get(id); } catch (e) { d = null; }
        const value = (d ? d.value || 1 : 1) * (n || 1);
        const sneak = C.skill('sneak', 15);
        const chance = C.clamp(0.06 + value / 260 + (d ? (d.weight || 0) : 1) * 0.025 - sneak / 260, 0.03, 0.95);
        if (Math.random() < chance) {
          G.notify('You were caught pickpocketing!');
          C.sfx('sneak_detected', { volume: 0.9 });
          actor.say(C.pick(['Hey! Get your hands out of my pockets!', 'Thief! THIEF!', 'You picked the wrong pocket, pal.', 'Do I look like a vending machine to you?!']), 3.5);
          if (!actor.essential) actor.setHostile(true);
          actor.awareness = 1;
          if (G.state === 'container') G.setState('play');
          if (actor.hostile()) C.enterCombat(actor, 'player');
          G.events.emit('crime', { type: 'pickpocket', owner: actor.faction, value, witness: actor });
        }
      };
      G.ui.openContainer(c);
    }
    /** Loot the corpse. */
    search() {
      if (!this.container && !G.isStub('containers') && typeof G.containers.fromActor === 'function') {
        try { this.container = G.containers.fromActor(this); } catch (e) { G.reportError('combat:fromActor', e); }
      }
      if (this.container && !G.isStub('ui') && typeof G.ui.openContainer === 'function') { G.ui.openContainer(this.container); return; }
      // Fallback: take everything straight into the inventory
      if (!G.isStub('inventory') && this.inventory.length) {
        for (const [id, n] of this.inventory) { try { if (id === 'caps' && G.inventory.addCaps) G.inventory.addCaps(n); else G.inventory.add(id, n); } catch (e) { /* ignore */ } }
        this.inventory = [];
        C.sfx('pickup');
      } else G.notify(this.name + ': nothing of value.');
    }
    interactPos() {
      if (this.dead) return _v2.set(this.pos.x, this.pos.y + 0.35, this.pos.z);
      return _v2.set(this.pos.x, this.pos.y + Math.min(this.height * 0.8, 1.7), this.pos.z);
    }

    // ---- weapon ---------------------------------------------------------
    _resolveWeapon(def) {
      let wid = def.weapon || null;
      if (wid === 'none') wid = null;
      else if (!wid && !this.isCreature && !this.isRobot && def.weapon === undefined) {
        const poolKey = def._pool !== undefined ? def._pool : (this.type === 'raider' ? 'raider' : this.type === 'super_mutant' ? 'super_mutant' : this.type === 'enclave_soldier' ? 'enclave_soldier'
          : this.type === 'brotherhood_paladin' ? 'brotherhood_paladin' : null);
        if (poolKey) wid = C.pickWeapon(poolKey, this.level);
      }
      this.equipWeapon(wid);
    }
    equipWeapon(wid) {
      if (this.weaponModel && this.char.setWeapon) { try { this.char.setWeapon(null); } catch (e) { /* ignore */ } }
      this.weaponModel = null;
      const info = wid ? C.weaponInfo(wid) : null;
      if (!info) { this.weaponId = null; this.weapon = null; if (!this.natural && !this.isCreature) this.natural = C.NATURAL.fists; return; }
      this.weaponId = wid; this.weapon = info; this.magLeft = info.magSize || 0;
      if (this.natural === C.NATURAL.fists) this.natural = null;
      try {
        const m = weaponModel(wid, info);
        if (m && this.char.setWeapon) { this.char.setWeapon(m); this.weaponModel = m; m.traverse((o) => { o.userData.actor = this; }); }
      } catch (e) { G.reportError('combat:setWeapon', e); }
      this.hitParts = null; // recompute (weapon part)
    }
    /** Knock the weapon out of the actor's hands (crippled arm / weapon shot). */
    disarm() {
      if (!this.weaponId || this.dead) return;
      const wid = this.weaponId;
      const p = this.weaponModel ? this.weaponModel.getWorldPosition(new THREE.Vector3()) : this.centerPos();
      if (G.isStub('items') && !this.char.placeholder && typeof this.char.dropWeapon === 'function') {
        try { this.char.dropWeapon(); } catch (e) { /* ignore */ }
        this.weaponModel = null;
      }
      this.equipWeapon(null);
      // Remove from its inventory and drop it in the world as a pickup
      const i = this.inventory.findIndex((e) => e[0] === wid); if (i >= 0) this.inventory.splice(i, 1);
      if (!G.isStub('items') && typeof G.items.spawnPickup === 'function') {
        p.y = Math.max(p.y - 0.8, G.world.groundHeight(p.x, p.z, p.y) + 0.05);
        try { const pk = G.items.spawnPickup(wid, 1, p); if (pk) this._droppedGun = { p: pk, wid }; } catch (e) { G.reportError('combat:spawnPickup', e); }
      }
      this.bark('disarmed', 0.9);
      this.disarmed = true; this._lostWeapon = wid; this._rearm = null;
    }
    /** World position of the weapon muzzle (or a sensible guess). */
    muzzlePos(out) {
      out = out || new THREE.Vector3();
      if (this.inScene && !this.char.placeholder && typeof this.char.getMuzzle === 'function') {
        try { const m = this.char.getMuzzle(out, _v1); if (m && m.pos) return out; } catch (e) { /* fall through */ }
      }
      if (this.weaponModel && this.inScene) {
        const mz = this.weaponModel.getObjectByName('muzzle');
        if (mz) return mz.getWorldPosition(out);
        this.weaponModel.getWorldPosition(out);
        const f = this.forward(_v1); out.addScaledVector(f, 0.45);
        return out;
      }
      const f = this.forward(_v1);
      out.set(this.pos.x + f.x * (this.radius + 0.25), this.pos.y + this.height * (this.hover ? 0.5 : 0.78), this.pos.z + f.z * (this.radius + 0.25));
      // offset to the right hand
      out.x += -f.z * 0.18; out.z += f.x * 0.18;
      return out;
    }

    // ---- inventory ------------------------------------------------------
    _buildInventory(def) {
      const inv = [];
      const add = (id, n) => {
        if (!id || n <= 0) return;
        if (!G.isStub('items') && typeof G.items.get === 'function' && id !== 'caps' && !G.items.get(id)) return;
        const e = inv.find((x) => x[0] === id); if (e) e[1] += n; else inv.push([id, n]);
      };
      if (def.inventory) for (const [id, n] of def.inventory) add(id, n || 1);
      if (this.weaponId) {
        add(this.weaponId, 1);
        const a = this.weapon && this.weapon.ammo;
        if (a && a !== this.weaponId) add(a, C.randInt(3, Math.max(4, Math.min(24, (this.weapon.magSize || 8)))));
      }
      if (!def.noLoot) {
        const L = C.LOOT[this.type] || (this.td.human ? C.LOOT.human : null);
        // rpg loot tables are keyed by actor type (raider, super_mutant, deathclaw…); fall back to our hints
        let rolled = false;
        if (!G.isStub('items') && typeof G.items.rollLoot === 'function') {
          for (const table of [def.loot, this.type, L && L.table].filter(Boolean)) {
            try { const r = G.items.rollLoot(table, this.level); if (Array.isArray(r) && r.length) { for (const [id, n] of r) add(id, n); rolled = true; break; } } catch (e) { /* table may not exist */ }
          }
        }
        if (!rolled && L && L.extra) {
          for (const ex of L.extra) { const id = ex[0], ch = ex[1], mn = ex[2], mx = ex[3]; if (Math.random() < ch) add(id, mn ? C.randInt(mn, mx || mn) : 1); }
        }
      }
      this.inventory = inv;
    }

    // ---- scene streaming --------------------------------------------------
    attach() {
      if (this.inScene) return;
      G.scene.add(this.char.root); this.inScene = true;
    }
    detach() {
      if (!this.inScene) return;
      if (this.char.root.parent) this.char.root.parent.remove(this.char.root);
      this.inScene = false;
      C.hideBark(this);
    }

    // ---- animation --------------------------------------------------------
    anim(name, speed) {
      if (this._anim === name && Math.abs((this._animSpeed || 1) - (speed || 1)) < 0.15) return;
      this._anim = name; this._animSpeed = speed || 1;
      try { this.char.setAnim(name, { loop: name !== 'death', speed: speed || 1 }); } catch (e) { /* ignore */ }
    }
    oneShot(name) { try { if (this.char.playOneShot) this.char.playOneShot(name); } catch (e) { /* ignore */ } }

    dispose() {
      this.detach();
      if (this.interactable) G.world.removeInteractable(this.interactable);
      if (this.radSource) G.world.removeRadSource(this.radSource);
      if (this.burnFx) C.stopBurnFx(this);
      if (this.pile && this.pile.parent) this.pile.parent.remove(this.pile);
      try { if (this.char.dispose) this.char.dispose(); } catch (e) { /* ignore */ }
      C.clearHighlight && C.clearHighlight(this);
    }
  }
  C.Actor = Actor;

  // ------------------------------------------------------------------
  // Hit capsules (per part). Computed lazily from the character's meshes,
  // in each part's local space so they follow the animation.
  // ------------------------------------------------------------------
  function partMeshes(obj, partSet, out) {
    if (obj.isMesh && obj.visible !== false) out.push(obj);
    for (const c of obj.children) if (!partSet.has(c)) partMeshes(c, partSet, out);
    return out;
  }
  const _box = new THREE.Box3(), _b2 = new THREE.Box3(), _inv = new THREE.Matrix4();
  C.computeHitParts = function (actor) {
    const ch = actor.char;
    ch.root.updateMatrixWorld(true);
    const parts = ch.parts || {};
    const partSet = new Set(Object.values(parts).filter(Boolean));
    const out = [];
    const markerMode = !ch.placeholder && typeof ch.raycast === 'function';
    const keys = Object.keys(parts).sort((p, q) => (p === 'torso' ? -1 : q === 'torso' ? 1 : 0));
    const seenObj = new Set();
    for (const key of keys) {
      const obj = parts[key]; if (!obj || !obj.isObject3D) continue;
      if (seenObj.has(obj)) continue; // G.chars aliases missing parts to the torso marker
      seenObj.add(obj);
      if (markerMode) {
        const r = (obj.userData && obj.userData.radius) || 0.2;
        out.push({ key, obj, a: new THREE.Vector3(), b: new THREE.Vector3(), r, wa: new THREE.Vector3(), wb: new THREE.Vector3(), label: C.partLabel(actor.type, key, ch), meshes: [] });
        continue;
      }
      const meshes = partMeshes(obj, partSet, []);
      _inv.copy(obj.matrixWorld).invert();
      _box.makeEmpty();
      for (const m of meshes) {
        const g = m.geometry; if (!g) continue;
        if (!g.boundingBox) g.computeBoundingBox();
        _b2.copy(g.boundingBox).applyMatrix4(_m.multiplyMatrices(_inv, m.matrixWorld));
        _box.union(_b2);
      }
      let a, b, r;
      if (_box.isEmpty()) {
        a = new THREE.Vector3(); b = new THREE.Vector3(); r = Math.max(0.08, actor.height * 0.08);
      } else {
        const size = _box.getSize(new THREE.Vector3()), c = _box.getCenter(new THREE.Vector3());
        // scale factor of the part (world) so radii are in world units
        const sc = new THREE.Vector3(); obj.matrixWorld.decompose(_v1, new THREE.Quaternion(), sc);
        const ax = size.x >= size.y && size.x >= size.z ? 'x' : size.y >= size.z ? 'y' : 'z';
        const others = ['x', 'y', 'z'].filter((k) => k !== ax);
        r = Math.max(0.05, Math.max(size[others[0]], size[others[1]]) * 0.5 * 0.92);
        const half = Math.max(0, size[ax] * 0.5 - r);
        a = c.clone(); b = c.clone(); a[ax] -= half; b[ax] += half;
        r *= (sc.x + sc.y + sc.z) / 3;
      }
      out.push({ key, obj, a, b, r, wa: new THREE.Vector3(), wb: new THREE.Vector3(), label: C.partLabel(actor.type, key, ch), meshes });
    }
    // Virtual 'weapon' part (shoot it out of their hands)
    if (actor.weaponModel) {
      const m = actor.weaponModel;
      _box.setFromObject(m);
      if (!_box.isEmpty()) {
        _inv.copy(m.matrixWorld).invert();
        const lb = _box.clone().applyMatrix4(_inv);
        const size = lb.getSize(new THREE.Vector3()), c = lb.getCenter(new THREE.Vector3());
        const r = Math.max(0.05, Math.min(size.x, size.y) * 0.5 + 0.02);
        const a = c.clone(), b = c.clone(); const half = Math.max(0, size.z * 0.5 - r); a.z -= half; b.z += half;
        out.push({ key: 'weapon', obj: m, a, b, r: Math.max(r, 0.06), wa: new THREE.Vector3(), wb: new THREE.Vector3(), label: 'Weapon', virtual: true });
      }
    }
    actor.hitParts = out;
    // Limb conditions (Fallout: each limb has its own health)
    for (const hp of out) {
      if (hp.virtual || actor.limbs[hp.key]) continue;
      const frac = hp.key === 'torso' ? 1.0 : hp.key === 'head' ? 0.45 : 0.5;
      actor.limbs[hp.key] = { hp: actor.maxHp * frac, max: actor.maxHp * frac, crippled: false };
    }
    return out;
  };
  /** Update world-space capsule endpoints (call once per frame per actor before tests). */
  C.updateHitParts = function (actor) {
    if (!actor.hitParts) C.computeHitParts(actor);
    if (actor._hpStamp === G.time.frame) return actor.hitParts;
    actor._hpStamp = G.time.frame;
    actor.char.root.updateMatrixWorld(true);
    for (const hp of actor.hitParts) {
      hp.wa.copy(hp.a).applyMatrix4(hp.obj.matrixWorld);
      hp.wb.copy(hp.b).applyMatrix4(hp.obj.matrixWorld);
    }
    return actor.hitParts;
  };
  C.partCenter = function (actor, key, out) {
    out = out || new THREE.Vector3();
    const parts = C.updateHitParts(actor);
    const hp = parts.find((p) => p.key === key);
    if (!hp) return actor.centerPos(out);
    return out.addVectors(hp.wa, hp.wb).multiplyScalar(0.5);
  };

  // ------------------------------------------------------------------
  // Floating barks (DOM subtitles above heads)
  // ------------------------------------------------------------------
  let barkLayer = null;
  const barks = new Map(); // actor → {el, t, dur}
  function ensureLayer() {
    if (barkLayer) return barkLayer;
    barkLayer = document.createElement('div');
    barkLayer.className = 'cmb-barks';
    document.body.appendChild(barkLayer);
    return barkLayer;
  }
  C.showBark = function (actor, text, dur) {
    if (!actor.inScene || G.SMOKE) { return; }
    if (actor.distToPlayer() > 40) return;
    ensureLayer();
    let b = barks.get(actor);
    if (!b) { const el = document.createElement('div'); el.className = 'cmb-bark'; barkLayer.appendChild(el); b = { el }; barks.set(actor, b); }
    b.el.innerHTML = '<span class="cmb-bark-name"></span><span class="cmb-bark-text"></span>';
    b.el.firstChild.textContent = actor.name;
    b.el.lastChild.textContent = text;
    b.t = 0; b.dur = dur || Math.min(6, 2.2 + text.length * 0.05);
    b.el.style.opacity = '0';
  };
  C.hideBark = function (actor) { const b = barks.get(actor); if (b) { b.el.remove(); barks.delete(actor); } };
  C.clearBarks = function () { for (const b of barks.values()) b.el.remove(); barks.clear(); };
  const _proj = new THREE.Vector3();
  C.updateBarks = function (realDt) {
    if (!barks.size) return;
    const hideAll = !(G.state === 'play' || G.state === 'vats' && C.vatsExecuting);
    const w = window.innerWidth, h = window.innerHeight;
    const shown = [];
    for (const [actor, b] of barks) {
      b.t += realDt;
      if (b.t > b.dur || !actor.inScene || actor.dead && b.t > 0.5) { b.el.remove(); barks.delete(actor); continue; }
      _proj.set(actor.pos.x, actor.pos.y + actor.height + 0.35, actor.pos.z).project(G.camera);
      const d = actor.distToPlayer();
      if (hideAll || _proj.z > 1 || _proj.z < -1 || d > 45) { b.el.style.opacity = '0'; continue; }
      const sc = C.clamp(1.15 - d / 60, 0.7, 1.1);
      // true on-screen size: unscaled layout box × our scale (scale is anchored bottom-centre)
      const bw = (b.el.offsetWidth || 200) * sc, bh = (b.el.offsetHeight || 34) * sc;
      shown.push({ b, d, sc, x: (_proj.x * 0.5 + 0.5) * w, y: (-_proj.y * 0.5 + 0.5) * h, bw, bh,
        fade: Math.min(1, b.t * 5, (b.dur - b.t) * 2) * C.clamp(1.3 - d / 40, 0.35, 1) });
    }
    // At most 3 subtitles at once (nearest speakers) — Fallout never floods the screen
    shown.sort((p, q) => p.d - q.d);
    for (let i = 3; i < shown.length; i++) shown[i].b.el.style.opacity = '0';
    if (shown.length > 3) shown.length = 3;
    // Clamp into the screen, then stack: each later (farther) bark climbs above whatever it overlaps
    const M = 8;
    const clampX = (S) => { S.x = C.clamp(S.x, S.bw / 2 + M, w - S.bw / 2 - M); S.y = C.clamp(S.y, S.bh + M, h - M); };
    for (let i = 0; i < shown.length; i++) {
      const A = shown[i]; clampX(A);
      for (let pass = 0; pass < 6; pass++) {
        let moved = false;
        for (let j = 0; j < i; j++) {
          const B = shown[j];
          // boxes: x centred, y = bottom edge
          if (Math.abs(A.x - B.x) < (A.bw + B.bw) / 2 + 6 && A.y > B.y - B.bh - 4 && A.y - A.bh < B.y + 4) { A.y = B.y - B.bh - 6; moved = true; }
        }
        if (!moved) break;
      }
      if (A.y - A.bh < M) { A.y = A.bh + M; A.b.el.style.opacity = '0'; A.hidden = true; }
    }
    for (const S of shown) {
      if (S.hidden) continue;
      S.b.el.style.opacity = S.fade.toFixed(2);
      S.b.el.style.transformOrigin = '50% 100%';
      S.b.el.style.transform = 'translate(' + Math.round(S.x) + 'px,' + Math.round(S.y) + 'px) translate(-50%,-100%) scale(' + S.sc.toFixed(2) + ')';
    }
  };

  const AP = G.actors;
  /** Spawn an actor. See CONTRACT §2 combat for def fields. */
  AP.spawn = function (def) {
    if (!def || !def.pos) { G.reportError('combat:spawn', new Error('actor def needs pos')); return null; }
    if (def.id && C.byId.has(def.id)) {
      // Re-spawning a stable id replaces the old instance (e.g. quest reload)
      AP.remove(C.byId.get(def.id));
    }
    let a;
    try { a = new Actor(def); } catch (e) { G.reportError('combat:spawn ' + def.type, e); return null; }
    C.list.push(a);
    if (a.stableId) C.byId.set(a.id, a);
    // Talk / Search interactable
    a.interactable = G.world.addInteractable({
      getPos: () => a.interactPos(),
      radius: Math.max(0.55, Math.min(1.4, a.radius + 0.25)),
      label: () => a.name,
      verb: () => (a.dead ? 'Search' : a.canPickpocket() ? 'Pickpocket' : 'Talk'),
      enabled: () => a.inScene && (a.dead ? !a.noSearch : (a.canTalk() || a.canPickpocket())),
      onInteract: () => (a.dead ? a.search() : a.canPickpocket() ? a.pickpocket() : a.talk()),
      priority: 0.3,
      actor: a,
    });
    // Saved state for stable ids (applied after load)
    if (a.stableId && C.pendingState.has(a.id)) { C.applySavedActorState(a, C.pendingState.get(a.id)); C.pendingState.delete(a.id); }
    // Glowing ones carry a radiation aura
    if (a.td.glowing && !a.dead) a.radSource = G.world.addRadSource({ x: a.pos.x, y: a.pos.y + 1, z: a.pos.z, radius: 7, strength: 3 });
    // Initial AI setup
    if (C.initAI) C.initAI(a, def);
    if (a.distToPlayer() < C.ACTIVE_R || def.alwaysActive) a.attach();
    return a;
  };
  AP.list = () => C.list;
  AP.get = (id) => C.byId.get(id) || C.list.find((a) => a.id === id) || null;
  AP.find = (pred) => C.list.filter(pred);
  AP.nearest = function (pos, maxDist, pred) {
    let best = null, bd = maxDist === undefined ? 1e9 : maxDist;
    for (const a of C.list) { if (pred && !pred(a)) continue; const d = Math.hypot(a.pos.x - pos.x, a.pos.z - pos.z); if (d < bd) { bd = d; best = a; } }
    return best;
  };
  AP.inRadius = (pos, r) => C.list.filter((a) => Math.hypot(a.pos.x - pos.x, a.pos.z - pos.z) <= r);
  AP.remove = function (a) {
    if (!a) return;
    const i = C.list.indexOf(a); if (i >= 0) C.list.splice(i, 1);
    if (a.stableId && C.byId.get(a.id) === a) C.byId.delete(a.id);
    a.dispose();
    a.removed = true;
  };
  AP.clear = function () { for (const a of C.list.slice()) AP.remove(a); C.list.length = 0; C.byId.clear(); C.clearBarks(); };
  AP.count = () => C.list.length;
  AP.alive = () => C.list.filter((a) => !a.dead);
  /** Hostile (to player), living actors in radius. */
  AP.hostilesNear = (pos, r) => C.list.filter((a) => !a.dead && a.hostile() && Math.hypot(a.pos.x - pos.x, a.pos.z - pos.z) <= r);
  /** Actors currently fighting the player (for HUD compass ticks). */
  AP.inCombatWithPlayer = () => C.list.filter((a) => !a.dead && a.inCombat && a.target === 'player');
  AP.types = () => Object.keys(C.TYPES);
  AP.typeInfo = (t) => C.TYPES[t] || null;
  AP.Actor = Actor;

  /** 'bos' | 'enclave' | null — faction colours of the body armour the player wears. */
  C.playerArmorFaction = function () {
    if (G.isStub('inventory') || typeof G.inventory.list !== 'function') return null;
    try {
      for (const e of G.inventory.list('apparel') || []) {
        if (!e || !e.equipped) continue;
        const d = e.def || (G.items && G.items.get ? G.items.get(e.id) : null) || {};
        const tags = d.tags || [];
        if (d.outfit === 'enclave_power_armor' || tags.indexOf('enclave') >= 0) return 'enclave';
        if (d.outfit === 'bos_power_armor' || tags.indexOf('brotherhood') >= 0) return 'bos';
      }
    } catch (e) { /* ignore */ }
    return null;
  };

  // Saved state helpers (stable-id actors)
  C.actorSaveState = (a) => ({ dead: a.dead, hp: Math.round(a.hp), x: +a.pos.x.toFixed(2), y: +a.pos.y.toFixed(2), z: +a.pos.z.toFixed(2), yaw: +a.yaw.toFixed(2),
    host: a.hostOverride, inv: a.dead ? a.inventory : undefined, follow: a.following === 'player' ? 'player' : undefined, name: a.name, ff: a.wasFriendly || undefined });
  C.applySavedActorState = function (a, s) {
    if (!s) return;
    if (typeof s.x === 'number') { a.pos.set(s.x, s.y || 0, s.z); a.yaw = s.yaw || 0; a.applyYaw(); }
    if (s.host !== undefined) a.hostOverride = s.host === null ? undefined : s.host;
    if (s.ff) a.wasFriendly = true;   // an essential the player angered can still calm down after a load (C.calmEssential)
    if (s.name) a.name = s.name;
    if (s.dead) { a.hp = 0; C.killActor(a, { silent: true, restore: true }); if (s.inv) a.inventory = s.inv; }
    else if (typeof s.hp === 'number') a.hp = Math.max(1, Math.min(a.maxHp, s.hp));
    if (s.follow === 'player') a.follow('player');
  };
