/* src/quests/60_finale.js */
/* =====================================================================
 * FALLOUT: NEW YORK — QUESTS — 60: FINALE — MQ06 "Give Me Liberty"
 *
 * Liberty Island. Colonel Augustus Crane, Zeke Marlow, and the
 * Knickerbocker Array hidden in the Statue of Liberty's torch.
 * Four outcomes at the torch terminal:
 *   RECLAMATION (purify the sky for everyone)      — good
 *   SELECTIVE   (Enclave Directive 7-A: genocide)   — very evil
 *   CODEX OVERRIDE (hand it to the Brotherhood)    — neutral
 *   OVERLOAD    (destroy it — nobody gets it)      — neutral/grey
 * …then a Fallout-style ending slideshow built from every flag the
 * player set along the way (G.quests.playEnding('main')).
 * ===================================================================== */
(function () {
  'use strict';
  const G = window.G;
  const L = G.quests.lib;
  const Q = G.quests;
  const pname = () => G.player.name || 'the Vault Dweller';
  const faction = () => L.flag('mq_faction') || 'none';
  const LIB = () => G.LAYOUT.LIBERTY;

  // ------------------------------------------------------------------
  // Placement helpers (statue anchors, else a ring around the pedestal)
  // ------------------------------------------------------------------
  function statueSpot(names, angle, dist) {
    const c = LIB();
    const dock = L.spots && L.spots.libertyDock ? L.spots.libertyDock() : { x: c.x + 30, z: c.z - 30 };
    // face the dock side (towards Manhattan)
    const base = Math.atan2(dock.z - c.z, dock.x - c.x) + (angle || 0);
    return { site: 'statue_of_liberty', anchors: names, fallback: { x: c.x + Math.cos(base) * dist, z: c.z + Math.sin(base) * dist }, maxR: 10 };
  }
  function torchTerminalSpot() {
    const a = statueSpot(['torch_control', 'terminal', 'control', 'pedestal_terminal'], 0.35, 20);
    return L.anchor(a.site, a.anchors, a.fallback, { maxR: 10 });
  }

  // ------------------------------------------------------------------
  // Quest
  // ------------------------------------------------------------------
  L.hook('init', function () {
    L.quest({
      id: 'mq06', main: true, order: 6, name: 'Give Me Liberty', location: 'Liberty Island',
      desc: () => 'Zeke is being held on Liberty Island by Colonel Augustus Crane of the Enclave, beneath the Knickerbocker Array hidden in the Statue of Liberty\'s torch. ' +
        'You carry both the Liberty Key and the Ignition Core. ' + (faction() === 'bos' ? 'The Brotherhood of Steel will support your landing.' : faction() === 'enclave' ? 'The Enclave expects you to finish what they started.' : 'You are on your own.'),
      stages: [
        { id: 'ferry', text: () => faction() === 'bos' ? 'Meet the Brotherhood squad at the Battery Park pier and cross to Liberty Island.' : 'Get to Liberty Island. Ossie runs a ferry from the Battery Park pier.',
          markers: () => [{ npc: 'bp_ossie', label: 'Battery Park Ferry' }] },
        { id: 'vertibird', text: 'Board the Enclave Vertibird with Lieutenant Hale at the UN Plaza.', markers: () => [{ npc: 'enc_hale', label: 'Lieutenant Hale' }] },
        { id: 'storm', text: 'Fight your way into the Statue of Liberty\'s pedestal.', markers: () => [{ npc: 'lib_crane', label: 'Colonel Crane', x: LIB().x, z: LIB().z }],
          onEnter() { stormBeach(); } },
        { id: 'crane', text: () => L.flag('crane_fight') ? 'Defeat Colonel Crane.' : 'Confront Colonel Crane in the pedestal command post.', markers: () => [{ npc: 'lib_crane', label: 'Colonel Crane' }] },
        { id: 'zeke', text: 'Free Zeke.', markers: () => [{ npc: 'lib_zeke', label: 'Zeke Marlow' }], onEnter() { L.say('lib_zeke', 'KID? Is that you? Over here!'); } },
        { id: 'torch', text: 'Decide the fate of the Knickerbocker Array at the torch control terminal.',
          markers: () => { const p = L.termPos('liberty_torch'); return p ? [{ x: p.x, z: p.z, label: 'Torch Control' }] : []; } },
        { id: 'cr_talk', text: '(Optional) Talk Colonel Crane into standing down.', sideObjective: true, optional: true,
          showIf: () => false, doneIf: () => !!L.flag('crane_surrendered') && !L.flag('crane_genome') },
        { id: 'cr_genome', text: '(Optional) Show Crane that his own men would fail the reference genome.', sideObjective: true, optional: true,
          showIf: () => false, doneIf: () => !!L.flag('crane_genome') },
        { id: 'cr_kill', text: '(Optional) Defeat Colonel Crane.', sideObjective: true, optional: true,
          showIf: () => false, doneIf: () => !!L.flag('crane_dead') },
        { id: 'cr_ally', text: '(Optional) Stand with Colonel Crane at the torch.', sideObjective: true, optional: true,
          showIf: () => false, doneIf: () => !!L.flag('crane_ally') },
        { id: 'done', text: () => ({
          purify: 'The torch is lit. The sky over Manhattan is clean — for everyone.',
          selective: 'The torch burned green. Directive 7-A has been carried out.',
          brotherhood: 'The Knickerbocker Array is in the hands of the Brotherhood of Steel.',
          destroy: 'The Knickerbocker Array is at the bottom of the harbour.',
        }[L.flag('ending_choice')] || 'The torch is lit.'), hidden: false },
      ],
      rewards: { xp: 1000 },
    });
  });

  // ------------------------------------------------------------------
  // Arrival on the island (ferry, vertibird, swimming — all count)
  // ------------------------------------------------------------------
  L.trigger({ id: 'liberty_arrive', once: false, cooldown: 2,
    cond: () => {
      const s = Q.stage('mq06'); if (s !== 'ferry' && s !== 'vertibird') return false;
      const c = LIB(), p = G.player.pos; return Math.hypot(p.x - c.x, p.z - c.z) < c.r + 28 || (p.x >= G.LAYOUT.INTERIOR_X_MIN && Math.hypot(p.x - 4000, p.z - 1200) < 150);
    },
    fire() {
      if (faction() === 'enclave') {
        Q.setStage('mq06', 'crane');
        L.later(1, () => L.say('lib_crane', 'There you are. Come, come — history is waiting, and it hates to be kept waiting.'));
      } else Q.setStage('mq06', 'storm');
    } });
  // No Hale, no Vertibird: fall back to Ossie's ferry.
  L.trigger({ id: 'vertibird_lost', once: false, cooldown: 3,
    cond: () => Q.stage('mq06') === 'vertibird' && (L.flag('hale_dead') || !L.alive('enc_hale')) && !!L.flag('hale_dead'),
    fire() { Q.setStage('mq06', 'ferry'); L.notify('With Lieutenant Hale gone, the Vertibird isn\'t going anywhere. Ossie\'s ferry, then.'); } });
  // Keep the Enclave garrison friendly on the island for Enclave-path players
  // (local truce zone around the statue; lifts itself if the Colonel is crossed).
  const libertyGuest = () => faction() === 'enclave' && !L.flag('crane_fight') && !L.flag('enclave_hostile');
  function libertyZone() { L.truceZone('liberty_guest', { faction: 'enclave', x: LIB().x, z: LIB().z, r: 200, cond: libertyGuest }); }
  L.hook('init', libertyZone);
  L.hook('reset', libertyZone);

  // Walking up to Crane moves the story on.
  L.trigger({ id: 'crane_near', once: false, cooldown: 2,
    pos: () => { const a = L.npc('lib_crane'); return a && !a.dead ? a.pos : null; }, r: 22,
    cond: () => Q.stage('mq06') === 'storm',
    fire() { Q.setStage('mq06', 'crane'); L.say('lib_crane', 'Hold your fire! HOLD. I want to speak to the tenant.'); } });

  function stormBeach() {
    L.hint('storm', 'The Enclave garrison is dug in around the statue. Use cover, V.A.T.S. and your Stimpaks.', 1);
    if (faction() === 'bos') {
      spawnBoSSquad();
      L.later(1.5, () => L.say('lib_bos_castillo', 'Brotherhood! Advance on the statue! Ad Victoriam!'));
    }
    if (L.flag('hale_defects')) {
      L.later(4, () => {
        L.notify('Every Enclave eyebot on the island drops out of the sky at once. Somewhere across the water, Lieutenant Hale has kept her word.');
        if (!G.isStub('actors') && G.actors.list) for (const a of G.actors.list()) if (!a.dead && a.type === 'eyebot' && Math.hypot(a.pos.x - LIB().x, a.pos.z - LIB().z) < 200) { try { a.kill(); } catch (e) { /* */ } }
      });
    }
    if (!G.isStub('music') && G.music.setMood) try { G.music.setMood('combat'); } catch (e) { /* */ }
  }
  function spawnBoSSquad() {
    const d = L.spots.libertyDock();
    const around = (dx, dz) => ({ fallback: { x: d.x + dx, z: d.z + dz, y: d.y }, maxR: 8 });
    if (!L.flag('stroud_dead')) {
      L.removeNPC('bos_stroud');
      L.spawnNPC({ id: 'bos_stroud', type: 'human', name: 'Paladin Commander Stroud', outfit: 'bos_power_armor', gender: 'm', faction: 'brotherhood', ai: 'follow', level: 14, hp: 400,
        at: around(2, 2), weapon: ['laser_rifle'], dialogue: 'dlg_bos_stroud', essential: true, barks: ['Push forward!', 'Target the heavy armour!', 'Steel stands!'] });
    }
    L.spawnNPC({ id: 'lib_bos_castillo', type: 'human', name: 'Knight Castillo', outfit: 'bos_power_armor', gender: 'f', faction: 'brotherhood', ai: 'follow', level: 10, hp: 300,
      at: around(-2, 3), weapon: ['laser_rifle', 'assault_rifle'], barks: ['Contact!', 'Moving up!', 'For the Codex!'] });
    L.spawnNPC({ id: 'lib_bos_albright', type: 'human', name: 'Knight-Sergeant Albright', outfit: 'bos_power_armor', gender: 'm', faction: 'brotherhood', ai: 'follow', level: 11, hp: 320,
      at: around(3, -2), weapon: ['minigun', 'laser_rifle'], barks: ['Suppressing fire!', 'Eat lead, black-hats!', 'Reloading!'] });
    for (const id of ['bos_stroud', 'lib_bos_castillo', 'lib_bos_albright']) { const a = L.npc(id); if (a && a.follow) { try { a.follow(G.player); } catch (e) { /* */ } } }
  }

  // ------------------------------------------------------------------
  // NPCs on the island
  // ------------------------------------------------------------------
  L.hook('spawn', function () {
    const ending = L.flag('ending_choice');
    if (!L.flag('crane_gone')) {
      const hostile = !!L.flag('crane_fight');
      L.spawnNPC({ id: 'lib_crane', type: 'human', name: 'Colonel Augustus Crane', outfit: 'enclave_power_armor', gender: 'm', faction: 'enclave', ai: 'guard', home: 'here', homeR: 3,
        hostile, essential: !hostile && !ending, level: 18, hp: 650, damage: 30, weapon: ['plasma_rifle', 'laser_rifle'],
        inventory: [['q_note_crane_log', 1], ['lady_libertys_torch', 1], ['ammo_flamer_fuel', 60], ['ammo_mf_cell', 40]],
        at: statueSpot(['commander', 'crane', 'boss'], 0, 16), dialogue: 'dlg_lib_crane',
        barks: ['God bless the Enclave.', 'Steady, soldiers.', 'History is watching.'], voice: 'm_officer' });
      L.spawnNPC({ id: 'lib_rourke', type: 'enclave_soldier', name: 'Sergeant Rourke', outfit: 'enclave_power_armor', faction: 'enclave', ai: 'guard', home: 'here', homeR: 4,
        hostile: faction() !== 'enclave' && !L.flag('crane_surrendered'), level: 14, weapon: ['plasma_rifle', 'laser_rifle'], at: statueSpot(['spawn_in1', 'spawn5', 'spawn1'], 0.25, 18),
        barks: ['Perimeter secure.', 'Eyes on the water!'] });
      L.spawnNPC({ id: 'lib_dunn', type: 'enclave_soldier', name: 'Private Dunn', outfit: 'enclave_power_armor', faction: 'enclave', ai: 'guard', home: 'here', homeR: 4,
        hostile: faction() !== 'enclave' && !L.flag('crane_surrendered'), level: 12, weapon: ['laser_rifle', 'plasma_rifle'], at: statueSpot(['spawn_in2', 'spawn6', 'spawn2'], -0.25, 18),
        barks: ['Yes, sir!', 'Is that the vault dweller?'] });
    }
    if (!L.flag('zeke_home')) {
      // faction 'vault' (not 'player'): a zip-tied captive must not 'fight' the garrison — as a player-faction actor he
      // stayed in combat with every Enclave soldier (even after Crane's surrender), so he could never be talked to
      L.spawnNPC({ id: 'lib_zeke', type: 'human', name: 'Zeke Marlow', outfit: 'vault_suit', gender: 'm', faction: 'vault', ai: 'static', essential: true, hostile: false,
        at: statueSpot(['prisoner', 'torch_display', 'footlocker'], -0.5, 19), dialogue: 'dlg_lib_zeke', hair: 'grey', skinTone: 0.35,
        barks: ['Water pressure on this island is a crime.', 'Kid? That you?'], voice: 'm_old' });
    } else {
      // After the finale Zeke goes home to Vault 120 (next to the Overseer).
      const V = L.vault && L.vault.spots;
      if (V && V.overseer) L.spawnNPC({ id: 'lib_zeke', type: 'human', name: 'Zeke Marlow', outfit: 'vault_suit', gender: 'm', faction: 'vault', ai: 'static', essential: true,
        pos: L.freeSpot(V.overseer.x + 1.5, V.overseer.z + 1.0, V.overseer.y, { maxR: 5 }), dialogue: 'dlg_lib_zeke', hair: 'grey', skinTone: 0.35 });
    }
    if (faction() === 'bos' && Q.isActive('mq06') && (Q.stage('mq06') === 'storm' || Q.stage('mq06') === 'crane' || Q.stage('mq06') === 'zeke' || Q.stage('mq06') === 'torch')) spawnBoSSquad();
  });

  L.hook('world', function () {
    // landmarks_b already places a terminal ('liberty_enclave_command') in the pedestal
    // command room; it IS the torch control. Otherwise we place our own.
    const lb = !G.isStub('landmarks') && G.landmarks.getAnchor ? G.landmarks.getAnchor('statue_of_liberty', 'terminal') : null;
    if (lb && isFinite(lb.x)) L.world.term_liberty_torch = { pos: L.V3(lb.x, (lb.y || 0) + 1.1, lb.z), handle: null, external: true };
    else { const t = torchTerminalSpot(); if (t) L.spawnTerminal('liberty_torch', t, 0); }
  });

  G.events.on('actorKilled', (a) => {
    if (!a) return;
    if (a.id === 'lib_crane') {
      L.setFlag('crane_dead');
      if (Q.stage('mq06') === 'crane' || Q.stage('mq06') === 'storm') Q.setStage('mq06', 'zeke');
      L.news('Reports from the harbour: gunfire on Liberty Island, and then — silence. The Enclave\'s Colonel Crane is dead. I never met the man. I\'m told he was very polite.');
    }
  });

  // Completability net: Crane spawns essential and friendly. Once the player (or the Brotherhood squad at
  // their side) hits him he turns hostile / enters combat, so he can no longer be talked to — and being
  // essential he could never die either. Any such hit before he is resolved opens the fight path instead.
  G.events.on('actorDamaged', (a, dmg, info) => {
    if (!a || a.id !== 'lib_crane' || a.dead || L.flag('crane_fight') || L.flag('crane_surrendered') || L.flag('crane_ally')) return;
    const s = Q.stage('mq06'); if (!Q.isActive('mq06') || (s !== 'storm' && s !== 'crane')) return;
    const src = info && info.source;
    if (src === 'player' || (src && src.faction === 'brotherhood') || (a.hostile && a.hostile())) craneFight();
  });

  // ------------------------------------------------------------------
  // Crane resolution helpers
  // ------------------------------------------------------------------
  function craneFight() {
    L.setFlag('crane_fight');
    const c = L.npc('lib_crane'); if (c) { c.essential = false; }
    L.hostile('lib_crane', true); L.hostile('lib_rourke', true); L.hostile('lib_dunn', true);
    const turncoat = faction() === 'enclave' && !L.flag('enclave_hostile');
    L.setFlag('enclave_hostile');
    // An Enclave recruit turning on the Colonel loses their welcome everywhere;
    // anyone else only has the island garrison to deal with.
    if (turncoat) L.factionTruce('enclave', undefined, undefined, 0, true);
    else L.factionTruce('enclave', LIB().x, LIB().z, 200, true);
    if (Q.isActive('mq06')) {
      if (Q.stage('mq06') !== 'crane') Q.setStage('mq06', 'crane');
      L.refresh('mq06');
    }
  }
  function craneSurrender() {
    L.setFlag('crane_surrendered');
    L.factionTruce('enclave', LIB().x, LIB().z, 220, false, { id: 'liberty_surrender' });
    L.xp(150, 'Colonel Crane surrendered');
    Q.setStage('mq06', 'zeke');
  }

  // ------------------------------------------------------------------
  // The choice
  // ------------------------------------------------------------------
  const hasKey = () => L.has('q_liberty_key');
  const hasCore = () => L.has('q_knickerbocker_core');
  const craneResolved = () => L.flag('crane_dead') || L.flag('crane_surrendered') || L.flag('crane_ally');
  const canChoose = () => hasKey() && hasCore() && craneResolved() && !L.flag('ending_choice') && (Q.stage('mq06') === 'torch' || Q.stage('mq06') === 'zeke');

  function choose(choice) {
    if (L.flag('ending_choice')) return;
    L.setFlag('ending_choice', choice);
    L.take('q_knickerbocker_core', 1);
    if (G.terminals && typeof G.terminals.close === 'function') { try { G.terminals.close(); } catch (e) { /* */ } }
    // The whole torch sequence runs in the 'cutscene' state (the world keeps running,
    // but no LEVEL UP screen, Pip-Boy or pause-worthy menu can pop over it); the ending
    // slideshow takes the same state over and hands 'play' back when it closes.
    if (G.state === 'terminal' || G.state === 'play') G.setState('cutscene');
    L.later(0.3, () => finaleCinematic(choice));
  }
  L.finale = { choose };

  /**
   * The torch sequence. Everything is scheduled in game time through the quest
   * scheduler (pauses with menus, cancelled by a reset/load) and the camera pan is a
   * lib.cinematic (update-driven; the player is always unfrozen at the end).
   */
  function finaleCinematic(choice) {
    const P = G.player, c = LIB();
    const torch = L.V3(c.x - 2, 88, c.z);
    // look up at the torch, hold, then let go
    const y0 = P.yaw, p0 = P.pitch;
    const y1 = Math.atan2(-(c.x - P.pos.x), -(c.z - P.pos.z));
    L.cinematic(7.5, (e, t) => {
      const k = Math.min(1, t * 2.5), ke = k * k * (3 - 2 * k);   // pan in the first 3 s, then hold
      P.yaw = y0 + G.util.wrapAngle(y1 - y0) * ke; P.pitch = p0 + (0.75 - p0) * ke;
    }, () => {
      // The quest is NOT completed here: completing pays 1000 XP, which would level the
      // player up and throw the LEVEL UP screen over the climax. The slideshow plays
      // first; the quest completes (and pays out) when the player is back in the world.
      Q.setStage('mq06', 'done');
      endingNews(choice);
      L.later(2.5, () => startEnding());
    });
    L.sfx('hum_electric', { volume: 1 });
    const col = { purify: '#bfe8ff', selective: '#6dff6a', brotherhood: '#e0c07a', destroy: '#ffb050' }[choice];
    const line = {
      purify: 'The torch ignites — not with fire, but with a cold blue light that rolls out across the harbour like a tide.',
      selective: 'The torch ignites with a sick green light. Across the water, in Times Square, the neon flickers — and somebody screams.',
      brotherhood: 'The array accepts the Codex Override. The torch dims, locks, and goes dark. Behind you, a Paladin says "Ad Victoriam" very quietly.',
      destroy: 'The core begins to whine. Then to scream. You run.',
    }[choice];
    L.later(1.2, () => L.notify(line));
    L.later(2.6, () => {
      if (!G.isStub('fx')) {
        try {
          if (choice === 'destroy') G.fx.explosion(torch, 14, 'normal');
          else G.fx.flash(torch, choice === 'selective' ? 0x55ff66 : choice === 'purify' ? 0x99ddff : 0xffe0a0, 40, 3);
        } catch (e) { /* */ }
      }
      if (choice === 'destroy') L.sfx('explosion_nuke'); else L.sfx(choice === 'selective' ? 'alarm' : 'levelup');
      P.shake(choice === 'destroy' ? 1 : 0.4, 2.5);
      L.overlay(col, 1.6, 0.9, 1.6, 0.92);
      applyWorldConsequences(choice);
    });
    // Crossing the Colonel at the last second: he doesn't get a firefight over the
    // slideshow — he gets a line, a slide, and a one-way Vertibird ride.
    if (L.flag('crane_ally') && choice !== 'selective' && L.alive('lib_crane')) {
      L.later(4.4, () => L.say('lib_crane', 'You — you absolute — I vouched for you. I VOUCHED for you!'));
      L.later(6.2, () => L.say('lib_rourke', 'Colonel, the Vertibird. Now, sir.'));
    }
  }

  function startEnding() {
    // The Colonel's garrison withdraws while the slides play (see the Enclave slide).
    if (L.flag('crane_betrayed') || L.flag('crane_surrendered')) {
      L.setFlag('crane_gone');
      for (const id of ['lib_crane', 'lib_rourke', 'lib_dunn']) L.removeNPC(id);
    }
    if (Q.isActive('mq06')) L.setFlag('mq06_pending_complete');
    settleT = 0;
    Q.playEnding('main');
    // No HUD slideshow to hand 'play' back (quests' own fallback overlay is used).
    if (G.isStub('hud') && G.state === 'cutscene') G.setState('play');
  }
  /**
   * After the slideshow ("Continue Exploring"): the game is back in 'play'. Only then is
   * "Give Me Liberty" completed — quest banner, the 1000 XP and any LEVEL UP screen all
   * arrive in the aftermath instead of interrupting the torch and the epilogue.
   */
  let settleT = 0;
  L.hook('update', function (dt, realDt) {
    if (!L.flag('mq06_pending_complete')) return;
    if (G.state !== 'play' || L.cineActive() || !(dt > 0)) { settleT = 0; return; }
    settleT += realDt;
    if (settleT < 1.2) return;
    L.setFlag('mq06_pending_complete', false);
    if (Q.isActive('mq06')) Q.complete('mq06');
    L.later(3.5, () => L.notify(L.flag('ending_choice') === 'purify' ? 'The sky over Manhattan is clear. The wasteland is still out there — and so are you.'
      : 'The torch is quiet. The wasteland is still out there — and so are you.'));
  });

  // ------------------------------------------------------------------
  // Lasting world consequences (weather). Atmosphere's automatic weather cycle
  // would bring radstorms straight back, so after the ending the quest module
  // drives the weather itself from a consequence-specific pool, and reapplies
  // it after every load.
  // ------------------------------------------------------------------
  const SKY_POOLS = {
    purify: { clear: 0.62, overcast: 0.2, rain: 0.12, fog: 0.06 },           // no radstorms, no green dust — ever
    selective: { radstorm: 0.4, dust: 0.25, overcast: 0.25, fog: 0.1 },     // the Cleaning poisoned the sky for good
  };
  function skyMode() { return L.flag('sky_clean') ? 'purify' : L.flag('the_cleaning') ? 'selective' : null; }
  let wxT = 0;
  function applySky(opts) {
    const mode = skyMode(); if (!mode || G.isStub('atmos') || !G.atmos.setWeather) return;
    opts = opts || {};
    const pool = SKY_POOLS[mode];
    let names = Object.keys(pool);
    try { const have = G.atmos.weatherNames ? G.atmos.weatherNames() : null; if (have && have.length) names = names.filter((n) => have.indexOf(n) >= 0); } catch (e) { /* */ }
    if (!names.length) return;
    let name = opts.name;
    if (!name) {
      let tot = 0; for (const n of names) tot += pool[n];
      let r = Math.random() * tot; name = names[names.length - 1];
      for (const n of names) { r -= pool[n]; if (r <= 0) { name = n; break; } }
    }
    try { G.atmos.setWeather(name, { auto: false, hold: 1e9, duration: opts.duration || 30, instant: !!opts.instant, silent: !!opts.silent }); } catch (e) { /* */ }
    wxT = 240 + Math.random() * 300;   // game seconds until the next change
  }
  L.hook('update', function (dt) {
    if (!(dt > 0) || !skyMode()) return;
    wxT -= dt;
    let cur = null; try { cur = G.atmos.getWeather ? G.atmos.getWeather() : null; } catch (e) { cur = null; }
    const pool = SKY_POOLS[skyMode()];
    if (wxT <= 0 || (cur && pool[cur] === undefined)) applySky({});   // someone set a forbidden weather: correct it
  });
  // Saved in the middle of the torch sequence: finish it (consequences, quest, slides).
  L.hook('loaded', function () {
    const choice = L.flag('ending_choice');
    if (!choice || L.flag('ending_played')) return;
    L.later(1.0, () => {
      applyWorldConsequences(choice);
      if (Q.isActive('mq06')) { Q.setStage('mq06', 'done'); endingNews(choice); }

      // The Cleaning: every ghoul and super mutant in the city dies.
      if (!G.isStub('actors') && G.actors.list) {
        for (const a of G.actors.list().slice()) {
          if (a.dead) continue;
          const doomed = a.ghoul || a.char && a.char.ghoul || ['ghouls', 'feral_ghouls', 'super_mutants'].indexOf(a.faction) >= 0 || a.type === 'feral_ghoul' || a.type === 'glowing_one' || a.type === 'super_mutant' || a.type === 'super_mutant_behemoth';
          if (!doomed) continue;
          if (a.essential) { a.essential = false; }
          try { a.kill(); } catch (e) { /* */ }
        }
      }
    } else if (choice === 'brotherhood') {
      L.karma(25, 'Handed the array to the Brotherhood');
      L.setFlag('enclave_hostile');
    } else if (choice === 'destroy') {
      L.karma(0, 'Destroyed the array');
      G.player.damage(15, { type: 'explosive' });

    // Zeke goes home to the vault after everything (unless the Cleaning broke him).
    if (choice !== 'selective') L.setFlag('zeke_home');
    if (L.flag('crane_ally') && choice !== 'selective' && L.alive('lib_crane')) L.setFlag('crane_betrayed');

' + (f('stroud_will_listen') && choice !== 'brotherhood' ? 'Paladin Commander Stroud never fully forgave the vault dweller for what happened at the torch — but he kept his word, and listened, and in time Scribe Wexler\'s footnotes became the Chapter\'s new Codex.' : 'Paladin Commander Stroud ruled with a steel hand and a closed fist; Manhattan was safer, and quieter, and not entirely free.'), 'thinking');
    if (f('crane_betrayed')) push('COLONEL CRANE', 'Colonel Augustus Crane was last seen climbing into a Vertibird on Liberty Island, still shouting. He never forgave the tenant who shook his hand at the foot of the torch and then chose differently. Enclave radio spent a year calling ' + N + ' a traitor to the Republic. Nobody in Manhattan could remember voting for it.', 'thinking');
    if (choice === 'selective') push('THE ENCLAVE', 'With the city "cleaned," the Enclave declared Manhattan the new capital of a reborn America. Colonel Crane was promoted. Lieutenant Hale resigned her commission and disappeared into the wasteland. They say she still wears her helmet, but only to hide her face.', 'injured');
    else if (f('hale_defects') || f('hale_knows')) push('LIEUTENANT HALE', 'Vesper Hale never went back to the Enclave. She took her helmet off for good and settled in Grand Central, where she taught the market\'s children to read — starting with the fine print.', 'thumbs_up');
    else push('THE ENCLAVE', 'The Enclave\'s eastern command never recovered from the loss of Liberty Island. Their eyebots still drift over the ruins at night, playing marching songs to a country that stopped listening a long time ago.', 'thinking');

    // 10. Radio
    push('RADIO FREE MANHATTAN', 'Vinnie Valentine broadcast for another twenty years, and told the story of the vault dweller from 120 so many times — and so differently — that by the end not even he knew which version was true. ' + (f('mq03_science') ? 'The Rockette Array lit up Radio City every night he was on the air.' : 'ROCKETTE-7 kept dancing. Somebody had to.'), 'thumbs_up');

    // 11. Borough
    if (f('borough_dead')) push('BOROUGH', 'Borough, the dog from Bethesda Terrace, fell defending the one he\'d chosen. There is a small cairn beside his first master\'s grave now. Somebody keeps a bowl of water by it. Somebody keeps it clean.', 'injured');
    else if (f('borough_recruited')) push('BOROUGH', 'Borough never left ' + N + '\'s side. He never learned "heel," either. He grew grey around the muzzle and lazy in the sun, and every mole rat in Central Park lived in terror of him until the end of his days.', 'thumbs_up');

    // 11b. Human companions (36_companions.js): Malone, Tull, Rook
    if (typeof L.companionSlides === 'function') {
      let cs = [];
      try { cs = L.companionSlides(choice, N) || []; } catch (e) { G.reportError('quests:companionSlides', e); }
      for (const c of cs) push(c.title, c.text, c.pose);
    }

    // 12. The Vault Dweller — the epilogue is written for the choice at the torch
    // first, then shaded by karma (never "gave back the sky" after a Cleaning).
    // Same scale as the Pip-Boy's karma title (rpg.karmaTier): Very Good/Good → good,
    // Evil/Very Evil → evil. Quest karma is sized FO3-style (big deeds ±100–400) so
    // a consistently kind or cruel run reliably lands outside "neutral".
    const tier = L.karmaTier();
    const kb = tier <= 1 ? 'good' : tier >= 3 ? 'evil' : 'neutral';
    void karma;
    const EPI = {
      purify: {
        good: 'And ' + N + '? The people of Manhattan tell a hundred stories about the kid in the blue jumpsuit who walked out of Central Park and gave the city back its sky — and in every one of them, the hero is kind. Some of the stories are even true.',
        neutral: 'And ' + N + '? The kid from Vault 120 gave Manhattan back its sky, then went on living the way wastelanders live — some days kind, some days not. The sky did not keep score. The sky was simply, stubbornly blue.',
        evil: 'And ' + N + '? They lit the torch for everyone — even for the people they had robbed and buried on the way there. Manhattan never did decide whether that made them a saint or a hypocrite. In the end, it settled on "complicated," and enjoyed the weather.',
      },
      selective: {
        good: 'And ' + N + '? Whatever good the kid from Vault 120 did before that day, it was weighed against one green hour on Liberty Island, and it was not enough. It was never going to be enough.',
        neutral: 'And ' + N + '? The Enclave put the name on a plaque. Nobody else in Manhattan would say it aloud. The kid from Vault 120 had turned the torch green, and the city remembered nothing else about them.',
        evil: 'And ' + N + '? Mothers in the ruins of Manhattan still frighten their children with the name. The kid from Vault 120, they say, who turned Lady Liberty\'s torch green and watched the Neon go dark.',
      },
      brotherhood: {
        good: 'And ' + N + '? The kid from Vault 120 put the city\'s last miracle into armoured hands and trusted them to use it well. Whether that trust was wisdom or naivety, history has not yet decided. The people they helped along the way remembered only the kindness.',
        neutral: 'And ' + N + '? The kid from Vault 120 chose steel over hope, and Manhattan learned to live with that choice — safer, quieter, and still coughing under a green sky.',
        evil: 'And ' + N + '? The kid from Vault 120 handed the torch to the men in the metal suits and walked away without looking back. Most of Manhattan was glad to see the back of them.',
      },
      destroy: {
        good: 'And ' + N + '? The kid from Vault 120 decided that nobody should hold that much power — not the Enclave, not the Brotherhood, not even themselves. Some called it cowardice. More called it the first honest thing anyone had done with pre-war technology in two hundred years.',
        neutral: 'And ' + N + '? The kid from Vault 120 blew up the future to keep it out of the wrong hands. Maybe, in New York, all hands were the wrong hands. The debate still starts fights in the Oyster Bar.',
        evil: 'And ' + N + '? The kid from Vault 120 blew Lady Liberty\'s torch into the harbour, and Manhattan remembered them the way it remembers every explosion: loudly, and not fondly.',
      },
    };
    let epi = (EPI[choice] || EPI.destroy)[kb];
    const title = L.karmaTitle();
    if (title) epi += (tier === 0 || tier === 4 ? ' Long after the dust settled, the wasteland still had a name for them: ' : ' On the radio, Vinnie Valentine had a name for them: ') + '"' + title + '."';
    push('THE VAULT DWELLER', epi, kb === 'good' && choice !== 'selective' ? 'thumbs_up' : kb === 'evil' || choice === 'selective' ? 'injured' : 'thinking');

    // Closing card: Vinnie's sign-off (an original nod to the franchise's refrain).
    push('', choice === 'selective' ? 'This is Radio Free Manhattan, signing off. The war changed its uniform again tonight. It always finds a new one.'
      : 'This is Radio Free Manhattan, signing off. The bombs changed everything, folks — except us. We\'re still here. Goodnight, New York.', null);
    return S;
  });
  L.ending('default', function () { return [{ title: 'FALLOUT: NEW YORK', text: 'The sirens stopped two hundred years ago. The city never did.' }]; });

  G.quests.debug.finale = (choice) => {
    L.setFlag('mq_faction', L.flag('mq_faction') || 'none');
    ['mq01', 'mq02', 'mq03', 'mq04', 'mq05'].forEach((id) => { if (!Q.isComplete(id)) Q.complete(id); });
    if (!Q.isStarted('mq06')) Q.start('mq06');
    L.give('q_liberty_key', 1, true); L.give('q_knickerbocker_core', 1, true);
    L.setFlag('crane_surrendered'); Q.setStage('mq06', 'torch');
    if (choice) choose(choice);
    return 'ok';
  };
})();
