      onInteract() {
        if (door._locked) {
          if (def.keyId && G.inventory.count && G.inventory.count(def.keyId) > 0) { door._locked = 0; }
          else if (G.lockpick && !G.isStub('lockpick') && G.lockpick.start) { G.lockpick.start(door._locked, () => { door._locked = 0; }); return; }
          else { G.sfx.play('door_locked'); G.notify('This door is locked.'); return; }
        }
        G.sfx.play(def.sound || 'door_open');
        G.fadeTransition(() => {
          G.player.teleport(def.target.x, def.target.y || 0, def.target.z, def.target.yaw || 0);
          if (def.onEnter) def.onEnter();
          G.events.emit('doorTransition', def);
        });
      },
    });
    door._locked = def.locked || 0;
    door.def = def;
    W.doors.push(door);
    return door;
  };
  /** Stable key for a door's lock state in saves (explicit def.id, else its rounded position). */
  function doorKey(d) {
    if (d.def && d.def.id) return String(d.def.id);
    const p = d.pos || (d.def && d.def.pos); if (!p) return null;
    return Math.round(p.x * 10) + ',' + Math.round((p.y || 0) * 10) + ',' + Math.round(p.z * 10);
  }
  /** Doors that were picked / keyed open this playthrough (lock > 0 by definition, now 0). */
  W.unlockedDoorKeys = function () {
    const out = [];
    for (const d of W.doors) if (d.def && d.def.locked > 0 && !d._locked) { const k = doorKey(d); if (k) out.push(k); }
    return out;
  };
  /** Restore every door to its defined lock, then re-open the given keys (save data). */
  W.restoreDoorLocks = function (unlocked) {
    const set = new Set(unlocked || []);
    for (const d of W.doors) { if (!d.def) continue; d._locked = set.has(doorKey(d)) ? 0 : (d.def.locked || 0); }
  };

  // ---- Distance LOD / culling ---------------------------------------
  /** Hide obj when camera farther than maxDist (checked 4×/s). */
  W.addLOD = function (obj, maxDist, center) { const e = { obj, max2: maxDist * maxDist, c: center || null }; W.lods.push(e); return e; };
  let lodTimer = 0;
  const _lp = new THREE.Vector3();
  function updateLODs(dt) {
    lodTimer -= dt; if (lodTimer > 0) return; lodTimer = 0.25;
    const cam = G.camera.position;
    for (const e of W.lods) {
      const p = e.c || e.obj.getWorldPosition(_lp);
      const dx = p.x - cam.x, dz = p.z - cam.z;
      e.obj.visible = dx * dx + dz * dz < e.max2;
    }
  }

  // ---- Radiation sources ---------------------------------------------
  /** {x,y,z, radius, strength (rads/sec at centre)} — player accumulates rads when inside. */
  W.addRadSource = function (src) { W.radSources.push(src); return src; };
  W.removeRadSource = function (src) { const i = W.radSources.indexOf(src); if (i >= 0) W.radSources.splice(i, 1); };
  W.radiationAt = function (p) {
    let r = 0;
    for (const s of W.radSources) {
      const d = Math.hypot(p.x - s.x, (p.y || 0) - (s.y || 0), p.z - s.z);
      if (d < s.radius) r += s.strength * (1 - d / s.radius);
    }
    return r;
  };

})();
