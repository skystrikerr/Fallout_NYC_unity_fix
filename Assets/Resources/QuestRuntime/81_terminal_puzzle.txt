  function shuffle(arr, r) { for (let i = arr.length - 1; i > 0; i--) { const j = Math.floor(r() * (i + 1)); const t = arr[i]; arr[i] = arr[j]; arr[j] = t; } return arr; }

  function makePuzzle(level, seed) {
    const r = G.util.rng(seed || ((Math.random() * 1e9) | 0));
    // FO3 word-length bands: VE 4-5 · E 6-8 · A 9-10 · H 11-12 · VH 13-15
    const band = level <= 12 ? [4, 5] : level <= 37 ? [6, 8] : level <= 62 ? [9, 10] : level <= 87 ? [11, 12] : [13, 15];
    let len = band[0] + Math.floor(r() * (band[1] - band[0] + 1));
    while (len > 4 && (!WORDS[len] || WORDS[len].length < 16)) len--;
    const pool = shuffle(WORDS[len].slice(), r);
    const password = pool[0];
    const others = pool.slice(1).map((w) => ({ w, l: likeness(w, password) }));
    const count = Math.min(others.length + 1, 9 + Math.round(level / 20) + (r() < 0.5 ? 0 : 1)); // total candidates incl. password
    // Deduction needs some near misses, but which ones (and how many) is random so the best
    // clue isn't always handed over. 2+ words sharing ≥1/4 of their letters, the rest random.
    const nearAll = shuffle(others.filter((o) => o.l >= Math.max(1, Math.floor(len / 4))), r);
    const nNear = Math.min(nearAll.length, Math.max(2, Math.round((count - 1) * (0.25 + r() * 0.25))));
    const near = nearAll.slice(0, nNear);
    const rest = shuffle(others.filter((o) => near.indexOf(o) < 0), r);
    const chosen = near.concat(rest).slice(0, count - 1).map((o) => o.w);
    const words = shuffle(chosen.concat([password]), r);
    // Lay out the continuous 408-char stream
    const buf = new Array(TOTAL);
    const BR = '()[]{}<>';
    for (let i = 0; i < TOTAL; i++) buf[i] = r() < 0.09 ? BR[Math.floor(r() * BR.length)] : JUNK[Math.floor(r() * JUNK.length)];
    const wordAt = new Array(TOTAL).fill(-1);
    const placed = [];
    const slots = Math.floor(TOTAL / words.length);
    words.forEach((w, k) => {
      for (let tries = 0; tries < 40; tries++) {
        const base = k * slots;
        const start = base + 1 + Math.floor(r() * Math.max(1, slots - w.length - 2));
        if (start + w.length >= TOTAL) continue;
        let ok = true;
        for (let i = start - 1; i <= start + w.length; i++) if (wordAt[i] !== undefined && wordAt[i] >= 0) { ok = false; break; }
        if (!ok) continue;
        for (let i = 0; i < w.length; i++) { buf[start + i] = w[i]; wordAt[start + i] = placed.length; }
        placed.push({ w, start, removed: false });
        break;
      }
    });
    // Inject bracket pairs inside single rows (no letters between)
    const rowsTotal = ROWS * 2;
    const nPairs = 4 + Math.floor(r() * 3);
    for (let p = 0, guard = 0; p < nPairs && guard < 200; guard++) {
      const row = Math.floor(r() * rowsTotal);
      const rs = row * COLS;
      const a = rs + Math.floor(r() * (COLS - 2));
      const span = 1 + Math.floor(r() * Math.min(5, COLS - 1 - (a - rs)));
      const b = a + span;
      if (b >= rs + COLS) continue;
      let clean = true;
      for (let i = a; i <= b; i++) if (wordAt[i] >= 0) { clean = false; break; }
      if (!clean) continue;
      const t = Math.floor(r() * 4);
      buf[a] = OPEN[t]; buf[b] = CLOSE[t];
      for (let i = a + 1; i < b; i++) if (OPEN.indexOf(buf[i]) >= 0 || CLOSE.indexOf(buf[i]) >= 0) buf[i] = '.';
      p++;
    }
    const addr0 = 0xF000 + Math.floor(r() * 0x9A) * 0x10;
    return { buf, wordAt, words: placed, password, len, attempts: 4, maxAttempts: 4, usedBrackets: new Set(), log: [], addr0, done: false };
  }

  /** Given a stream index, return the selectable span {a,b,kind,word?} (FO3 hover rules). */
  function spanAt(P, i) {
    if (i < 0 || i >= TOTAL) return null;
    const wi = P.wordAt[i];
    if (wi >= 0 && !P.words[wi].removed) { const w = P.words[wi]; return { a: w.start, b: w.start + w.len - 1, kind: 'word', wi }; }
    const ch = P.buf[i];
    const t = OPEN.indexOf(ch);
    if (t >= 0 && !P.usedBrackets.has(i)) {
      const rowEnd = (Math.floor(i / COLS) + 1) * COLS;
      for (let j = i + 1; j < rowEnd; j++) {
        if (P.wordAt[j] >= 0 && !P.words[P.wordAt[j]].removed) break;
        if (P.buf[j] === CLOSE[t]) return { a: i, b: j, kind: 'bracket' };
      }
    }
    return { a: i, b: i, kind: 'char' };
  }
