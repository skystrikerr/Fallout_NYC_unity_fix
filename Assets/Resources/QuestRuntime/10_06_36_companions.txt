/* =====================================================================
 * FALLOUT: NEW YORK — QUESTS — 36: HUMAN COMPANIONS
 *   FO3/NV-style followers, one at a time (Borough the dog is extra):
 *     • "Gentleman" Jim Malone — ghoul prizefighter-turned-mercenary at the
 *       Neon's Rad-Lounge. Hired for caps; walks out on a Very Evil boss.
 *       Personal quest "The Rematch" (Madison Square Garden).
 *     • Ambrose Tull — expelled Brotherhood scribe in Grand Central's
 *       waiting room. Needs Good karma; +Science/+Repair and field repairs.
 *       Personal quest "Chapter and Verse" (the Guggenheim's Holo-Press).
 *     • Rook — raider defector camped under the Washington Square Arch.
 *       Needs Evil karma; +Lockpick/+Sneak. Personal quest "Severance
 *       Package" (her stash at the Woolworth raider den).
 *   Commands (FO3 companion wheel as dialogue): follow / wait, keep distance
 *   near / far, melee / ranged, aggressive / passive, trade items (their own
 *   inventory via the container UI; they wield the best weapon they carry),
 *   part ways. Location + combat + karma-reaction barks, karma-threshold
 *   farewell scenes, and epilogue slides (L.companionSlides for 60_finale).
 *   All state lives in quest flags (comp_<key>) so it rides the quests save.
 * ===================================================================== */

  const G = window.G;
  const L = G.quests.lib;

  const KEYS = ['malone', 'tull', 'rook'];
  const KID_SPOT = { x: -82, z: -5 };          // 34th Street, west of 7th Avenue: the ruins of the Garden
  const now = () => (G.time ? G.time.elapsed : 0);
  const pick = (arr) => arr[Math.floor(Math.random() * arr.length)];
  const tier = () => L.karmaTier();             // 0 Very Good … 4 Very Evil

  // ------------------------------------------------------------------
  // Per-companion state (persistent, inside quest flags)
  // ------------------------------------------------------------------
  function st(k) {
    let s = L.flag('comp_' + k);
    if (!s || typeof s !== 'object') {
      s = { met: false, rec: false, hired: false, wait: null, far: false, melee: false, passive: false, inv: null,
        left: false, dismissed: false, hostile: false, paid: 0, n: 0, lastDay: -1 };
      L.setFlag('comp_' + k, s);
    }
    return s;
  }
  const cfg = {};                      // roster, filled below
  const actor = (k) => L.npc(cfg[k].id);
  /** The human companion currently hired (following or waiting), or null. */
  function current() { for (const k of KEYS) { const s = st(k); if (s.rec && !s.hostile) return k; } return null; }
  function following(k) { const s = st(k); const a = actor(k); return s.rec && !s.wait && a && !a.dead ? a : null; }
  /** Response ids that change after every command, so orders never show as "already said". */
  const cid = (k, name) => name + '#' + (st(k).n || 0);
  const bump = (k) => { const s = st(k); s.n = (s.n || 0) + 1; };

  // ------------------------------------------------------------------
  // Inventory & weapons
  // ------------------------------------------------------------------
  function resolveInv(list) {
    const out = [];
    for (const e of list) { const id = L.item(e[0]); if (id) out.push([id, e[1]]); }
    return out;
  }
  function ensureInv(k) {
    const s = st(k);
    if (!Array.isArray(s.inv)) s.inv = resolveInv(cfg[k].startInv);
    return s.inv;
  }
  function weaponInfo(id) {
    if (!id || id === 'caps' || G.isStub('items') || !G.items.get) return null;
    const d = G.items.get(id);
    if (!d || d.type !== 'weapon') return null;
    let w = d.weapon || null;
    if (!w && !G.isStub('weapons') && G.weapons.getDef) { try { const wd = G.weapons.getDef(id); w = wd && (wd.weapon || wd); } catch (e) { w = null; } }
    if (!w) return null;
    if (w.skill === 'explosives' || /grenade|mine/.test(id)) return null;   // thrown things stay in the pack
    const melee = !!(w.melee || w.skill === 'melee_weapons' || w.skill === 'unarmed' || w.reloadType === 'melee');
    const dps = (w.damage || 5) * Math.max(0.3, Math.min(6, w.rof || 1)) * (w.pellets || 1);
    return { id, melee, score: dps + (d.value || 0) * 0.02 };
  }
  /** FO3: companions use the best weapon they carry, of the preferred kind if they have one. */
  function ensureWeapon(k, force) {
    const a = actor(k); const s = st(k);
    if (!a || a.dead || typeof a.equipWeapon !== 'function') return;
    const ws = ensureInv(k).map((e) => weaponInfo(e[0])).filter(Boolean);
    if (!ws.length) {
      // everything was taken off them: fight with fists (FO3), don't keep swinging a gun they no longer carry
      if (a.weaponId) { try { a.equipWeapon(null); } catch (e) { G.reportError('quests:companionWeapon', e); } }
      return;
    }
    const pref = ws.filter((w) => w.melee === !!s.melee);
    const pool = pref.length ? pref : ws;
    pool.sort((x, y) => y.score - x.score);
    const best = pool[0].id;
    if (force || a.weaponId !== best) { try { a.equipWeapon(best); } catch (e) { G.reportError('quests:companionWeapon', e); } }
  }
  function openInventory(k) {
    const a = actor(k); const s = st(k); const c0 = cfg[k];
    if (!a || a.dead) return;
    if (G.isStub('containers') || !G.containers.create) { L.notify(c0.short + ' shrugs. (No container system.)'); return; }
    const cid2 = 'q_comp_inv_' + k;
    let c = G.containers.get ? G.containers.get(cid2) : null;
    if (!c) c = G.containers.create({ id: cid2, name: c0.short, items: [], model: false, interact: false, owner: null });
    if (!c) return;
    // The quest flags are authoritative: refill from them every time.
    const items = new Map(), conds = new Map();
    for (const e of ensureInv(k)) { if (e[1] > 0) { items.set(e[0], (items.get(e[0]) || 0) + e[1]); if (Array.isArray(e[2]) && e[2].length) conds.set(e[0], e[2].slice()); } }
    c.items = items; c.conds = conds; c.owner = null; c.locked = 0; c.playerStored = true;
    c.onClose = function () {
      const before = a.weaponId;
      s.inv = [];
      for (const [id, n] of c.items) if (n > 0) s.inv.push(c.conds.has(id) ? [id, n, c.conds.get(id).slice()] : [id, n]);
      a.inventory = s.inv.map((e) => [e[0], e[1]]);
      ensureWeapon(k);
      if (a.weaponId && a.weaponId !== before) {
        const wn = (G.items.get(a.weaponId) || {}).name || 'this';
        L.later(0.4, () => bark(k, c0.newWeapon(wn), 4, true));
      }
    };
    if (G.ui && typeof G.ui.openContainer === 'function' && !G.isStub('ui')) G.ui.openContainer(c);
    else if (G.containers.open) G.containers.open(c);
  }

  // ------------------------------------------------------------------
  // Perks (companion bonuses through rpg's effect system)
  // ------------------------------------------------------------------
  const perkOn = {};
  function syncPerk(k) {
    if (G.isStub('rpg') || !G.rpg.addEffect) return;
    const s = st(k); const want = s.rec && !s.hostile;
    const p = cfg[k].perk();
    const sig = want ? p.name + JSON.stringify(p.mods) : '';
    if (perkOn[k] === sig) return;
    try { G.rpg.removeEffect('comp_perk_' + k); } catch (e) { /* */ }
    if (want) {
      try { G.rpg.addEffect({ id: 'comp_perk_' + k, name: p.name, kind: 'buff', duration: 0, mods: p.mods, source: 'companion', desc: p.desc }); } catch (e) { G.reportError('quests:companionPerk', e); }
    }
    perkOn[k] = sig;
  }

  // ------------------------------------------------------------------
  // Commands
  // ------------------------------------------------------------------
  function homeOf(k) {
    const c0 = cfg[k];
    if (!c0._home) c0._home = c0.home() || { x: 0, y: 0, z: 0 };
    return c0._home;
  }
  function setGuard(a, x, z, r) {
    a.aiMode = 'guard';
    a.home = { x, z, r: r || 2 };
    try { a.follow(null); } catch (e) { /* */ }
  }
  function startFollow(k) {
    const a = actor(k); const s = st(k);
    s.wait = null;
    if (!a || a.dead) return;
    a.faction = 'player'; a.essential = true;
    try { if (a.setHostile) a.setHostile(false); } catch (e) { /* */ }
    a.aiMode = 'follow';
    try { a.follow('player'); } catch (e) { G.reportError('quests:follow', e); }
  }
  function recruit(k, how) {
    const s = st(k); const c0 = cfg[k];
    const other = current();
    if (other && other !== k) return false;
    s.rec = true; s.hired = true; s.met = true; s.left = false; s.dismissed = false; s.wait = null; s.how = how || s.how || 'hired';
    s.hiredAt = G.time ? G.time.day : 1;
    startFollow(k);
    ensureWeapon(k, true);
    syncPerk(k);
    setBarks(k);
    bump(k);
    L.notify(c0.short + ' has joined you.');
    L.hint('companions', 'Talk to your companion to give orders: wait or follow, keep your distance, switch weapons, change combat stance, or trade items. Only one hired companion can travel with you at a time.', 1.2);
    return true;
  }
  function doWait(k) {
    const a = actor(k); const s = st(k); bump(k);
    if (!a) return;
    s.wait = { x: +a.pos.x.toFixed(2), y: +a.pos.y.toFixed(2), z: +a.pos.z.toFixed(2) };
    setGuard(a, s.wait.x, s.wait.z, 1.5);
  }
  function doFollow(k) { bump(k); startFollow(k); }
  /** Part ways (dismiss or karma walk-out): they head home and keep their gear. */
  function release(k, why) {
    const a = actor(k); const s = st(k); const h = homeOf(k);
    s.rec = false; s.wait = null; bump(k);
    if (why === 'karma') { s.left = true; s.leftTier = tier(); } else s.dismissed = true;
    if (a && !a.dead) {
      a.faction = cfg[k].faction;
      setGuard(a, h.x, h.z, 2);
      try { if (a.moveTo) a.moveTo(h.x, h.z); } catch (e) { /* */ }
      a._compHomeT = now();
    }
    syncPerk(k);
    setBarks(k);
  }
  function setBarks(k) {
    const a = actor(k); const c0 = cfg[k];
    if (a) a.customBarks = { idle: st(k).rec ? c0.idle : c0.ambient, greet: c0.greet, alert: c0.combat, kill: c0.kill };
  }

  // ------------------------------------------------------------------
  // Barks
  // ------------------------------------------------------------------
  let barkGate = 0;                      // one companion line every few seconds at most
  function bark(k, text, dur, force) {
    if (!text) return false;
    const a = actor(k); if (!a || a.dead) return false;
    if (!force && now() < barkGate) return false;
    if (a.distToPlayer && a.distToPlayer() > 30) return false;
    barkGate = now() + 6;
    try { a.say(text, dur || 4.5); } catch (e) { return false; }
    return true;
  }

  function spawnCompanion(k) {
    const c0 = cfg[k]; const s = st(k);
    const inv = ensureInv(k);
    let at;
    const P = G.player.pos;
    if (s.rec && !s.wait && !s.hostile) at = { fallback: { x: P.x + 1.5, z: P.z + 1.5, y: P.y }, maxR: 6 };
    else if (s.rec && s.wait) at = { fallback: { x: s.wait.x, z: s.wait.z, y: s.wait.y }, maxR: 4 };
    else { const h = homeOf(k); at = { fallback: { x: h.x, z: h.z, y: h.y }, maxR: 3 }; }
    const followingNow = s.rec && !s.wait && !s.hostile;
    const a = L.spawnNPC(Object.assign({}, c0.look, {
      id: c0.id, name: c0.name, faction: s.hostile ? 'raiders' : followingNow || s.rec ? 'player' : c0.faction,
      ai: followingNow ? 'follow' : 'guard', home: 'here', homeR: 2, at, hostile: s.hostile ? true : false,
      essential: !s.hostile, dialogue: s.hostile ? undefined : c0.dlg, inventory: inv.map((e) => [e[0], e[1]]), weapon: 'none', noRespawn: true,
      barks: { idle: s.rec ? c0.idle : c0.ambient, greet: c0.greet, alert: c0.combat, kill: c0.kill },
    }));
    if (!a) return;
    a.inventory = inv.map((e) => [e[0], e[1]]);
    a.essential = !s.hostile;
    ensureWeapon(k, true);
    if (followingNow) startFollow(k);
    else if (s.rec && s.wait) setGuard(a, s.wait.x, s.wait.z, 1.5);
    else if (!s.hostile) { const h = homeOf(k); a.faction = c0.faction; setGuard(a, h.x, h.z, 2); }
    else { a.faction = 'raiders'; try { a.setHostile(true); } catch (e) { /* */ } }
  }

  function statusLine(k) {
    const s = st(k);
    return '[' + (s.far ? 'Keeping distance' : 'Staying close') + '  ·  ' + (s.melee ? 'Melee' : 'Ranged') + '  ·  ' + (s.passive ? 'Passive' : 'Aggressive') + ']';

  function commonNodes(k, extraResponses) {
    const c0 = cfg[k];
    const S = () => st(k);
    return {
      comp: {
        speaker: c0.name,
        text: () => {
          const s = S();
          if (s.wait) return pick(c0.greet) + ' ' + pick(['Still here, like you asked.', 'Waiting, like a good little hireling.', 'Holding position.']);
          return pick(c0.greet) + (c0.compLine ? ' ' + c0.compLine() : '');
        },
        responses: () => [
          { id: cid(k, 'tac'), text: 'Let\'s talk about how you\'re handling things.', next: 'tactics' },
          { id: cid(k, 'wait'), text: 'Wait here.', cond: () => !S().wait, action: () => { doWait(k); L.later(0.3, () => bark(k, pick(c0.wait), 3, true)); } },
          { id: cid(k, 'fol'), text: 'Follow me.', cond: () => !!S().wait, action: () => { doFollow(k); L.later(0.3, () => bark(k, pick(c0.follow), 3, true)); } },
          { id: cid(k, 'trade'), text: 'Let\'s trade some equipment.', action: () => { bump(k); L.later(0.1, () => openInventory(k)); } },
        ].concat(extraResponses ? extraResponses() : []).concat([
          { id: cid(k, 'part'), text: 'I think it\'s time we parted ways.', next: 'part' },
          { id: cid(k, 'bye'), text: 'Never mind.', end: true },
        ]),
      },
      tactics: {
        speaker: c0.name,
        text: () => { const s = S(); const ack = s.ack || 'How do you want it done?'; s.ack = null; return ack + '\n\n' + statusLine(k); },

          { id: cid(k, 'far'), text: 'Keep your distance.', cond: () => !S().far, action: () => { S().far = true; S().ack = c0.ack.far; bump(k); }, next: 'tactics' },

  /** Tull's field repair (once a day): equipped weapon + armour up to 90% condition. */
  function tullRepair() {
    const s = st('tull'); s.lastDay = G.time ? G.time.day : 0;
    if (G.isStub('inventory') || !G.inventory.equipped || !G.inventory.setCondition) return 0;
    let n = 0;
    for (const slot of ['weapon', 'body', 'head']) {
      let id = null; try { id = G.inventory.equipped(slot); } catch (e) { id = null; }
      if (!id) continue;
      let c = 1; try { c = G.inventory.condition(id); } catch (e) { c = 1; }
      if (typeof c === 'number' && c < 0.9) { try { G.inventory.setCondition(id, 0.9); n++; } catch (e) { /* */ } }
    }
    if (n) L.sfx('repair', { volume: 0.6 });
    return n;
  }

  function farProxy(k, a) {
    let p = proxies[k];
    if (!p) p = proxies[k] = { pos: new THREE.Vector3(), dead: false, isProxy: true };
    const P = G.player.pos;
    let dx = a.pos.x - P.x, dz = a.pos.z - P.z; let d = Math.hypot(dx, dz);
    if (d < 0.5) { dx = Math.sin(G.player.yaw); dz = Math.cos(G.player.yaw); d = 1; }
    const want = 7.5;
    p.pos.set(P.x + dx / d * (want - 2.2), P.y, P.z + dz / d * (want - 2.2));
    return p;
  }
  function engage(a, t) {
    if (typeof a.attack === 'function') { try { a.attack(t); return; } catch (e) { /* fall through */ } }
    // Mirror of combat's own enterCombat for a follower (no public API yet — see NOTES).
    try {
      a.target = t; a.inCombat = true; a.setState('combat');
      a.lastSeenT = now(); a.aimT = now() + 0.35 + Math.random() * 0.3;
      a.lastKnown = (a.lastKnown || new THREE.Vector3()).copy(t.pos);
      if (Math.random() < 0.5) a.bark('alert', 1);
    } catch (e) { /* */ }
  }
  function aggressiveScan(k, a) {
    if (a.state !== 'follow' || a.inCombat) return;
    const P = G.player.pos;
    let list = [];
    try { list = G.actors.hostilesNear ? G.actors.hostilesNear(P, 32) : (G.actors.inRadius(P, 32) || []).filter((o) => o.hostile && o.hostile()); } catch (e) { list = []; }
    let best = null, bd = 1e9;
    for (const o of list) {
      if (!o || o.dead || o === a || !o.inScene || o.state === 'dormant' || o.questTruce) continue;
      const d = o.distTo(a.pos); if (d > 34 || d >= bd) continue;
      if (G.world.lineOfSight) {
        const e1 = a.eyePos ? a.eyePos() : new THREE.Vector3(a.pos.x, a.pos.y + 1.6, a.pos.z);
        const e2 = o.centerPos ? o.centerPos() : new THREE.Vector3(o.pos.x, o.pos.y + 1, o.pos.z);
        if (!G.world.lineOfSight(e1, e2)) continue;
      }
      best = o; bd = d;
    }
    if (best) engage(a, best);
  }
  function locationComment(k) {
    const P = G.player.pos;
    if (P.x >= (G.LAYOUT.INTERIOR_X_MIN || 1e9)) return;
    const s = st(k); s.saw = s.saw || {};
    const lines = cfg[k].loc;
    for (const l of LOCS()) {
      const key = l.id === 'bethesda' || l.id === 'belvedere' || l.id === 'reservoir' ? 'central_park' : l.id === 'dumbo' ? 'brooklyn_bridge' : l.id;
      if (!lines[key] || s.saw[key]) continue;
      if (Math.hypot(P.x - l.x, P.z - l.z) > Math.max(22, (l.r || 30) * 0.9)) continue;
      if (bark(k, lines[key], 6)) s.saw[key] = true;
      return;
    }
  }
  /** FO-style catch-up: a follower that has made no progress towards the player for ~5 s
   *  (wedged on geometry the steering can't route around) is moved to a free spot behind the
   *  player, out of view. Never mid-fight. */
  function unstick(k, a, s) {
    const d = a.distToPlayer(); const want = s.far ? 13 : 9;
    if (d < want || a.inCombat || a.state !== 'follow') { s._stuckN = 0; s._stuckD = d; return; }
    if (s._stuckD !== undefined && d > s._stuckD - 1.0) s._stuckN = (s._stuckN || 0) + 1; else s._stuckN = 0;
    s._stuckD = d;
    if (s._stuckN < 3) return;
    s._stuckN = 0;
    const P = G.player; const back = s.far ? 6 : 2.5;
    const sp = L.freeSpot(P.pos.x + Math.sin(P.yaw) * back, P.pos.z + Math.cos(P.yaw) * back, P.pos.y, { maxR: 5 });
    if (sp && !sp.forced) { a.pos.set(sp.x, sp.y, sp.z); if (a.attach) { try { a.attach(); } catch (e) { /* */ } } s._stuckD = a.distToPlayer(); }
  }
  function tickCompanion(k, slow) {
    const s = st(k); const a = actor(k);
    if (!a || a.dead) return;
    // Friendly fire past the warning: they turn on you (FO3).
    if (s.rec && a.hostile && a.hostile() && !s.hostile) {
      s.hostile = true; s.rec = false; a.essential = false; a.dialogue = null; syncPerk(k);
      L.notify(cfg[k].short + ' has turned against you.');
      return;
    }
    if (s.hostile) return;
    if (s.rec && !s.wait) {
      a.faction = 'player'; a.essential = true;
      if (a.aiMode !== 'follow') a.aiMode = 'follow';
      if (!a.knockedOut && a.state !== 'knocked' && (a.state === 'idle' || a.state === 'search' || (a.state === 'move' && !a.moveGoal) || !a.following)) {
        try { a.follow('player'); } catch (e) { /* */ }
      }
      if (a.state === 'follow') {
        if (s.far) a.followTarget = farProxy(k, a);
        else if (a.followTarget && a.followTarget.isProxy) a.followTarget = 'player';
      }
      if (s.passive) a._nextFollowScan = now() + 5;       // combat's follower scan stays quiet: fight only when attacked
      else if (a._nextFollowScan > now() + 1.5) a._nextFollowScan = 0;
      if (!s.passive && aggroT <= 0) aggressiveScan(k, a);
      if (slow) unstick(k, a, s);
      if (slow) {
        if (farewellFor === null && cfg[k].leaves() && G.state === 'play' && !(G.combat.inCombat && G.combat.inCombat()) && !L.cineActive() && a.distToPlayer() < 12) {
          farewellFor = k;
          L.later(0.2, () => { farewellFor = null; if (st(k).rec && cfg[k].leaves() && G.state === 'play') L.startDialogue(cfg[k].dlg, cfg[k].id); });
        }
        const P = G.player;
        if (P.hp < P.maxHp * 0.25 && P.alive !== false && (!s._hurtT || now() - s._hurtT > 60)) { s._hurtT = now(); bark(k, pick(cfg[k].hurt), 3.5, true); }
        else if (!(G.combat.inCombat && G.combat.inCombat())) {
          locationComment(k);
          if (!s._idleT) s._idleT = now() + 90 + Math.random() * 60;
          if (now() > s._idleT) { s._idleT = now() + 120 + Math.random() * 120; bark(k, pick(cfg[k].idle), 5); }
        }
      }
    } else if (s.rec && s.wait) {
      if (a.aiMode !== 'guard' || !a.home || Math.hypot(a.home.x - s.wait.x, a.home.z - s.wait.z) > 1) setGuard(a, s.wait.x, s.wait.z, 1.5);
      if (s.rec && cfg[k].leaves() && slow) { /* they refuse when next spoken to (start node → farewell) */ }
    } else {
      // Not hired: live at home. Walked home from far away? Arrive while the player isn't looking.
      const h = homeOf(k);
      a.faction = cfg[k].faction;
      if (a.aiMode !== 'guard' || !a.home || Math.hypot(a.home.x - h.x, a.home.z - h.z) > 1) setGuard(a, h.x, h.z, 2);
      if (slow && Math.hypot(a.pos.x - h.x, a.pos.z - h.z) > 25 && a.distToPlayer() > 70) {
        a.pos.set(h.x, h.y || 0, h.z); if (a.attach) { try { a.attach(); } catch (e) { /* */ } }
      }
    }
  }
  L.hook('update', function (dt, realDt) {

  });

  // Keep followers with the player across doors, ferries and fast travel.
  function regroup() {
    for (const k of KEYS) {
      const a = following(k); if (!a) continue;
      L.later(0.2, () => {
        if (a.dead || !following(k)) return;
        if (Math.hypot(a.pos.x - G.player.pos.x, a.pos.z - G.player.pos.z) < 20) return;
        try {
          const p = G.player.pos, sp = L.freeSpot(p.x - 1.3, p.z + 1.3, p.y, { maxR: 4 });
          a.pos.set(sp.x, sp.y, sp.z); if (a.attach) a.attach();
          // a knocked-out companion is carried along but left to get up by itself: forcing 'follow' here
          // would skip combat's 'knocked' recovery and leave them at 1 HP / knockedOut forever
          if (!a.knockedOut && a.state !== 'knocked') a.follow('player');
        } catch (e) { /* */ }
      });
    }
  }
