  function subwayCoreSpot() {
    // Prefer map anchors; else next to the subway's inbound door target.
    const tries = [() => G.map.getAnchor && G.map.getAnchor('subway', 'loot'), () => G.map.getAnchor && G.map.getAnchor('subway', 'core'),
      () => G.map.anchors && G.map.anchors.subway && (G.map.anchors.subway.loot || G.map.anchors.subway.car)];
    for (const t of tries) { let a = null; try { a = t(); } catch (e) { a = null; } if (a && isFinite(a.x)) return { x: a.x, y: a.y || 0, z: a.z }; }
    const d = L.doorInto('subway');
    const o = G.LAYOUT.INTERIORS.subway.origin;
    if (d) { const t = d.def.target; return L.freeSpot(t.x + (o.x - t.x) * 0.6, t.z + (o.z - t.z) * 0.6, t.y || 0, { maxR: 12 }); }
    return L.freeSpot(o.x, o.z, o.y || 0, { maxR: 20 });
  }

  function plantIsotope(placebo) {
    if (placebo) {
      L.take('q_blessed_isotope', 1);
      L.take(L.item('nuka_cola_quantum', /quantum/i), 1);
      L.setFlag('atom_placebo_planted');
      L.notify('You empty the isotope into a lead-lined gutter and seat the glowing Nuka-Cola Quantum in the reliquary instead. It looks very holy.');
      L.xp(60, 'Science: Divine intervention');
      L.karma(150, 'Saved Grand Central\'s water');
    } else {
      L.take('q_blessed_isotope', 1);
      L.setFlag('atom_isotope_planted');
      L.notify('The isotope slips into the intake with a soft, green plop. Somewhere below, a thousand taps begin to glow.');
      L.karma(-400, 'Poisoned Grand Central\'s water');
      G.world.addRadSource({ x: 240, y: 1, z: -222, radius: 26, strength: 1.2 });
      L.news('Health bulletin, folks, and it\'s not a good one. The taps at Grand Central Market are running green. Doc Hollis says don\'t drink the water, don\'t bathe in the water, don\'t look at the water. The Children of Atom are calling it a miracle. Vinnie is calling it something I can\'t say on the air.');
    }
    if (Q.stage('sq_atom') === 'plant') Q.setStage('sq_atom', 'return');
  }
  function neonResolve(how) {
    L.setFlag('neon_core_resolved'); L.setFlag('neon_outcome', how);
    if (how === 'crowe' || how === 'repair') {
      L.news('Good news from Times Square! The lights of the Neon are shining brighter than they have in a decade. Mayor Crowe says drinks are on the house tonight. Mayor Crowe has since clarified that drinks are NOT on the house. But the lights are nice.');
      Q.complete('sq_neon');
    } else {
      L.news(how === 'atom' ? 'Times Square went dark last night, friends. The Neon\'s generator gave up the ghost, and word is the Children of Atom are very, very happy about it. Say a prayer for the ghouls in the dark.'
        : 'Times Square went dark last night. Somebody sold the Neon\'s last fusion core to the Brotherhood. The ghouls are lighting candles. Candles, folks. In 2291.');
      Q.complete('sq_neon');
    }
  }
  L.neon = { neonResolve, plantIsotope };

  // Selling the neon core to the Brotherhood / Atom (from their dialogue — see below hooks)
  L.hook('register', function () {
    // Extra response injected into Reyes' and Ezra's trees at runtime: selling the Neon's core.
    const addSell = (dlgId, nodeId, text, how, caps, karma) => {
      const tree = L.DIALOGUES.get(dlgId); if (!tree || !tree.nodes[nodeId]) return;
      const rs = tree.nodes[nodeId].responses;
      if (rs.some((r) => r._neonSell)) return;
      rs.splice(rs.length - 1, 0, { _neonSell: true, text, cond: () => L.has('q_neon_core') && !L.flag('neon_core_resolved'),
        karma, action: () => { L.take('q_neon_core', 1); L.giveCaps(caps); neonResolve(how); } });
    };
    addSell('dlg_bos_reyes', 'hello', 'I have a fusion core to sell. (400 caps)', 'sold', 400, -100);
    addSell('dlg_atom_ezra', 'again', 'I have the Neon\'s fusion core. The ghouls can have the dark. (250 caps)', 'atom', 250, -150);
  });
