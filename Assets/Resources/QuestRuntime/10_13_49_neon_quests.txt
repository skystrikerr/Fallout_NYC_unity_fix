  function ballSpots() {
    const T = N();
    const top = T.has(['ball_top']);
    const ots = T.has(['one_times_square']);
    const base = T.spot(['ball_base'], { near: ['one_times_square', 'entrance_south'], dx: 0, dz: -4 });
    let hi = null, lo = null;
    if (top) { hi = { x: top.x, y: top.y, z: top.z }; const b = T.has(['ball_bottom']); lo = b ? { x: b.x, y: b.y, z: b.z } : { x: top.x, y: top.y - 11, z: top.z }; }
    else if (ots) { hi = { x: ots.x, y: ots.y + 12.5, z: ots.z }; lo = { x: ots.x, y: ots.y + 1.8, z: ots.z }; }
    return { base, hi, lo };
  }
  // (a metre off the anchor: Wexler's T-45 manual can sit on the same NYPL 'loot1' spot, and two items on one
  //  point hide each other from the interaction focus)
  function motorSpot() { const p = L.anchor('nypl', ['loot2', 'loot1', 'loot', 'stacks', 'boss', 'entrance'], { x: 70, z: -160 }); return p ? L.freeSpot(p.x + 1.1, p.z + 0.4, p.y, { maxR: 3, radius: 0.25 }) : p; }
  const BALL = { seq: null, gatherOn: false, sprite: null, lastDay: -1, mesh: null };
  /**
   * Stopgap Ball (used only while landmarks_a has no times_square.setBallDrop): a quests-owned faceted
   * crystal ball — red / blue / white / gold panels, unlit so it blooms — that rides the mast down.
   * It is built when a drop starts and disposed when the fireworks end (1 draw call, 19 s a night).
   */
  function ballMesh() {
    if (BALL.mesh) return BALL.mesh;
    const geo = new THREE.IcosahedronGeometry(1.85, 2).toNonIndexed();
    const n = geo.attributes.position.count, col = new Float32Array(n * 3);
    const PAL = [[1, 1, 1], [1, 0.32, 0.3], [0.38, 0.62, 1], [1, 0.84, 0.36], [0.86, 0.94, 1], [1, 1, 1]];
    for (let f = 0; f < n / 3; f++) {
      const c = PAL[(f * 7 + (f >> 2)) % PAL.length], k = 0.62 + ((f * 13) % 7) / 16;
      for (let v = 0; v < 3; v++) { const i = (f * 3 + v) * 3; col[i] = c[0] * k; col[i + 1] = c[1] * k; col[i + 2] = c[2] * k; }
    }
    geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
    const m = new THREE.Mesh(geo, new THREE.MeshBasicMaterial({ vertexColors: true }));
    m.name = 'neon_ball_stopgap'; m.castShadow = false; m.receiveShadow = false;
    G.scene.add(m);
    BALL.mesh = m;
    return m;
  }
  function dropBallMesh() {
    const m = BALL.mesh; BALL.mesh = null;
    if (!m) return;
    try { if (m.parent) m.parent.remove(m); m.geometry.dispose(); m.material.dispose(); } catch (e) { /* ignore */ }
  }
  function ballFixed() { return !!L.flag('neon_ball_fixed'); }
  function landmarkBall(t) {
    try { const ts = !G.isStub('landmarks') && G.landmarks.times_square; if (ts && typeof ts.setBallDrop === 'function') { ts.setBallDrop(t); return true; } } catch (e) { /* ignore */ }
    return false;
  }
  function crowdNear(r) { return N().recs().filter((rec) => rec.a && !rec.a.dead && rec.a.inScene && !rec.def.guard && G.player && Math.hypot(rec.a.pos.x - G.player.pos.x, rec.a.pos.z - G.player.pos.z) < (r || 45)); }
  function shout(text, n) { const c = crowdNear(50); for (let i = 0; i < Math.min(n || 3, c.length); i++) { const a = pick(c).a; if (!a.talking) a.say(text, 1.1); } }
  function firework(sp) {
    const col = pick([0xff5050, 0x50a0ff, 0xffe070, 0x70ff90, 0xff70e0, 0xffffff]);
    const p = new THREE.Vector3(sp.x + (Math.random() - 0.5) * 40, sp.y + 18 + Math.random() * 22, sp.z + (Math.random() - 0.5) * 40);
    if (!G.isStub('fx')) {
      try { if (G.fx.flash) G.fx.flash(p, col, 2.5, 0.5); } catch (e) { /* ignore */ }
      try { if (G.fx.sparks) for (let k = 0; k < 3; k++) G.fx.sparks(p.clone().add(new THREE.Vector3((Math.random() - 0.5) * 3, (Math.random() - 0.5) * 3, (Math.random() - 0.5) * 3))); } catch (e) { /* ignore */ }
    }
    L.sfx('explosion', { pos: p, volume: 0.22, pitch: 1.9 + Math.random() * 0.5 });
  }
  function startDrop() {
    const S = ballSpots();
    if (!S.hi) return;
    BALL.seq = { t: 0, n: 10, S, fireT: 0 };
    shout('Here it comes!', 2);
    L.notify('The Times Square Ball is dropping!');
  }
  function tickDrop(dt) {
    const q = BALL.seq; if (!q) return;
    q.t += dt;
    const S = q.S;
    const k = Math.min(1, q.t / 10);
    const y = S.hi.y + (S.lo.y - S.hi.y) * k;
    const pos = new THREE.Vector3(S.hi.x + (S.lo.x - S.hi.x) * k, y, S.hi.z + (S.lo.z - S.hi.z) * k);
    if (!landmarkBall(k)) {
      // stopgap: our own crystal Ball rides the mast down (the landmark's ball mesh is static)
      const m = ballMesh();
      m.position.copy(pos); m.rotation.y += dt * 0.6;
      const pulse = 1 + Math.sin(q.t * 9) * 0.05 * (1 - k); m.scale.setScalar(pulse);
      if (!BALL.sprite && !G.isStub('fx') && G.fx.glowSprite) { try { BALL.sprite = G.fx.glowSprite(pos.clone(), 0xfff2d0, 6); } catch (e) { BALL.sprite = null; } }
      if (BALL.sprite && BALL.sprite.position) BALL.sprite.position.copy(pos);
      if (k < 1 && Math.random() < dt * 5 && !G.isStub('fx') && G.fx.sparks) try { G.fx.sparks(pos.clone().add(new THREE.Vector3(0, 1.9, 0))); } catch (e) { /* ignore */ }
    }
    // countdown: one number a second
    const n = 10 - Math.floor(q.t);
    if (n < q.n && n >= 1) { q.n = n; shout(String(n) + '!', 4); L.sfx('ui_click', { volume: 0.4 }); }
    if (q.t >= 10 && !q.cheered) {
      q.cheered = true;
      shout(pick(['HAPPY NEW YEAR!', 'IT DROPPED! IT ACTUALLY DROPPED!', 'Happy New Year, Neon!', 'Two hundred years! TWO HUNDRED YEARS!']), 8);
      if (!G.isStub('fx') && G.fx.flash) try { G.fx.flash(pos, 0xffffff, 4, 1.2); } catch (e) { /* ignore */ }
      if (G.player && Math.hypot(G.player.pos.x - S.lo.x, G.player.pos.z - S.lo.z) < 140 && !G.isStub('rpg') && G.rpg.addEffect) {
        try { G.rpg.addEffect({ id: 'neon_new_year', name: 'Neon New Year', kind: 'buff', duration: 1200, mods: { special: { C: 1, L: 1 } }, desc: 'CHA +1, LCK +1' }); } catch (e) { /* ignore */ }
        L.notify('Neon New Year: CHA +1, LCK +1.');
      }
      L.incFlag('neon_ball_drops');
      if (Q.stage('sq_neon_ball') === 'midnight') L.later(3, () => { Q.complete('sq_neon_ball'); L.news('The Times Square Ball dropped at midnight for the first time since 2077. Ghouls, smoothskins and at least one brahmin were seen weeping in the streets of the Neon.'); });
    }
    if (q.t >= 10) { q.fireT -= dt; if (q.fireT <= 0) { q.fireT = 0.35 + Math.random() * 0.5; firework(S.lo); } }
    if (q.t > 19) {
      BALL.seq = null;
      if (BALL.sprite) { try { if (BALL.sprite.parent) BALL.sprite.parent.remove(BALL.sprite); } catch (e) { /* ignore */ } BALL.sprite = null; }
      dropBallMesh();
      landmarkBall(0);   // back to the top for tomorrow night
    }
  }
  function gatherSpotFor(rec) {
    if (!rec._gather) {
      const S = ballSpots();
      const c = N().has(['plaza_centre', 'screen_view', 'center']) || { x: TS.x, y: 0, z: TS.z };
      const ref = S.base || c;
      const ang = Math.random() * Math.PI * 2, r = 4 + Math.random() * 9;
      const s = L.freeSpot((ref.x + c.x) / 2 + Math.cos(ang) * r, (ref.z + c.z) / 2 + Math.sin(ang) * r, ref.y || 0, { maxR: 6 });
      const face = S.hi ? Math.atan2(-(S.hi.x - s.x), -(S.hi.z - s.z)) : 0;
      rec._gather = { x: s.x, y: s.y, z: s.z, yaw: face };
    }
    return rec._gather;
  }
  function ballTick(dt) {
    if (!ballFixed()) return;
    const h = G.time.hour;
    const want = (h >= 23.6 || h < 0.25) && nearNeon(220);
    if (want !== BALL.gatherOn) {
      BALL.gatherOn = want;
      N().setGather(want ? { who: (rec) => !rec.def.guard && ['folk', 'hawker', 'singer', 'merchant', 'director', 'dealer'].indexOf(rec.def.role) >= 0 && !(rec.a.pos.x >= IXM()), at: gatherSpotFor } : null);
      if (!want) for (const r of N().recs()) delete r._gather;
    }
    // the drop: 23:55 game time (10 real seconds ≈ the last 5 game minutes)
    const midnightDay = h >= 12 ? G.time.day + 1 : G.time.day;
    if (!BALL.seq && h >= 23.915 && L.flag('neon_ball_last') !== midnightDay && nearNeon(170)) { L.setFlag('neon_ball_last', midnightDay); startDrop(); }
  }
  function installBall(how) {
    L.setFlag('neon_ball_fixed', how);
    L.take('q_neon_winch_motor', 1); L.take('q_neon_cable', 1);
    Q.setStage('sq_neon_ball', 'midnight');
    L.sfx('terminal_success');
  }

      pos: { x: p.x, y: p.y, z: p.z }, home: { x: p.x, z: p.z, r: 3 }, level: 5, weapon: 'switchblade', dialogue: 'dlg_neon_kip', voice: 'settler',

  function raidOrigin() {
    const gw = N().has(['gate_w']) || N().has(['guard3']) || { x: -92, y: 0, z: -280 };

  function spawnRaid() {
    if (RAID.spawned) return;
    RAID.spawned = true; RAID.actors = [];
    const o = raidOrigin(); if (!o) return;
    const gw = N().has(['gate_w']) || N().has(['guard3']) || { x: -92, z: -280 };
    const c = N().has(['plaza_centre', 'center']) || TS;
    const lvl = Math.max(4, L.level() + 1);
    const path = [{ x: gw.x, z: gw.z }, { x: c.x, z: c.z }];
    const W = ['assault_rifle', 'hunting_rifle', 'sawed_off_shotgun', 'smg_10mm', 'pipe_rifle', 'pistol_10mm', 'combat_shotgun'];
    for (let i = 0; i < 7; i++) {
      const boss = i === 0;
      const a = L.spawnNPC({ id: 'neon_raid' + i, type: 'raider', faction: 'raiders', name: boss ? 'Slag' : 'Terminal Boy', level: boss ? lvl + 3 : lvl, hostile: true,
        pos: { x: o.x + (Math.random() - 0.5) * 8, y: o.y, z: o.z + (Math.random() - 0.5) * 8 }, ai: 'patrol', patrol: path, weapon: boss ? 'combat_shotgun' : W[i % W.length],
        voice: 'raider', noRespawn: true, hp: boss ? 260 : undefined, inventory: boss ? [['caps', 120], ['stimpak', 2]] : [['caps', 12]],
        barks: boss ? { alert: ['Lights out, deadheads!', 'The Neon\'s OURS tonight!'], combat: ['Burn it! Burn it all!', 'Where\'s my lights, Kip?!'] } : undefined });
      if (a) { a.safeExempt = true; a.questRaid = true; RAID.actors.push(a); }
    }
    const capt = L.npc('neon_capt');
    if (capt && !capt.dead) capt.say('Here they come! West gate — everybody on the wall!', 4);
    L.notify('Raiders are storming the west gate!');
    // lights out, if the saboteur's cut was never spliced
    if (!L.flag('power_spliced')) blackoutStart();
  }
  /**
   * The black-out. landmarks_a's times_square.setPower(k) dims the square's neon when it exists; until then
   * (and on top of it) quests stages the failure itself: transformers blow in a chain across the billboard
   * walls (flash-pool pops + spark showers), the cut feed at the breaker cabinet keeps arcing, the Oracle's
   * and the notice board's lamps die, and the crowd panics. Power comes back when the raid is beaten.
   */
  const BLACK = { on: false, t: 0, pops: [], popT: 0, arcT: 0 };
  function landmarkPower(k) {
    try { const ts = !G.isStub('landmarks') && G.landmarks.times_square; if (ts && typeof ts.setPower === 'function') { ts.setPower(k); return true; } } catch (e) { /* ignore */ }
    return false;
  }
  function billboardPoints() {
    const c = N().has(['plaza_centre', 'center']) || { x: TS.x, y: 0, z: TS.z };
    const out = [];
    for (let i = 0; i < 10; i++) { const a = i * 2.39996 + 0.4, r = 30 + (i % 3) * 7; out.push(new THREE.Vector3(c.x + Math.cos(a) * r, 11 + (i * 7) % 14, c.z + Math.sin(a) * r)); }
    return out;
  }
  function servicesLamps(on) { try { const S = L.neonServices; if (S && typeof S.lamps === 'function') S.lamps(on); } catch (e) { /* ignore */ } }
  function blackoutStart() {
    if (BLACK.on) return;
    BLACK.on = true; BLACK.t = 0; BLACK.popT = 0; BLACK.arcT = 0; BLACK.pops = billboardPoints();
    BLACK.landmark = landmarkPower(0.15);
    servicesLamps(false);
    L.sfx('explosion', { volume: 0.35, pitch: 0.45 });
    L.notify('The lights of the Neon go out.');
    L.later(0.8, () => shout(pick(['The lights! THE LIGHTS!', 'Who turned off the Neon?!', 'Stay in the square! Stay together!']), 5));
  }
  function blackoutEnd(silent) {
    if (!BLACK.on) return;
    BLACK.on = false;
    landmarkPower(1);
    servicesLamps(true);
    if (!silent) { L.sfx('terminal_success', { volume: 0.5 }); L.later(0.5, () => shout(pick(['Lights! We got lights!', 'THERE she is!', 'The Neon\'s back!']), 5)); }
  }
  function blackoutTick(dt) {
    if (!BLACK.on || !nearNeon(220) || G.isStub('fx')) return;
    BLACK.t += dt;
    // the first seconds: a chain of blown transformers across the billboards
    if (BLACK.pops.length && BLACK.t > 0.3) {
      BLACK.popT -= dt;
      if (BLACK.popT <= 0) {
        BLACK.popT = 0.28 + Math.random() * 0.3;
        const p = BLACK.pops.shift();
        try { if (G.fx.flash) G.fx.flash(p, 0xbfe4ff, 3.2, 0.18); } catch (e) { /* ignore */ }
        try { if (G.fx.sparks) { G.fx.sparks(p); G.fx.sparks(p.clone().add(new THREE.Vector3(0, -1.2, 0))); } } catch (e) { /* ignore */ }
        L.sfx('explosion', { pos: p, volume: 0.18, pitch: 2.2 + Math.random() * 0.4 });
      }
    }
    // then: the cut feed arcs every few seconds
    BLACK.arcT -= dt;
    if (BLACK.arcT <= 0) {
      BLACK.arcT = 1.5 + Math.random() * 2.5;
      const pp = powerPanelSpot();
      if (pp) {
        const p = new THREE.Vector3(pp.x, (pp.y || 0) + 1.9, pp.z);
        try { if (G.fx.sparks) G.fx.sparks(p); } catch (e) { /* ignore */ }
        if (Math.random() < 0.5) try { if (G.fx.flash) G.fx.flash(p, 0xa8d8ff, 1.6, 0.1); } catch (e) { /* ignore */ }
      }
    }
  }
  /** Stand-in breaker cabinet + conduit riser (only while no power_generator / power_panel anchor is built). */
  function buildCabinet(p) {
    const parts = [
      [new THREE.BoxGeometry(1.3, 1.9, 0.55), 0x4a5048, 0, 0.95, 0], [new THREE.BoxGeometry(1.36, 0.08, 0.62), 0x2e322c, 0, 1.94, 0],
      [new THREE.BoxGeometry(1.1, 1.5, 0.03), 0x5d665a, 0, 1.0, 0.29], [new THREE.BoxGeometry(0.16, 0.16, 0.05), 0xd8c070, -0.3, 1.5, 0.31],
      [new THREE.BoxGeometry(0.16, 0.16, 0.05), 0xd8c070, 0.3, 1.5, 0.31], [new THREE.BoxGeometry(0.7, 0.05, 0.05), 0x1c1c1c, 0, 1.2, 0.31],
      [new THREE.BoxGeometry(0.7, 0.05, 0.05), 0x1c1c1c, 0, 1.0, 0.31], [new THREE.BoxGeometry(0.7, 0.05, 0.05), 0x1c1c1c, 0, 0.8, 0.31],
      [new THREE.BoxGeometry(0.34, 0.22, 0.02), 0xc8b020, 0, 0.45, 0.31], [new THREE.BoxGeometry(0.12, 9.5, 0.12), 0x2a2a26, -0.45, 6.7, -0.1],
      [new THREE.BoxGeometry(0.09, 9.5, 0.09), 0x1a1a18, 0.45, 6.7, -0.1], [new THREE.BoxGeometry(0.5, 0.35, 0.3), 0x3a3e38, 0, 2.15, 0],
    ];
    const geos = parts.map(([g, c, x, y, z]) => {

  const EXP = { run: null, driver: null, brahmin: null, ambushed: false, done: false, waitT: 0, pts: [], idx: 0, doneT: 0 };
  function expressRoute(dest) {
    const start = N().spot(['caravan_arrive', 'caravan_park_1', 'depot_door'], { near: ['entrance_south', 'subway'], dx: 0, dz: 6 });
    const loc = G.LAYOUT.getLocation(dest);
    if (!start || !loc) return null;
    const Ly = G.LAYOUT;
    // L-shaped street route: along the avenue to the destination's street, then along the street
    const zS = loc.z, xA = -40;
    const pts = [{ x: start.x, z: start.z }];
    const corner1 = { x: xA, z: start.z }, corner2 = { x: xA, z: zS }, end = { x: loc.x, z: loc.z };
    const onRoad = (p) => Ly.isRoad(p.x, p.z);
    if (onRoad(corner1)) pts.push(corner1);
    pts.push(onRoad(corner2) ? corner2 : { x: xA, z: zS });
    // pull the end point a little out of the settlement footprint onto the street
    const dx = end.x - corner2.x, dz = end.z - corner2.z, d = Math.hypot(dx, dz) || 1;
    const stop = L.freeSpot(end.x - dx / d * Math.min(12, d * 0.3), end.z - dz / d * Math.min(12, d * 0.3), 0, { maxR: 12 });
    pts.push({ x: stop.x, z: stop.z });
    return { start, pts };
  }
  /**
   * resume = the saved progress {q, idx, x, z, amb, n} (flag neon_exp_state, written every 2 s while
   * the wagon rolls): after a load the driver + brahmin reappear where the save left them, on the
   * same leg of the route, instead of restarting from the Depot.
   */
  function startExpress(run, resume) {
    const r = expressRoute(run.dest); if (!r) { L.notify('The wagon isn\'t ready. Try again later.'); return false; }
    if (!resume) L.incFlag('neon_exp_n');
    EXP.run = run; EXP.pts = r.pts; EXP.idx = 1; EXP.ambushed = false; EXP.done = false; EXP.waitT = 0; EXP.doneT = 0; EXP.saveT = 0;
    EXP.driverId = 'neon_exp_driver' + (resume && resume.n !== undefined ? resume.n : (Number(L.flag('neon_exp_n')) || 0));
    let at = r.start;
    if (resume) {
      EXP.idx = Math.max(1, Math.min(r.pts.length - 1, resume.idx || 1));
      EXP.ambushed = !!resume.amb;
      const f = L.freeSpot(resume.x, resume.z, 0, { maxR: 10 });
      if (f && !f.forced) at = f;
    }
    EXP.driver = L.spawnNPC({ id: EXP.driverId, type: 'human', faction: 'settlers', name: run.driver, outfit: 'merchant', gender: run.q === 'sq_neon_express2' ? 'f' : 'm', hat: 'cowboy', hatColor: 0x4a3420,
      pos: { x: at.x, y: at.y || 0, z: at.z }, ai: 'patrol', patrol: [EXP.pts[EXP.idx]], hp: 220, level: 6, weapon: 'pistol_10mm', voice: 'merchant', noRespawn: true,
      barks: { idle: ['Steady, girl. Steady.', 'Keep your eyes on the rooftops.', 'Nice and slow.'], combat: ['Protect the wagon!', 'Ambush! AMBUSH!'] } });
    const bp = L.freeSpot(at.x + 2, at.z + 2, at.y, { maxR: 5 });
    try { if (!G.isStub('ambient') && G.ambient.brahmin && EXP.driver) EXP.brahmin = G.ambient.brahmin({ pos: { x: bp.x, y: bp.y, z: bp.z }, variant: 1, pack: true, ai: 'follow', follow: EXP.driver, name: 'Neon Express Brahmin' }); } catch (e) { EXP.brahmin = null; }
    if (EXP.driver) EXP.driver.say(resume ? 'There you are, partner. ' + run.name + ' ain\'t getting any closer by itself.' : 'Alright, partner. ' + run.name + ', nice and easy. Stay close.', 4);
    if (EXP.driver) saveExpress();
    return !!EXP.driver;
  }
  function saveExpress() {
    const d = EXP.driver; if (!EXP.run || !d || !d.pos) return;
    L.setFlag('neon_exp_state', { q: EXP.run.q, idx: EXP.idx, x: +d.pos.x.toFixed(1), z: +d.pos.z.toFixed(1), amb: EXP.ambushed, n: Number(EXP.driverId.replace(/\D+/g, '')) || 0 });
  }
  function ambush() {
    if (EXP.ambushed || !EXP.driver) return;
    EXP.ambushed = true;
    const d = EXP.driver, nx = EXP.pts[Math.min(EXP.idx, EXP.pts.length - 1)];
    const dx = nx.x - d.pos.x, dz = nx.z - d.pos.z, dl = Math.hypot(dx, dz) || 1;
    const ax = d.pos.x + dx / dl * 28, az = d.pos.z + dz / dl * 28;
    const lvl = Math.max(3, L.level());
    const n = 3 + (EXP.run.q === 'sq_neon_express3' ? 2 : EXP.run.q === 'sq_neon_express2' ? 1 : 0);
    for (let i = 0; i < n; i++) {
      const s = L.freeSpot(ax + (Math.random() - 0.5) * 10, az + (Math.random() - 0.5) * 10, 0, { maxR: 8 });
      L.spawnNPC({ id: 'neon_exp_raider' + RUNS.indexOf(EXP.run) + '_' + (Number(L.flag('neon_exp_n')) || 0) + '_' + i, type: 'raider', faction: 'raiders', level: lvl, hostile: true,
        pos: { x: s.x, y: s.y, z: s.z }, ai: 'patrol', patrol: [{ x: d.pos.x, z: d.pos.z }, { x: ax, z: az }], noRespawn: true, voice: 'raider' });
    }
    d.say('Raiders! Protect the wagon!', 3);
  }
  function expressTick(dt) {
    const run = EXP.run; if (!run) return;
    const d = EXP.driver;
    if (!Q.isActive(run.q)) { cleanupExpress(); return; }
    if (!d || d.removed) { cleanupExpress(); return; }
    if (d.dead) { Q.fail(run.q); L.notify('The Neon Express driver is dead. Contract failed.'); cleanupExpress(); return; }
    EXP.saveT = (EXP.saveT || 0) + dt; if (EXP.saveT > 2 && !EXP.done) { EXP.saveT = 0; saveExpress(); }
    if (EXP.done) { EXP.doneT += dt; if (EXP.doneT > 40 || (G.player && Math.hypot(G.player.pos.x - d.pos.x, G.player.pos.z - d.pos.z) > 90)) cleanupExpress(); return; }
    const P = G.player.pos;
    const far = Math.hypot(P.x - d.pos.x, P.z - d.pos.z);
    const tgt = EXP.pts[EXP.idx];
    if (d.inCombat || d.state === 'combat' || d.state === 'flee' || d.state === 'cower') return;
    // wait for the escort
    if (far > 35) { if (d.aiMode !== 'static') { d.aiMode = 'static'; d.say(pick(['Hey! Where\'d you go? I\'m not moving without you.', 'Partner? PARTNER?']), 3); } return; }
    if (d.aiMode !== 'patrol' || !d.patrol || d.patrol[0] !== tgt) { d.aiMode = 'patrol'; d.patrol = [tgt]; d.patrolIdx = 0; }
    if (Math.hypot(d.pos.x - tgt.x, d.pos.z - tgt.z) < 3.5) {
      if (EXP.idx < EXP.pts.length - 1) EXP.idx++;
      else {
        EXP.done = true; d.aiMode = 'static';
        d.say('We made it! ' + run.name + '! Eustace\'ll have your caps — here, take your cut now.', 5);
        Q.complete(run.q);
      }
    }
    // ambush at about half way
    if (!EXP.ambushed && EXP.idx >= Math.max(1, Math.floor(EXP.pts.length / 2)) && Math.random() < dt * 0.25) ambush();
  }
  function cleanupExpress() {
    if (EXP.driverId) L.removeNPC(EXP.driverId);
    if (EXP.brahmin && !EXP.brahmin.removed && !G.isStub('actors')) try { G.actors.remove(EXP.brahmin); } catch (e) { /* ignore */ }
    EXP.run = null; EXP.driver = null; EXP.brahmin = null;
    L.setFlag('neon_exp_state', null);
  }

  });

  G.events.on('questUpdated', (q) => { if (q && q.id === 'sq_neon_power' && Q.stage('sq_neon_power') === 'culprit') spawnKip(); });
  G.events.on('itemAdded', (id) => {
    if (id === 'q_neon_winch_motor' || id === 'q_neon_cable') { if (!Q.isStarted('sq_neon_ball') && id === 'q_neon_winch_motor') Q.start('sq_neon_ball'); ballPartsCheck(); }
    if (id === 'q_neon_pay_note' && Q.stage('sq_neon_power') === 'culprit') kipKnown('note');
  });
  G.events.on('actorKilled', (a) => {
    if (!a) return;
    const b = BOUNTIES.find((x) => x.id === a.id);
    if (b && Q.stage('sq_neon_guard') === b.stage) { Q.setStage('sq_neon_guard', 'r' + (BOUNTIES.indexOf(b) + 1)); L.notify(b.name + ' is dead. Report to Captain Kincaid.'); }
    // Sprocket is the only source of the Ball's cable: if she dies first, the spool is on her (lootable)
    if (a.id === 'neon_sprocket' && !Q.isComplete('sq_neon_ball') && !L.flag('neon_ball_fixed') && !L.has('q_neon_cable') && Array.isArray(a.inventory)) a.inventory.push(['q_neon_cable', 1]);
    if (a.id === 'neon_smiley' && Q.isActive('sq_neon_club') && Q.stage('sq_neon_club') === 'meet') { L.setFlag('club_card_only'); Q.complete('sq_neon_club'); }
    if (a.id === 'neon_kip' && (Q.stage('sq_neon_power') === 'confront' || Q.stage('sq_neon_power') === 'culprit')) { L.setFlag('kip_gone'); kipResolved('dead'); if (Q.stage('sq_neon_power') === 'culprit') Q.setStage('sq_neon_power', 'defend'); }
  });

  let t4 = 0;
  L.hook('update', function (dt) {
    if (dt <= 0) return;
    if (BALL.seq) { try { tickDrop(dt); } catch (e) { G.reportError('quests:neon ball', e); BALL.seq = null; } }
    try { expressTick(dt); } catch (e) { G.reportError('quests:neon express', e); cleanupExpress(); }
    t4 += dt; if (t4 < 0.5) return; t4 = 0;
    try { ballTick(); } catch (e) { G.reportError('quests:neon ballTick', e); }
    try { raidTick(); } catch (e) { G.reportError('quests:neon raid', e); }
    try { blackoutTick(0.5); } catch (e) { G.reportError('quests:neon blackout', e); BLACK.on = false; }
    try { showTick(); } catch (e) { G.reportError('quests:neon show', e); }
  });

  // Debug (dev console)
  Q.debug.neonBall = () => { L.setFlag('neon_ball_fixed', 'debug'); G.time.hour = 23.9; };
  Q.debug.neonRaid = () => { if (!Q.isStarted('sq_neon_power')) Q.start('sq_neon_power'); Q.setStage('sq_neon_power', 'defend'); G.time.hour = 1.97; };
  L.neonQuests = { startDrop, spawnRaid, spawnBounty, BOUNTIES, RUNS, startExpress, EXP, clubAdmit, ballSpots };
})();
